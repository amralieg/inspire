def inspire_normalize_identifier(identifier: str) -> str:
    """
    Strip backticks from an identifier.
    
    Args:
        identifier: A SQL identifier that may or may not have backticks
        
    Returns:
        The identifier without backticks
        
    Examples:
        inspire_normalize_identifier("`my-schema`") -> "my-schema"
        inspire_normalize_identifier("my_table") -> "my_table"
    """
    if identifier is None:
        return ""
    return identifier.strip().strip('`')

def inspire_quote_identifier(identifier: str) -> str:
    """
    Add backticks to an identifier for safe SQL usage.
    First normalizes (strips existing backticks) then adds fresh backticks.
    
    Args:
        identifier: A SQL identifier (normalized or not)
        
    Returns:
        The identifier wrapped in backticks
        
    Examples:
        inspire_quote_identifier("my-schema") -> "`my-schema`"
        inspire_quote_identifier("`my-schema`") -> "`my-schema`" (not double-quoted)
    """
    if identifier is None:
        return "``"
    normalized = inspire_normalize_identifier(identifier)
    return "`" + normalized.replace("`", "``") + "`"

def inspire_quote_qualified_name(name: str) -> str:
    parts = [p for p in str(name or "").split(".") if p.strip()]
    return ".".join(inspire_quote_identifier(p) for p in parts)



def inspire_table_scope_key(name) -> str:
    return str(name or "").replace("`", "").strip().lower()

def inspire_scope_projection(frame, scope):
    return frame.select(*[f"`{c}`" for c in scope["columns"]]) if scope and scope["columns"] else frame

def _scoped_read_entry(scope, entry):
    if not scope:
        return entry
    entry["where"] = " AND ".join(p for p in (scope["where"], entry["where"]) if p)
    entry["columns"] = scope["columns"]
    return entry

def inspire_table_scope(name):
    key = inspire_table_scope_key(name)
    run = globals().get("INSPIRE_RUN") or {}
    scope = (run.get("table_scopes") or {}).get(key)
    if scope is None:
        scope = (INSPIRE_TABLE_SCOPES or {}).get(key)
    if scope is None and key in {str(k) for k in run.get("scoped_tables") or []}:
        raise ValueError(f"{name} has a Table Scope, but this run does not carry its scope, so Inspire will not read the whole table; "
                         f"build the use case again")
    return scope

INSPIRE_SESSION_ID_RANGE = (-(2 ** 63), 2 ** 63 - 1)


def inspire_session_id(value) -> str:
    if value is None:
        return ""
    if isinstance(value, bool) or not isinstance(value, (int, str)):
        raise ValueError("an Inspire session id is a whole number written in digits, never a fraction, a flag or a list")
    text = str(value).strip()
    digits = text[1:] if text.startswith("-") else text
    if text and not (digits.isascii() and digits.isdigit()
                     and INSPIRE_SESSION_ID_RANGE[0] <= int(text) <= INSPIRE_SESSION_ID_RANGE[1]):
        raise ValueError("an Inspire session id is a whole number that fits in a BIGINT, written in ASCII digits "
                         "with an optional leading minus")
    return str(int(text)) if text else ""

def inspire_table_parts(name):
    parts = [part.strip() for part in str(name or "").replace("`", "").split(".")]
    if len(parts) != 3 or not all(parts):
        raise ValueError(f"expected a three-level table name catalog.schema.table, got {name!r}")
    return parts


def inspire_companion(table, suffix: str) -> str:
    # REQ-264: the companion table of a result table, named by the one rule the App reads (inspire_shell_facts.companion); a companion
    # needs a three-level table name, so the name is validated first.
    inspire_table_parts(table)
    return inspire_shell_facts.companion(table, suffix)


def inspire_quote_principal(principal) -> str:
    return inspire_shell_facts.quote_principal(principal)


def inspire_table_name(name, suffix=""):
    catalog, schema, table = inspire_table_parts(name)
    return ".".join(f"`{part}`" for part in (catalog, schema, table + suffix))


def inspire_sql_literal(text):
    return "'" + str("" if text is None else text).replace("\\", "\\\\").replace("'", "\\'") + "'"


def inspire_column(name, alias=""):
    return (f"{alias}." if alias else "") + "`" + str(name).replace("`", "``") + "`"


def inspire_rows(frame):
    return [row.asDict() if hasattr(row, "asDict") else dict(row) for row in frame.collect()]


def inspire_table_described(spark, table, extended=False):
    try:
        rows = inspire_rows(spark.sql(f"DESCRIBE TABLE {'EXTENDED ' if extended else ''}{inspire_table_name(table)}"))
    except Exception as error:
        if "TABLE_OR_VIEW_NOT_FOUND" in str(error) or "cannot be found" in str(error):
            return None
        raise
    columns, comment, section = [], "", "columns"
    for row in rows:
        name = str(row.get("col_name") or "").strip()
        if section == "columns" and name and not name.startswith("#"):
            columns.append((name, " ".join(str(row.get("data_type") or "").split()), " ".join(str(row.get("comment") or "").split())))
            continue
        if not extended:
            break
        section = "detail" if name == "# Detailed Table Information" else "other" if section == "columns" else section
        if section == "detail" and name.lower() == "comment":
            comment = " ".join(str(row.get("data_type") or "").split())
    return {"columns": columns, "comment": comment} if columns else None

def _match_index(columns_by_table) -> dict:
    return {str(table).lower(): (str(table), {str(col).lower(): str(col) for col in columns or ()})
            for table, columns in (columns_by_table or {}).items()}


def _match_evidence(entry, index):
    """One evidence entry kept only when its table and every column exist in the scanned metadata (their spelling). None otherwise."""
    if not isinstance(entry, dict):
        return None
    table = index.get(str(entry.get("table") or "").strip().lower())
    columns = entry.get("columns")
    if table is None or not isinstance(columns, (list, tuple)) or not columns:
        return None
    kept = []
    for column in columns:
        real = table[1].get(str(column).strip().lower())
        if real is None:
            return None
        if real not in kept:
            kept.append(real)
    return {"table": table[0], "columns": kept, "why": " ".join(str(entry.get("why") or "").split())}


def _match_confidence(value) -> int:
    try:
        return max(0, min(100, int(float(value) + 0.5)))
    except (TypeError, ValueError, OverflowError):
        return 0


def _match_int(value, default=None):
    try:
        return int(value)
    except (TypeError, ValueError):
        return default


def inspire_match_validate(report, columns_by_table) -> dict:
    """The Match report as the contract states it. Code, never the model, decides what is real: evidence naming a table or column that
    the scan did not list is dropped; a piece is 'matched' only with confidence >= 80 and evidence, 'partial' with evidence below that,
    'missing' without evidence (confidence 0). The verdict is 'full' when every piece is matched, 'none' when none is matched or partial,
    else 'partial'; the confidence is the rounded mean of the pieces'."""
    report = report if isinstance(report, dict) else {}
    index = _match_index(columns_by_table)
    pieces = []
    for number, raw in enumerate(report.get("pieces") or [], 1):
        if not isinstance(raw, dict):
            continue
        evidence = [item for item in (_match_evidence(entry, index) for entry in raw.get("evidence") or []) if item]
        confidence = _match_confidence(raw.get("confidence"))
        gap = " ".join(str(raw.get("gap") or "").split())
        if not evidence:
            status, confidence = "missing", 0
            gap = gap or "No table in the scanned metadata answers this piece."
        elif confidence >= MATCH_MATCHED_CONFIDENCE:
            status = "matched"
        else:
            status = "partial"
            gap = gap or "The columns found answer this piece only in part."
        pieces.append({"piece_no": _match_int(raw.get("piece_no"), number), "requirement": " ".join(str(raw.get("requirement") or "").split()),
                       "status": status, "confidence": confidence, "evidence": evidence, "gap": gap,
                       "known_data_option": _match_evidence(raw.get("known_data_option"), index)})
    matched = sum(1 for piece in pieces if piece["status"] == "matched")
    answered = sum(1 for piece in pieces if piece["status"] != "missing")
    verdict = "full" if pieces and matched == len(pieces) else ("partial" if answered else "none")
    overall = int(sum(piece["confidence"] for piece in pieces) / len(pieces) + 0.5) if pieces else 0
    return {"input_no": _match_int(report.get("input_no"), report.get("input_no")), "name": str(report.get("name") or ""),
            "verdict": verdict, "confidence": overall, "pieces": pieces}


def inspire_match_breakdown(parsed) -> list:
    """The pieces of a breakdown answer as [{piece_no, requirement}]: numbered from 1 when the model gave no number, an empty
    requirement or a repeated number dropped."""
    pieces, seen = [], set()
    for number, raw in enumerate((parsed.get("pieces") if isinstance(parsed, dict) else None) or [], 1):
        if not isinstance(raw, dict):
            continue
        requirement = " ".join(str(raw.get("requirement") or "").split())
        piece_no = _match_int(raw.get("piece_no"), number)
        if requirement and piece_no not in seen:
            seen.add(piece_no)
            pieces.append({"piece_no": piece_no, "requirement": requirement})
    return pieces


def inspire_match_merge(breakdown, validated_batches) -> list:
    """One piece per breakdown piece: the highest-confidence validated answer over all metadata batches (the earlier batch on a tie), so
    its evidence comes from that batch only. The requirement stays the breakdown's. A piece no batch answered is missing."""
    merged = []
    for piece in breakdown:
        best = None
        for batch in validated_batches:
            for found in batch:
                if found["piece_no"] == piece["piece_no"] and (best is None or found["confidence"] > best["confidence"]):
                    best = found
        merged.append(dict(best or {"status": "missing", "confidence": 0, "evidence": [], "gap": "", "known_data_option": None},
                           piece_no=piece["piece_no"], requirement=piece["requirement"]))
    return merged


def _match_at(row):
    try:
        return row["event_at"]
    except (KeyError, IndexError, TypeError, ValueError):
        return None


def inspire_match_latest_rows(rows, key) -> list:
    """For each key(row) the row with the latest event_at (the later row of a tie), in key order."""
    best = {}
    for row in rows or []:
        found = key(row)
        current = best.get(found)
        mark = (_match_at(row) is not None, _match_at(row))
        if current is None or mark >= current[0]:
            best[found] = (mark, row)
    return [best[found][1] for found in sorted(best)]


def _match_names_columns(item):
    return isinstance(item, dict) and bool(item.get("table")) and bool(item.get("columns"))


def inspire_match_verdict(raw_pieces, chosen):
    """The accept step's verdict over the pieces the person kept, re-checked from the fields (the stored status is not trusted):
    supported = matched at MATCH_MATCHED_CONFIDENCE or more with a table and columns of evidence, or use_known_data on an option that
    names a table and columns. full when every kept piece is supported; none when no kept piece is supported or has evidence; else
    partial. The confidence is the rounded mean over the kept pieces (dropped never counted), 0 for none."""
    kept = [raw for raw in raw_pieces if chosen.get(_match_int(raw.get("piece_no")), "keep") != "drop"]
    supported, scored, scores = 0, False, []
    for raw in kept:
        decision = chosen.get(_match_int(raw.get("piece_no")), "keep")
        evidence = [item for item in raw.get("evidence") or [] if _match_names_columns(item)]
        confidence = _match_int(raw.get("confidence")) or 0
        known = decision == "use_known_data" and _match_names_columns(raw.get("known_data_option"))
        ok = known or (raw.get("status") == "matched" and confidence >= MATCH_MATCHED_CONFIDENCE and bool(evidence))
        supported += 1 if ok else 0
        if ok or raw.get("evidence"):
            scored = True
        scores.append(confidence)
    if not kept or not scored:
        return "none", 0
    return ("full" if supported == len(kept) else "partial"), int(sum(scores) / len(scores) + 0.5)


def inspire_match_accept(report, decisions):
    """Apply the person's decisions to one stored report: drop leaves a piece out, use_known_data rewrites it to its known-data option
    (a piece without one is kept as it is), keep or no decision keeps it; the latest event_at wins. The name stays verbatim. None when
    no piece is left."""
    report = report if isinstance(report, dict) else {}
    input_no = _match_int(report.get("input_no"))
    mine = [row for row in decisions or []
            if _match_int(row["input_no"]) == input_no and _match_int(row["piece_no"]) is not None]
    chosen = {}
    for row in inspire_match_latest_rows(mine, lambda r: _match_int(r["piece_no"])):
        chosen[_match_int(row["piece_no"])] = str(row["decision"] or "").strip().lower()
    pieces = []
    for raw in report.get("pieces") or []:
        decision = chosen.get(_match_int(raw.get("piece_no")), "keep")
        if decision == "drop":
            continue
        piece = {"piece_no": raw.get("piece_no"), "requirement": raw.get("requirement", ""), "status": raw.get("status", ""),
                 "confidence": raw.get("confidence", 0), "evidence": list(raw.get("evidence") or []), "gap": raw.get("gap", ""),
                 "decision": "keep"}
        known = raw.get("known_data_option")
        if decision == "use_known_data" and isinstance(known, dict):
            piece.update(evidence=[dict(known)], status="matched", gap="", decision="use_known_data")
        pieces.append(piece)
    if not pieces:
        return None
    tables = sorted({item["table"] for piece in pieces for item in piece["evidence"]})
    dropped = [{"piece_no": raw.get("piece_no"), "requirement": raw.get("requirement", "")} for raw in report.get("pieces") or []
               if chosen.get(_match_int(raw.get("piece_no")), "keep") == "drop"]
    verdict, confidence = inspire_match_verdict(report.get("pieces") or [], chosen)
    return {"input_no": report.get("input_no"), "name": str(report.get("name") or ""), "pieces": pieces, "tables": tables,
            "dropped": dropped, "verdict": verdict, "confidence": confidence}

              else key == INSPIRE_GROUP_BY and INSPIRE_GROUP_CUSTOM in [part.lower() for part in inspire_group_by_parts(text)])

def inspire_setting_problem(setting, value):
    setting = setting or {}
    name, text = str(setting.get("name") or ""), str("" if value is None else value).strip()
    custom = inspire_custom_sql_problem(name, text)
    if custom:
        return custom
    if setting.get("kind") == "choice":
        choices = [str(choice) for choice in setting.get("choices") or []]
        picked = [part.strip() for part in text.split(",") if part.strip()] if setting.get("many") else [text]
        allowed = choices + ([INSPIRE_SETTING_ALL] if setting.get("many") else [])
        wrong = [part for part in picked if choices and part not in allowed]
        if (not text and not setting.get("optional")) or wrong:
            return f"{name} is '{text}'; allowed: {', '.join(allowed) if choices else 'any text'}"
        alone = [part for part in picked if part in (setting.get("solo") or [])]
        if alone and len(picked) > 1:
            return f"{name} is '{text}'; {' and '.join(alone)} stand{'s' if len(alone) == 1 else ''} alone, never with another choice"
        return ""
    if not text and setting.get("optional"):
        return ""
    low, high, whole = setting.get("low"), setting.get("high"), bool(setting.get("whole"))
    low_open, high_open = bool(setting.get("low_open")), bool(setting.get("high_open"))
    number = float(text) if _inspire_re.fullmatch(INSPIRE_SETTING_NUMBER, text) and not (whole and "." in text) else None
    if number is None or (low is not None and (number <= low if low_open else number < low)) or (
            high is not None and (number >= high if high_open else number > high)):
        return f"{name} is '{text}'; allowed: {setting.get('allowed') or 'a number'}"
    return ""


def inspire_assumption_setting(name, kind, low=None, high=None):

    return None


def inspire_value_bounds(value, formula):
    listed = list((formula or {}).get("assumptions") or [])
    if not listed:
        return []
    declared = (value or {}).get("bounds")
    if declared is None and not (value or {}).get("bounds_required"):
        return []
    rule = (f"define {INSPIRE_VALUE_BOUNDS_NAME} at top level next to {INSPIRE_VALUE_FORMULA_NAME}: for every assumption constant the "
            f"formula names, its [lowest, highest] sensible value, such as {INSPIRE_VALUE_BOUNDS_EXAMPLE}")
    widgets = {str(name) for name in (value or {}).get("widget_names") or ()}
    problems = [f"{a['name']} is a widget, so it is a setting of the run and never an assumption constant of {INSPIRE_VALUE_FORMULA_NAME}; "
                f"write its value as a numeric column of every row and name that column in the formula" for a in listed if a["name"] in widgets]
    named = {a["name"] for a in listed}
    listed = [a for a in listed if a["name"] not in widgets]
    if not listed:
        return problems
    if declared is None:
        return problems + [f"the notebook has no {INSPIRE_VALUE_BOUNDS_NAME}, so the App cannot keep {', '.join(a['name'] for a in listed)} "
                           f"inside a sensible range when an owner changes it; {rule}"]
    if not isinstance(declared, dict):
        return problems + [f"{INSPIRE_VALUE_BOUNDS_NAME} must be a dict of each constant's name to its [lowest, highest]; {rule}"]
    problems += [f"{INSPIRE_VALUE_BOUNDS_NAME} names {extra}, which is not an assumption constant {INSPIRE_VALUE_FORMULA_NAME} names"
                 for extra in sorted(set(map(str, declared)) - named)]
    for item in listed:
        name, kind, held = item["name"], item["kind"], float(item["value"])
        pair = inspire_declared_range(declared.get(name))
        if name not in declared:
            problems.append(f"{INSPIRE_VALUE_BOUNDS_NAME} gives no range for {name}; {rule}")
            continue
        if pair is None:
            problems.append(f"{INSPIRE_VALUE_BOUNDS_NAME}[{name!r}] is {declared[name]!r}; it must be [lowest, highest], two numbers")
            continue
        low, high = pair
        shown = f"{inspire_number_text(low)} to {inspire_number_text(high)}"
        if not low <= held <= high:
            problems.append(f"{name} is {inspire_number_text(held)}, outside its own range {shown} in {INSPIRE_VALUE_BOUNDS_NAME}")
        elif kind == "percent" and (low < 0 or high > 1):
            problems.append(f"{name} is a share (:percent), so its range must sit inside 0 to 1, not {shown}")
        elif kind == "money" and low < 0:
            problems.append(f"{name} is money (:money), so its range cannot go below 0, not {shown}")
        else:
            item.update(low=low, high=high)
    return problems

    try:

def inspire_kpi_first(spark, table, use_case_id, kpi):
    name, unit = (" ".join(str(kpi.get(key) or "").split()) for key in ("name", "unit"))
    same = f"kpi_query = {inspire_sql_literal(' '.join(str(kpi.get('query') or '').split()))}"

    rows = inspire_rows(spark.sql(
        f"SELECT kpi_latest AS v, CAST(kpi_measured_at AS DATE) AS d FROM {inspire_table_name(inspire_ai_summary_table(table))} "
        f"WHERE use_case_id = {inspire_sql_literal(use_case_id)} AND kpi_name = {inspire_sql_literal(name)} "
        f"AND {same} AND kpi_status = 'measured' AND kpi_latest IS NOT NULL "
        "ORDER BY kpi_measured_at ASC NULLS LAST LIMIT 1"))
    value = inspire_kpi_number(rows[0].get("v")) if rows else None
    return (value, str(rows[0].get("d") or "")[:10]) if value is not None else (None, "")


def inspire_save_kpi(spark, table, use_case_id, run_id, run=None, out=print):
    run = (globals().get("INSPIRE_RUN") or {}) if run is None else run
    kpi = (run or {}).get("target_kpi")
    if not isinstance(kpi, dict) or not " ".join(str(kpi.get("name") or "").split()):
        return None
    name, query = " ".join(str(kpi["name"]).split()), str(kpi.get("query") or "")
    status, latest, reason = "not_measured", None, " ".join(str(kpi.get("reason") or "").split())
    if kpi.get("computable") and _inspire_re.fullmatch(INSPIRE_KPI_QUERY, query):
        try:
            rows = inspire_rows(spark.sql(query))
            latest = inspire_kpi_number(rows[0].get("v") if rows else None)
            status, reason = ("measured", "") if latest is not None else ("not_measured", "its query returned no value")
        except Exception as error:
            status, reason = "failed", f"its query failed: {inspire_redact_text(str(error), set(), '')[:300]}"
    elif kpi.get("computable"):
        reason = "its query is not one read-only SELECT, so Inspire does not run it"
    elif not reason:
        reason = "Inspire has no query for it"
    flag = " ".join(str(kpi.get("baseline_reason") or "").split())
    reason = flag if status == "measured" else "; ".join(part for part in (reason, flag) if part)
    try:
        inspire_ensure_summary(spark, table)
        if status == "measured" and inspire_kpi_number(kpi.get("baseline")) is None:
            first, first_on = inspire_kpi_first(spark, table, use_case_id, kpi)
            kpi = dict(kpi, baseline=latest if first is None else first,
                       baseline_on=first_on or _inspire_datetime.datetime.now(_inspire_datetime.timezone.utc).date().isoformat())
        values = inspire_kpi_values(kpi, status, latest, reason)
        spark.sql(inspire_ai_summary_merge_sql(table, use_case_id, run_id, values))
    except Exception as error:
        out(f"Target KPI: run {run_id} wrote its rows, but Inspire could not record {name} in {inspire_ai_summary_table(table)}: "
            f"{inspire_redact_text(str(error), set(), '')[:300]}")
        return None
    unit = " ".join(str(kpi.get("unit") or "").split())
    unit = unit if unit == "%" else f" {unit}" if unit else ""
    out(f"Target KPI: {name} is {inspire_number_text(latest)}{unit} this run{'; ' + flag if flag else ''}"
        if status == "measured" else f"Target KPI: {name} is not measured this run, because {reason}")
    return values

def inspire_share_cuts(value):
    shares = [inspire_amount(((value or {}).get("shares") or {}).get(name)) for name in INSPIRE_SHARE_NAMES]
    passed = (value or {}).get("passed")
    if passed is None or None in shares:
        return None
    act_now, flagged = int(_inspire_math.floor(shares[0] * passed + 1e-9)), int(_inspire_math.floor((shares[0] + shares[1]) * passed + 1e-9))
    if flagged >= passed:
        act_now = min(act_now, max(INSPIRE_FULL_SHARE_ACT_MIN, -(-passed * INSPIRE_FULL_SHARE_ACT_PERCENT // 100)))
    return (act_now, flagged)

def inspire_value_problems(value):
    problems = []
    if value["basis"] not in INSPIRE_DECISION_BASES:
        problems.append(f"the run's decision basis is '{value['basis']}', and it must be one of {', '.join(INSPIRE_DECISION_BASES)}")
    money = value.get("kind", "money") == "money"
    if money and value["currency"] not in INSPIRE_CURRENCY_CODES:
        problems.append(f"the currency widget is '{value['currency']}'; set its default to the 3-letter ISO code of the money "
                        f"{INSPIRE_VALUE_AT_STAKE} holds, such as USD")
    for name, text in value["amounts"].items():
        if inspire_amount(text) is None:
            problems.append(f"the {name} widget is '{text}'; set its default to an amount of at least 0 in the "
                            + (f"currency of {INSPIRE_VALUE_AT_STAKE}" if money else f"unit of {INSPIRE_VALUE_AT_STAKE} "
                               f"({inspire_value_unit(value)})") + ", since this use case acts on measured amounts")
    order = inspire_amount_order_problem(*inspire_value_amounts(value))
    if value["basis"] == "predicted" and value.get("shares_required") and value.get("accepted") is not False and value.get("passed") is None:
        problems.append("pass passed=<the number of rows that pass the decision test> to inspire_write_result, so Inspire can check the "
                        "share tiers and the App can show how a share change moves the records")
    return problems + ([order] if order else [])


def inspire_value_formula(value, schema):
    text = (value or {}).get("formula")
    rule = (f"define {INSPIRE_VALUE_FORMULA_NAME} at top level, after the Parameters cell: the {INSPIRE_VALUE_AT_STAKE} formula in "
            "plain words for the people who act on the records, each amount a {name} that names a numeric column of the rows or "
            "a named assumption constant, followed by :money, :number or :percent (a share from 0 to 1), such as "
            f"{INSPIRE_VALUE_FORMULA_EXAMPLE!r}; Inspire fills in each record's numbers and adds ' = ' and its {INSPIRE_VALUE_AT_STAKE}")
    if not isinstance(text, str) or not text.strip():
        return None, [f"the notebook has no {INSPIRE_VALUE_FORMULA_NAME}, so the App cannot show how each record's "
                      f"{inspire_stake_words(value)} is worked out; {rule}"]
    text = " ".join(text.split())
    numeric = {str(name).lower(): name for name, kind in schema or [] if _inspire_re.match(INSPIRE_NUMERIC_TYPES, str(kind).strip())}
    known = {str(name).lower() for name, _kind in schema or []}
    constants = (value or {}).get("constants") or {}
    problems, parts, at, columns = [], [], 0, []
    if len(text) > INSPIRE_VALUE_FORMULA_MAX_CHARS:
        problems.append(f"{INSPIRE_VALUE_FORMULA_NAME} is {inspire_plural(len(text), 'character')}; at most {INSPIRE_VALUE_FORMULA_MAX_CHARS} are allowed")
    for match in _inspire_re.finditer(INSPIRE_VALUE_FORMULA_PLACEHOLDER, text):
        parts.append(("text", text[at:match.start()]))
        at = match.end()
        name, kind = match.group(1).strip(), (match.group(2) or "number").strip()
        if kind not in INSPIRE_VALUE_FORMULA_KINDS:
            problems.append(f"{{{match.group(1)}:{match.group(2)}}} in {INSPIRE_VALUE_FORMULA_NAME} has the format {kind!r}; use "
                            + ", ".join(INSPIRE_VALUE_FORMULA_KINDS))
        if name.lower() == INSPIRE_VALUE_AT_STAKE:
            problems.append(f"{INSPIRE_VALUE_FORMULA_NAME} names {INSPIRE_VALUE_AT_STAKE} itself; name the amounts it is worked out "
                            "from, since Inspire adds the result")
        elif name.lower() in numeric and not _inspire_re.fullmatch(INSPIRE_VALUE_NAME, numeric[name.lower()]):
            problems.append(inspire_value_name_problem(f"{{{name}}} in {INSPIRE_VALUE_FORMULA_NAME}", numeric[name.lower()]))
        elif name.lower() in numeric:
            parts.append(("column", numeric[name.lower()], kind))
            columns.append(numeric[name.lower()])
        elif name.lower() in known:
            problems.append(f"{{{name}}} in {INSPIRE_VALUE_FORMULA_NAME} names a column of the rows that is not a number")
        elif inspire_number_constant(constants.get(name)) and not _inspire_re.fullmatch(INSPIRE_VALUE_NAME, name):
            problems.append(inspire_value_name_problem(f"{{{name}}} in {INSPIRE_VALUE_FORMULA_NAME}"))
        elif inspire_number_constant(constants.get(name)):
            parts.append(("constant", float(constants[name]), kind, name))
        else:
            problems.append(f"{{{name}}} in {INSPIRE_VALUE_FORMULA_NAME} is neither a numeric column of the rows nor a number "
                            "constant defined at top level")
    parts.append(("text", text[at:]))
    words = "".join(part[1] for part in parts if part[0] == "text")
    if "{" in words or "}" in words:
        problems.append(f"{INSPIRE_VALUE_FORMULA_NAME} has a brace outside a {{name}} or {{name:format}}")
    if "=" in words:
        problems.append(f"{INSPIRE_VALUE_FORMULA_NAME} holds '='; leave out the result, since Inspire adds ' = ' and the record's "
                        f"{INSPIRE_VALUE_AT_STAKE}")
    if not columns and not problems:
        problems.append(f"{INSPIRE_VALUE_FORMULA_NAME} names no column of the rows, so every record would show the same numbers; {rule}")
    if problems:
        return None, problems
    assumptions = {}
    for part in parts:
        if part[0] == "constant":
            assumptions.setdefault(part[3], {"name": part[3], "value": part[1], "kind": part[2]})
    return {"parts": [part for part in parts if part != ("text", "")], "columns": list(dict.fromkeys(columns)), "text": text,
            "assumptions": list(assumptions.values())}, []


def inspire_expression_tokens(text):
    tokens, at, pattern = [], 0, _inspire_re.compile(INSPIRE_VALUE_EXPRESSION_TOKEN)
    while text[at:].strip():
        match = pattern.match(text, at)
        if not match:
            return None, (f"{INSPIRE_VALUE_EXPRESSION_NAME} has {text[at:].strip()[:20]!r}, which is not a number, a name, + - * / ( ) "
                          "or a comma")
        number, name, sign = match.groups()
        tokens.append(("number", number) if number else ("name", name) if name else ("sign", sign))
        at = match.end()
    return tokens, ""


def inspire_expression_tree(tokens):
    at = [0]

    def peek():
        return tokens[at[0]] if at[0] < len(tokens) else (None, None)

    def take(kind=None, text=None):
        token = peek()
        if token[0] is None or (kind and token[0] != kind) or (text and token[1] != text):
            raise ValueError(f"expected {text or kind} at {' '.join(t[1] for t in tokens[at[0]:at[0] + 3]) or 'the end'}")
        at[0] += 1
        return token

    def factor():
        kind, text = peek()
        if (kind, text) == ("sign", "-"):
            take()
            return ("negative", factor())
        if (kind, text) == ("sign", "("):
            take()
            inner = expression()
            take("sign", ")")
            return inner
        if kind == "number":
            return ("number", float(take()[1]))
        if kind == "name":
            take()
            if peek() == ("sign", "("):
                if text.lower() not in INSPIRE_VALUE_EXPRESSION_FUNCTIONS:
                    raise ValueError(f"{text}( ) is not greatest( ) or least( )")
                take()
                args = [expression()]
                while peek() == ("sign", ","):
                    take()
                    args.append(expression())
                take("sign", ")")
                if len(args) < 2:
                    raise ValueError(f"{text}( ) needs at least two values")
                return ("call", text.lower(), args)
            if text.lower() in INSPIRE_VALUE_EXPRESSION_FUNCTIONS:
                raise ValueError(f"{text} is a function, so it takes values in ( )")
            if not _inspire_re.fullmatch(INSPIRE_VALUE_NAME, text):
                raise ValueError(f"{text[:20]}... is longer than {INSPIRE_VALUE_NAME_CHARS} characters")
            return ("name", text)
        raise ValueError(f"expected a number, a name or ( at {text or 'the end'}")

    def term():
        node = factor()
        while peek() in (("sign", "*"), ("sign", "/")):
            node = ("operation", take()[1], node, factor())
        return node

    def expression():
        node = term()
        while peek() in (("sign", "+"), ("sign", "-")):
            node = ("operation", take()[1], node, term())
        return node

    tree = expression()
    if at[0] != len(tokens):
        raise ValueError(f"unexpected {tokens[at[0]][1]}")
    return tree


def inspire_expression_names(tree):
    if tree[0] == "name":
        return [tree[1]]
    if tree[0] == "negative":
        return inspire_expression_names(tree[1])
    if tree[0] == "operation":
        return inspire_expression_names(tree[2]) + inspire_expression_names(tree[3])
    if tree[0] == "call":
        return [name for arg in tree[2] for name in inspire_expression_names(arg)]
    return []


def inspire_expression_sql(tree, named):
    if tree[0] == "number":
        return repr(tree[1])
    if tree[0] == "name":
        return named(tree[1])
    if tree[0] == "negative":
        return f"(- {inspire_expression_sql(tree[1], named)})"
    if tree[0] == "call":
        return f"{tree[1].upper()}({', '.join(inspire_expression_sql(arg, named) for arg in tree[2])})"
    left, right = inspire_expression_sql(tree[2], named), inspire_expression_sql(tree[3], named)
    return f"({left} / NULLIF({right}, 0))" if tree[1] == "/" else f"({left} {tree[1]} {right})"


def inspire_value_expression(value, schema, formula):
    text = (value or {}).get("expression")
    shown = [a["name"] for a in (formula or {}).get("assumptions") or []]
    rule = (f"define {INSPIRE_VALUE_EXPRESSION_NAME} at top level next to {INSPIRE_VALUE_FORMULA_NAME}: the same formula as "
            "arithmetic over the same names, with only numbers, + - * / ( ) and greatest( ) or least( ), such as "
            f"{INSPIRE_VALUE_EXPRESSION_EXAMPLE!r}")
    if not isinstance(text, str) or not text.strip():
        if shown and (value or {}).get("expression_required"):
            return None, [f"the notebook has no {INSPIRE_VALUE_EXPRESSION_NAME}, so the App cannot work out the "
                          f"{inspire_stake_words(value)} again when someone changes {', '.join(shown)}; {rule}"]
        return None, []
    text = " ".join(text.split())
    if len(text) > INSPIRE_VALUE_FORMULA_MAX_CHARS:
        return None, [f"{INSPIRE_VALUE_EXPRESSION_NAME} is {inspire_plural(len(text), 'character')}; at most {INSPIRE_VALUE_FORMULA_MAX_CHARS} are allowed"]
    tokens, problem = inspire_expression_tokens(text)
    if problem:
        return None, [f"{problem}; {rule}"]
    try:
        tree = inspire_expression_tree(tokens)
    except ValueError as error:
        return None, [f"{INSPIRE_VALUE_EXPRESSION_NAME} {text!r} is not arithmetic Inspire can read: {inspire_redact_text(str(error), set(), text)}; {rule}"]
    numeric = {str(name).lower(): name for name, kind in schema or [] if _inspire_re.match(INSPIRE_NUMERIC_TYPES, str(kind).strip())}
    known = {str(name).lower() for name, _kind in schema or []}
    constants = (value or {}).get("constants") or {}
    problems, columns, used = [], [], []
    for name in dict.fromkeys(inspire_expression_names(tree)):
        if name.lower() in numeric and not _inspire_re.fullmatch(INSPIRE_VALUE_NAME, numeric[name.lower()]):
            problems.append(inspire_value_name_problem(f"{name} in {INSPIRE_VALUE_EXPRESSION_NAME}", numeric[name.lower()]))
        elif name.lower() in numeric:
            columns.append(numeric[name.lower()])
        elif name.lower() in known:
            problems.append(f"{name} in {INSPIRE_VALUE_EXPRESSION_NAME} names a column of the rows that is not a number")
        elif inspire_number_constant(constants.get(name)):
            used.append(name)
        else:
            problems.append(f"{name} in {INSPIRE_VALUE_EXPRESSION_NAME} is neither a numeric column of the rows nor a number constant "
                            "defined at top level")
    if not problems and set(used) != set(shown):
        extra, missing = sorted(set(used) - set(shown)), sorted(set(shown) - set(used))
        problems.append(f"{INSPIRE_VALUE_EXPRESSION_NAME} must name the same assumption constants as {INSPIRE_VALUE_FORMULA_NAME}, so "
                        f"the App shows every assumption that changes the {inspire_stake_noun(value)}: "
                        + "; ".join(([f"only the expression names {', '.join(extra)}"] if extra else [])
                                    + ([f"only the formula names {', '.join(missing)}"] if missing else [])))
    if not problems and not columns:
        problems.append(f"{INSPIRE_VALUE_EXPRESSION_NAME} names no column of the rows, so every record would have the same "
                        f"{inspire_stake_words(value)}")
    if problems:
        return None, problems
    return {"text": text, "tree": tree, "columns": columns}, []


def inspire_value_expression_sql(expression, value, alias=""):
    constants = (value or {}).get("constants") or {}
    numeric = {str(c).lower(): c for c in expression["columns"]}
    return inspire_expression_sql(expression["tree"], lambda name: (f"CAST({inspire_column(numeric[name.lower()], alias)} AS DOUBLE)"
                                                                      if name.lower() in numeric else repr(float(constants[name]))))


def inspire_amount_sql(number, kind, currency):
    scaled = f"CAST({number} AS DOUBLE)" + (" * 100" if kind == "percent" else "")
    shown = f"regexp_replace(format_number({scaled}, {1 if kind == 'percent' else 2}), {inspire_sql_literal(INSPIRE_TRAILING_ZEROS)}, '')"
    suffix = {"money": " " + str(currency) if currency else "", "percent": "%"}.get(kind, "")
    return f"concat({shown}, {inspire_sql_literal(suffix)})" if suffix else shown


def inspire_value_formula_sql(formula, value, alias):
    pieces = []
    for part in formula["parts"]:
        if part[0] == "text":
            pieces.append(inspire_sql_literal(part[1]))
        else:
            number = inspire_column(part[1], alias) if part[0] == "column" else repr(part[1])
            pieces.append(inspire_amount_sql(number, part[2], value["currency"]))
    stake = inspire_column(INSPIRE_VALUE_AT_STAKE, alias)
    pieces += [inspire_sql_literal(" = "), inspire_amount_sql(stake, "money", inspire_value_unit(value))]
    return f"CASE WHEN {stake} IS NULL THEN NULL ELSE concat({', '.join(pieces)}) END"


def inspire_run_profile_sql(stage, columns, value=None, formula=None, expression=None):
    blank = "COUNT_IF({0} IS NULL OR TRIM(CAST({0} AS STRING)) = '')"
    named = {str(c).lower(): inspire_column(c) for c in columns}
    checks = ""
    if value and INSPIRE_VALUE_AT_STAKE in named:
        stake, tier = named[INSPIRE_VALUE_AT_STAKE], inspire_tier_column(columns)
        act_now, review = inspire_value_amounts(value)
        checks += f", COUNT_IF({stake} < 0) AS negative_stake, SUM({stake}) AS stake_total"
        if formula:
            missing = " OR ".join(f"{inspire_column(c)} IS NULL" for c in formula["columns"])
            checks += f", COUNT_IF({stake} IS NOT NULL AND ({missing})) AS unshown_stake"
        if expression:
            worked = inspire_value_expression_sql(expression, value)
            gap, share = INSPIRE_VALUE_TOLERANCE
            checks += (f", COUNT_IF({stake} IS NOT NULL AND ({worked} IS NULL OR abs({worked} - {stake}) > "
                       f"greatest({gap!r}, abs({stake}) * {share!r}))) AS off_expression")
        if value["basis"] == "predicted" and value.get("accepted") is False and tier:
            checks += f", COUNT_IF({inspire_column(tier)} IN ({inspire_sql_list(INSPIRE_FLAGGED_TIERS)})) AS unaccepted_flagged"
        cuts = inspire_share_cuts(value) if value["basis"] == "predicted" and value.get("accepted") is not False else None
        if cuts and tier and "rank" in named:
            ok, rank = (f"{named['data_quality_flag']} = 'ok'" if "data_quality_flag" in named else "TRUE"), named["rank"]
            checks += (f", COUNT_IF(NOT ({inspire_column(tier)} <=> CASE WHEN {ok} AND {rank} <= {cuts[0]} THEN 'Act now' "
                       f"WHEN {ok} AND {rank} <= {cuts[1]} THEN 'Review' ELSE 'Monitor' END)) AS off_share_tier"
                       f", COUNT_IF({ok}) AS ok_rows")
        if value["basis"] == "measured" and tier and act_now is not None and review is not None:
            ok = (f"{named['data_quality_flag']} = 'ok'" if "data_quality_flag" in named else "TRUE") + inspire_flag_segments_sql(value, named)
            checks += (f", COUNT_IF(NOT ({inspire_column(tier)} <=> CASE WHEN {ok} AND {stake} >= {act_now!r} THEN 'Act now' "
                       f"WHEN {ok} AND {stake} >= {review!r} THEN 'Review' ELSE 'Monitor' END)) AS off_tier")
    if INSPIRE_CONFIDENCE in named:
        level = named[INSPIRE_CONFIDENCE]
        checks += (f", COUNT_IF({level} IS NULL OR {level} NOT IN ({inspire_sql_list(INSPIRE_CONFIDENCE_LEVELS)})) "
                   "AS bad_levels")
    if INSPIRE_AI_FACTS in named:
        tier = inspire_tier_column(columns)
        flagged = f"{inspire_column(tier)} IN ({inspire_sql_list(INSPIRE_FLAGGED_TIERS)})" if tier else "FALSE"
        checks += f", COUNT_IF({flagged} AND ({named[INSPIRE_AI_FACTS]} IS NULL OR TRIM({named[INSPIRE_AI_FACTS]}) = '')) AS bare_flagged"
    return (f"SELECT COUNT(*) AS row_count, COUNT(DISTINCT use_case_id) AS use_cases, MIN(use_case_id) AS use_case_id, "
            f"COUNT(DISTINCT run_id) AS runs, MIN(run_id) AS run_id, COUNT_IF(as_of_date IS NULL) AS undated, "
            f"{blank.format('use_case_id')} AS no_use_case, {blank.format('run_id')} AS no_run, "
            f"{inspire_unique_sql(columns, 'entity_id')} AS id_unique, {inspire_unique_sql(columns, 'entity_label')} AS label_unique"
            f"{checks} FROM {inspire_table_name(stage)}")

    return f" AND {named['segment_or_item']} IN ({inspire_sql_list(picked)})"


def inspire_technique_records(technique):
    if INSPIRE_TECHNIQUE_RECORDS is None and INSPIRE_TECHNIQUE_RECORDS_LOADER is None:
        raise RuntimeError("Inspire's technique table is not loaded, so it cannot tell whether a technique works in groups")
    table = INSPIRE_TECHNIQUE_RECORDS if INSPIRE_TECHNIQUE_RECORDS is not None else INSPIRE_TECHNIQUE_RECORDS_LOADER()
    label = " ".join(str(technique if technique is not None else "").split()).lower()
    return next((str(kind) for name, kind in table.items() if name.lower() == label), "")


def inspire_grouping_applies(technique):
    return inspire_technique_records(technique) == INSPIRE_RECORDS_GROUPS


INSPIRE_ONE_MODEL_REFUSED = ("this use case fits one model across all of the data, and Inspire registers one model per use case, so "
                             "splitting the data into groups would need one registered model per group, which Inspire does not register; "
                             "choose all, the whole data as one group")




def inspire_group_by_parts(text):
    picked = []
    for part in str("" if text is None else text).split(","):
        name = part.strip()
        if name and name.lower() not in [seen.lower() for seen in picked]:
            picked.append(name)
    return picked


def inspire_group_by_mode(text):
    picked = inspire_group_by_parts(text)
    lowered = [name.lower() for name in picked]
    if INSPIRE_GROUP_CUSTOM in lowered:
        return INSPIRE_GROUP_CUSTOM, picked
    if INSPIRE_SETTING_ALL in lowered and len(picked) > 1:
        return "mixed", picked
    if not picked or lowered == [INSPIRE_SETTING_ALL]:
        return INSPIRE_SETTING_ALL, []
    return "columns", picked


def inspire_group_by_plain(text):
    kept = [name for name in inspire_group_by_parts(text) if name.lower() != INSPIRE_GROUP_CUSTOM]
    return ", ".join(kept) if kept else INSPIRE_SETTING_ALL


def inspire_group_by_columns(text):
    mode, picked = inspire_group_by_mode(text)
    return picked if mode == "columns" else []


def inspire_group_minimum(time_series):
    return INSPIRE_GROUP_MIN_PERIODS if time_series else INSPIRE_GROUP_MIN_RECORDS


def inspire_group_part(column, value):
    if value is None or (isinstance(value, float) and value != value):
        return f"no {column}"
    return str(value).strip() or f"no {column}"


def inspire_effective_groups(group_by, groups, time_series=False):
    columns = inspire_group_by_columns(group_by)
    rows = [{"path": list(item.get("path") or []), "records": int(item.get("records") or 0),
             "periods": set(range(int(item["periods"][0]), int(item["periods"][1]) + 1)) if time_series else set()}
            for item in groups or []]
    settled = [None] * len(rows)
    for depth in range(len(columns), 0, -1):
        keys = [tuple(inspire_group_part(column, value) for column, value in zip(columns[:depth], row["path"])) for row in rows]
        pools = {}
        for at, row in enumerate(rows):
            if settled[at] is None:
                pool = pools.setdefault(keys[at], {"records": 0, "periods": set()})
                pool["records"] += row["records"]
                pool["periods"] |= row["periods"]
        for at in range(len(rows)):
            pool = pools.get(keys[at]) if settled[at] is None else None
            if pool and (len(pool["periods"]) if time_series else pool["records"]) >= inspire_group_minimum(time_series):
                settled[at] = (INSPIRE_GROUP_SEPARATOR.join(keys[at]), columns[depth - 1])
    return [{"path": row["path"], "group": (settled[at] or (INSPIRE_GROUP_GLOBAL, INSPIRE_SETTING_ALL))[0],
             "level": (settled[at] or (INSPIRE_GROUP_GLOBAL, INSPIRE_SETTING_ALL))[1]} for at, row in enumerate(rows)]


def inspire_group_column_usable(profile):
    groups, records = int((profile or {}).get("groups") or 0), int((profile or {}).get("records") or 0)
    return INSPIRE_GROUP_MIN_GROUPS <= groups <= INSPIRE_GROUP_MAX_GROUPS and records >= groups * INSPIRE_GROUP_MIN_RECORDS


def inspire_group_recommendation(profiles, nests=()):
    nested = {(str(coarse).lower(), str(fine).lower()) for coarse, fine, holds in nests or () if holds}
    usable = sorted((p for p in profiles or [] if inspire_group_column_usable(p)), key=lambda p: (int(p["groups"]), str(p["column"]).lower()))
    picked = []
    for profile in usable:
        if not picked or (str(picked[-1]).lower(), str(profile["column"]).lower()) in nested:
            picked.append(str(profile["column"]))
    return picked


class InspireRunStop(BaseException):
    pass


def inspire_group_stop(why, run=None, step="Grouping"):
    run = (globals().get("INSPIRE_RUN") or {}) if run is None else run
    link = str(((run or {}).get("grouping") or {}).get("link") or "")
    return InspireRunStop(f"{step}: {why}, so this run stopped before it scored any record; change the {step.lower()} on the Run page"
                          + (f": {link}" if link else " of this use case in the Inspire App"))


def inspire_group_hash(mode, columns, time_column, minimum):
    shown = {"mode": mode, "group_by": [str(c).lower() for c in columns], "time_column": str(time_column or "").lower(), "minimum": minimum}
    return _inspire_hashlib.sha256(_inspire_json.dumps(shown, sort_keys=True).encode("utf-8")).hexdigest()


def _inspire_group_pandas(frame, columns, time_column):
    import pandas as _inspire_pd
    out = frame.copy()
    parts = []
    for column in columns:
        text = out[column].astype("string").str.strip()
        parts.append(text.where(text.notna() & (text != ""), f"no {column}").astype(object))
    group = _inspire_pd.Series([None] * len(out), index=out.index, dtype=object)
    level = _inspire_pd.Series([None] * len(out), index=out.index, dtype=object)
    for depth in range(len(columns), 0, -1):
        key = parts[0]
        for part in parts[1:depth]:
            key = key + INSPIRE_GROUP_SEPARATOR + part
        open_rows = group.isna()
        grouped = (out.loc[open_rows, time_column] if time_column else key[open_rows]).groupby(key[open_rows])
        sizes = grouped.nunique() if time_column else grouped.size()
        hit = open_rows & key.isin(set(sizes.index[sizes >= inspire_group_minimum(bool(time_column))]))
        group = group.where(~hit, key)
        level = level.where(~hit, columns[depth - 1])
    out[INSPIRE_GROUP_COLUMN] = group.where(group.notna(), INSPIRE_GROUP_GLOBAL)
    out[INSPIRE_GROUP_LEVEL_COLUMN] = level.where(level.notna(), INSPIRE_SETTING_ALL)
    counts = {}
    for name, part in out.groupby(INSPIRE_GROUP_LEVEL_COLUMN)[INSPIRE_GROUP_COLUMN]:
        counts[str(name)] = (int(part.nunique()), int(len(part)))
    return out, counts


def _inspire_group_spark(frame, columns, time_column):
    from pyspark.sql import functions as _inspire_f
    parts = []
    for column in columns:
        text = _inspire_f.trim(_inspire_f.col(inspire_column(column)).cast("string"))
        parts.append(_inspire_f.when(text.isNull() | (text == ""), _inspire_f.lit(f"no {column}")).otherwise(text))
    work = (frame.withColumn(INSPIRE_GROUP_COLUMN, _inspire_f.lit(None).cast("string"))
            .withColumn(INSPIRE_GROUP_LEVEL_COLUMN, _inspire_f.lit(None).cast("string")))
    size = _inspire_f.countDistinct(_inspire_f.col(inspire_column(time_column))) if time_column else _inspire_f.count(_inspire_f.lit(1))
    for depth in range(len(columns), 0, -1):
        keyed = work.withColumn(INSPIRE_GROUP_KEY, _inspire_f.concat_ws(INSPIRE_GROUP_SEPARATOR, *parts[:depth]))
        passing = (keyed.filter(_inspire_f.col(INSPIRE_GROUP_COLUMN).isNull()).groupBy(INSPIRE_GROUP_KEY)
                   .agg(size.alias(INSPIRE_GROUP_SIZE))
                   .filter(_inspire_f.col(INSPIRE_GROUP_SIZE) >= inspire_group_minimum(bool(time_column)))
                   .select(INSPIRE_GROUP_KEY, _inspire_f.lit(True).alias(INSPIRE_GROUP_PASS)))
        hit = _inspire_f.col(INSPIRE_GROUP_COLUMN).isNull() & _inspire_f.coalesce(_inspire_f.col(INSPIRE_GROUP_PASS), _inspire_f.lit(False))
        work = (keyed.join(passing, on=INSPIRE_GROUP_KEY, how="left")
                .withColumn(INSPIRE_GROUP_LEVEL_COLUMN,
                            _inspire_f.when(hit, _inspire_f.lit(columns[depth - 1])).otherwise(_inspire_f.col(INSPIRE_GROUP_LEVEL_COLUMN)))
                .withColumn(INSPIRE_GROUP_COLUMN,
                            _inspire_f.when(hit, _inspire_f.col(INSPIRE_GROUP_KEY)).otherwise(_inspire_f.col(INSPIRE_GROUP_COLUMN)))
                .drop(INSPIRE_GROUP_KEY, INSPIRE_GROUP_PASS))
    work = (work.withColumn(INSPIRE_GROUP_COLUMN, _inspire_f.coalesce(_inspire_f.col(INSPIRE_GROUP_COLUMN), _inspire_f.lit(INSPIRE_GROUP_GLOBAL)))
            .withColumn(INSPIRE_GROUP_LEVEL_COLUMN,
                        _inspire_f.coalesce(_inspire_f.col(INSPIRE_GROUP_LEVEL_COLUMN), _inspire_f.lit(INSPIRE_SETTING_ALL))))
    rows = (work.groupBy(INSPIRE_GROUP_LEVEL_COLUMN).agg(_inspire_f.countDistinct(INSPIRE_GROUP_COLUMN).alias("groups"),
                                                         _inspire_f.count(_inspire_f.lit(1)).alias("records")).collect())
    return work, {str(row[INSPIRE_GROUP_LEVEL_COLUMN]): (int(row["groups"]), int(row["records"])) for row in rows}


def inspire_group_record_text(record):
    levels = record.get("levels") or []
    if record.get("group_by") == INSPIRE_SETTING_ALL:
        return f"{inspire_plural(record.get('groups'), 'group')}, {INSPIRE_GROUP_GLOBAL}, since Data groups is {INSPIRE_SETTING_ALL}"
    shown = "; ".join(f"{inspire_plural(item['groups'], 'group')} by {item['level']}" for item in levels)
    return (f"{inspire_plural(record.get('groups'), 'group')} by {', '.join(record.get('group_by') or [])}"
            + (f" ({shown})" if len(levels) > 1 else "")
            + f"; a group with fewer than {inspire_plural(record.get('minimum'), 'period' if record.get('time_column') else 'record')} "
            "moves up a level")


def inspire_group_column_of(name, have):
    low = str(name).strip().lower()
    return have.get(low) or (have.get(low.rsplit(".", 1)[-1]) if "." in low else None)


def inspire_group_join(name, join):
    if not isinstance(join, dict) or not all(str(join.get(part) or "").strip() for part in ("table", "key", "ref_column")):
        return None
    column = str(name).strip().rsplit(".", 1)[-1]
    ref, key, value, named = (inspire_column(join["ref_column"]), inspire_column(INSPIRE_GROUP_JOIN_KEY), inspire_column(INSPIRE_GROUP_JOIN_VALUE),
                              inspire_column(column))
    return {"table": str(join["table"]), "key": str(join["key"]), "ref_column": str(join["ref_column"]), "column": column,
            "where": f"{ref} IS NOT NULL", "select": [f"CAST({ref} AS STRING) AS {key}", f"{named} AS {value}"],
            "sql": (f"SELECT {key}, MAX({value}) AS {named} FROM (SELECT CAST({ref} AS STRING) AS {key}, {named} AS {value} FROM "
                    f"{inspire_table_name(join['table'])} WHERE {ref} IS NOT NULL) AS d GROUP BY {key}")}


def inspire_group_join_frame(join, dimension):
    return (dimension.where(join["where"]).selectExpr(*join["select"]).groupBy(INSPIRE_GROUP_JOIN_KEY)
            .agg({INSPIRE_GROUP_JOIN_VALUE: "max"}).toDF(INSPIRE_GROUP_JOIN_KEY, join["column"]))


def inspire_group_joined(frame, wanted, out=print, spark_session=None, run=None):
    run = (globals().get("INSPIRE_RUN") or {}) if run is None else run
    joins = {str(k).strip().lower(): v for k, v in (((run or {}).get("grouping") or {}).get("joins") or {}).items() if isinstance(v, dict)}
    have = {str(name).lower(): str(name) for name in frame.columns}
    spark_frame = hasattr(frame, "groupBy") and hasattr(frame, "withColumn")
    for name in wanted:
        if "." not in str(name) or inspire_group_column_of(name, have) is not None:
            continue
        join = inspire_group_join(name, joins.get(str(name).strip().lower()))
        if join is None:
            raise inspire_group_stop(f"the Grouping names {name}, a column of another table, but this run's plan carries no join to "
                                     f"that table, so the records have no {name} to group by", run)
        key = have.get(join["key"].lower())
        if key is None:
            raise ValueError(f"group_by names {name}, which inspire_group_records reads from {join['table']} on the key {join['key']}, "
                             f"but the records passed to inspire_group_records have no column {join['key']}; keep {join['key']} in the records")
        bare = join["column"]
        session = spark_session if spark_session is not None else globals().get("spark")
        try:
            dim = inspire_group_join_frame(join, inspire_read_source(join["table"], session))
            if spark_frame:
                frame = frame.join(dim, frame[key].cast("string") == dim[INSPIRE_GROUP_JOIN_KEY], "left").drop(INSPIRE_GROUP_JOIN_KEY)
            else:
                table = dim.toPandas()
                table = table.assign(**{INSPIRE_GROUP_JOIN_KEY: table[INSPIRE_GROUP_JOIN_KEY].astype("string")})
                frame = (frame.assign(**{INSPIRE_GROUP_JOIN_KEY: frame[key].astype("string")})
                         .merge(table, how="left", on=INSPIRE_GROUP_JOIN_KEY).drop(columns=[INSPIRE_GROUP_JOIN_KEY]).set_axis(frame.index))
            matched = frame.filter(frame[bare].isNotNull()).limit(1).count() if spark_frame else int(frame[bare].notna().sum())
        except Exception as error:
            raise inspire_group_stop(f"the Grouping names {name}, but Inspire could not join {join['table']} on {join['key']} = "
                                     f"{join['ref_column']} to read it ({inspire_error_line(error)[:INSPIRE_RUN_REASON_CHARS]})", run)
        if not matched:
            raise inspire_group_stop(f"the Grouping names {name}, but no {key} of the records matches {join['ref_column']} in {join['table']}, "
                                     f"so every record would fall in the one {INSPIRE_GROUP_GLOBAL} group", run)
        have[bare.lower()] = bare
        out(f"[GROUP-JOIN FIRED] Grouping {name}: joined {join['table']} on {key} = {join['ref_column']} to build the groups only; "
            f"the records come back without {bare}")
    return frame


INSPIRE_COUNT_ATTEMPTS = 2


def inspire_retry_count(count):
    """Runs a count. A failure is tried once more; a second failure is raised, never answered as a pass."""
    for attempt in range(INSPIRE_COUNT_ATTEMPTS):
        try:
            return count()
        except Exception:
            if attempt == INSPIRE_COUNT_ATTEMPTS - 1:
                raise


def inspire_group_distinct_values(frame, column, spark_frame):
    """The values of a Grouping column in the records, counted the way the build counts them inside the scope."""
    if spark_frame:
        from pyspark.sql import functions as _inspire_f
        text = _inspire_f.trim(_inspire_f.col(inspire_column(column)).cast("string"))
        key = _inspire_f.when(text.isNull() | (text == ""), _inspire_f.lit(f"no {column}")).otherwise(text)
        return int(frame.select(key.alias("inspire_value")).distinct().count())
    text = frame[column].astype("string").str.strip()
    return int(text.where(text.notna() & (text != ""), f"no {column}").nunique())


def inspire_one_value_words(columns):
    """The plain reason a Grouping column that holds one value is refused, and the choice that stands in its place."""
    return (" and ".join(f"{name} has one value in this data, so it makes one group" for name in columns)
            + f"; choose {INSPIRE_SETTING_ALL}, the whole data as one group")


def inspire_scopes_in_force(scopes=None) -> dict:
    return {**INSPIRE_TABLE_SCOPES, **(scopes or {})}


def inspire_scoped_read_source(name, scopes=None, alias=None) -> str:
    scope = inspire_scopes_in_force(scopes).get(inspire_table_scope_key(name))
    if scope is None:
        plain = inspire_quote_qualified_name(name)
        return f"{plain} AS {alias}" if alias else plain
    if scope.get("ready"):
        return f"{scope['view_ref']} AS {alias}" if alias else scope["view_ref"]
    return f"{scope['source']} AS {alias or inspire_quote_identifier(str(name).split('.')[-1])}"


def inspire_implement_one_value_columns(spark, logger, uc_id, table, scopes, columns, joins=None) -> tuple:
    """The columns that hold one value in the scope. A column in `joins` is counted through its join: the table's records read
    through the join's table, counted on the joined value. Only COUNT(DISTINCT) comes back, one per column."""
    one, unchecked = [], []
    for column in columns:
        spec = (joins or {}).get(column)
        if spec is None:
            sql = (f"SELECT COUNT(DISTINCT COALESCE(NULLIF(TRIM(CAST({inspire_column(column)} AS STRING)), ''), 'no {column}')) AS distinct_values "
                   f"FROM {inspire_scoped_read_source(table, scopes)}")
        else:
            join, key = spec
            ref, named = inspire_column(join["ref_column"]), inspire_column(join["column"])
            sql = (f"SELECT COUNT(DISTINCT COALESCE(NULLIF(TRIM(CAST(g.{named} AS STRING)), ''), 'no {join['column']}')) AS distinct_values "
                   f"FROM {inspire_scoped_read_source(table, scopes, 'm')} LEFT JOIN (SELECT CAST({ref} AS STRING) AS inspire_join_key, MAX({named}) AS {named} "
                   f"FROM {inspire_scoped_read_source(join['table'], scopes)} WHERE {ref} IS NOT NULL GROUP BY CAST({ref} AS STRING)) g "
                   f"ON CAST(m.{inspire_column(key)} AS STRING) = g.inspire_join_key")
        def count(sql=sql):
            rows = inspire_rows(spark.sql(sql))
            return int((rows[0] if rows else {}).get("distinct_values") or 0)
        try:
            value = inspire_retry_count(count)
        except Exception as error:
            logger.warning(f"[IMPLEMENT-GROUPING] {uc_id}: could not count the values of {column} in {table} after one retry: "
                                f"{inspire_redact_text(str(error), set(), table)[:300]}")
            unchecked.append(column)
            continue
        if value == 1:
            one.append(column)
    return one, unchecked


def inspire_group_launch_one_value(session, table, columns, joins=None):
    """The Grouping columns that hold one value in the Launch scope of their table: the Data Filters the wizard counts, before a First run
    narrows the records. A joined column is counted through its join. Returns (one_value_columns, unchecked_columns)."""
    import logging
    return inspire_implement_one_value_columns(session, logging.getLogger("inspire"), "Grouping", table, INSPIRE_TABLE_SCOPES, columns, joins)


def inspire_group_one_value_held(frame, columns, spark_frame, group_launch=None):
    """The Grouping columns that hold one value. A First run narrows the records, so when `group_launch` (from inspire_group_launch) is
    given, a column of the table or a column that a join brings in is counted in the Launch scope, the count the wizard made. A count
    that fails twice raises, so the run stops."""
    launch_one = {}
    if group_launch is not None:
        counted = [column for column in columns if column in group_launch["passed"] or column in group_launch["joins"]]
        joins = {column: group_launch["joins"][column] for column in counted if column in group_launch["joins"]}
        one, unchecked = inspire_group_launch_one_value(group_launch["session"], group_launch["table"], counted, joins)
        if unchecked:
            raise RuntimeError(f"could not count the values of {', '.join(unchecked)} in the Launch scope")
        launch_one = {column: column in one for column in counted}
    held = []
    for column in columns:
        if column in launch_one:
            is_one = launch_one[column]
        else:
            is_one = inspire_retry_count(lambda column=column: inspire_group_distinct_values(frame, column, spark_frame)) == 1
        if is_one:
            held.append(column)
    return held


def inspire_register_schema(frame, extra=()):
    """Column names are schema, not data. A message that quotes one keeps it, so every frame the run reads registers its names, and
    the Grouping names it was given (a joined name such as table.column)."""
    try:
        names = [str(name) for name in frame.columns]
    except Exception:
        return
    globals().setdefault("INSPIRE_SCHEMA_NAMES", set()).update(names + [str(name) for name in extra])


def inspire_group_launch(spark_session, passed, wanted, named, have):
    """What the one-value check needs to read the Launch scope when a First run narrows the records: the session, the table, the
    records' own columns, and each joined Grouping column with its join and its key. None when the records are the Launch scope."""
    scope = globals().get("INSPIRE_SCOPE") or {}
    if not scope.get("table") or scope.get("scope") in (None, INSPIRE_SCOPES[0]):
        return None
    joins = {str(k).strip().lower(): v for k, v in (((globals().get("INSPIRE_RUN") or {}).get("grouping") or {}).get("joins") or {}).items()
             if isinstance(v, dict)}
    group_launch = {"session": spark_session if spark_session is not None else globals().get("spark"), "table": scope["table"],
              "passed": set(passed), "joins": {}}
    for name, found in zip(wanted, named):
        join = inspire_group_join(name, joins.get(str(name).strip().lower())) if "." in str(name) else None
        if found is not None and join is not None and found not in passed:
            group_launch["joins"][found] = (join, have.get(join["key"].lower()))
    return group_launch


def inspire_group_records(frame, group_by=None, time_column=None, out=print, spark_session=None):
    text = (inspire_widget(INSPIRE_GROUP_BY, INSPIRE_SETTING_ALL) if group_by is None
            else ", ".join(str(name) for name in group_by) if isinstance(group_by, (list, tuple)) else str(group_by))
    run_grouping = (globals().get("INSPIRE_RUN") or {}).get("grouping") or {}
    if run_grouping.get("one_model") and inspire_group_by_mode(text)[0] != INSPIRE_SETTING_ALL:
        raise inspire_group_stop(f"the Grouping is {text!r}: " + INSPIRE_ONE_MODEL_REFUSED)
    spark_frame = hasattr(frame, "groupBy") and hasattr(frame, "withColumn")
    if not spark_frame and not hasattr(frame, "groupby"):
        raise TypeError("inspire_group_records takes the records as a Spark or a pandas DataFrame")
    mode, wanted = inspire_group_by_mode(text)
    if mode == INSPIRE_GROUP_CUSTOM:
        text = inspire_group_by_plain(text)
        out(INSPIRE_CUSTOM_SQL_PLAIN.format(part="Grouping", plain=f"the Grouping {text}"))
        mode, wanted = inspire_group_by_mode(text)
    if mode == "mixed":
        raise inspire_group_stop(f"the Grouping is {text!r}, but {INSPIRE_SETTING_ALL} stands alone")
    else:
        passed = [str(name) for name in frame.columns]
        frame = inspire_group_joined(frame, wanted, out, spark_session)
        have = {str(name).lower(): str(name) for name in frame.columns}
        named = [inspire_group_column_of(name, have) for name in wanted]
        missing = [name for name, found in zip(wanted, named) if found is None]
        if missing:
            raise ValueError(f"group_by is {text!r}, but the records passed to inspire_group_records have no column "
                             f"{', '.join(missing)}; group_by takes these records' own column names, coarse to fine, such as "
                             f"region, country, or {INSPIRE_SETTING_ALL}")
        inspire_register_schema(frame, wanted)
        group_launch = inspire_group_launch(spark_session, passed, wanted, named, have)
        try:
            held = inspire_group_one_value_held(frame, named, spark_frame, group_launch)
        except Exception as error:
            raise inspire_group_stop(f"the Grouping {text!r} could not be checked for one value in this data "
                                     f"({inspire_error_line(error)[:INSPIRE_RUN_REASON_CHARS]})")
        if held and len(held) == len(named):
            raise inspire_group_stop(f"the Grouping is {text!r}: " + inspire_one_value_words(held))
        time_name = have.get(str(time_column).lower()) if time_column else None
        if time_column and time_name is None:
            raise ValueError(f"time_column is {time_column!r}, but the records passed to inspire_group_records have no such column")
        grouped, counts = (_inspire_group_spark if spark_frame else _inspire_group_pandas)(frame, named, time_name)
        joined = [str(name) for name in grouped.columns
                  if str(name) not in passed and str(name) not in (INSPIRE_GROUP_COLUMN, INSPIRE_GROUP_LEVEL_COLUMN)]
        if joined:
            grouped = grouped.drop(*joined) if spark_frame else grouped.drop(columns=joined)
        levels = [{"level": name, "groups": counts[name][0], "records": counts[name][1]} for name in named + [INSPIRE_SETTING_ALL]
                  if name in counts]
        minimum = inspire_group_minimum(bool(time_name))
        record = {"mode": "columns" if named else INSPIRE_SETTING_ALL, "group_by": named or INSPIRE_SETTING_ALL,
                  "time_column": time_name or "", "minimum": minimum, "groups": sum(item["groups"] for item in levels), "levels": levels,
                  "hash": inspire_group_hash("columns" if named else INSPIRE_SETTING_ALL, named, time_name, minimum)}
    globals()["INSPIRE_RUN_GROUPING"] = record
    out(f"Data groups: {inspire_group_record_text(record)}")
    return grouped


def inspire_run_problems(profile, value=None):
    problems = []
    if int(profile.get("negative_stake") or 0):
        problems.append(f"{inspire_plural(profile['negative_stake'], 'row has', 'rows have')} a {INSPIRE_VALUE_AT_STAKE} below 0; it holds the "
                        f"{inspire_stake_noun(value)} the record puts at stake, 0 when nothing is at stake")
    if int(profile.get("unshown_stake") or 0):
        problems.append(f"{inspire_plural(profile['unshown_stake'], 'row has', 'rows have')} a {INSPIRE_VALUE_AT_STAKE} but NULL in a column the "
                        f"{INSPIRE_VALUE_FORMULA_NAME} names, so the App cannot show how it was worked out; a record the formula "
                        f"cannot price has a NULL {INSPIRE_VALUE_AT_STAKE}")
    if int(profile.get("off_expression") or 0):
        problems.append(f"{inspire_plural(profile['off_expression'], 'row has', 'rows have')} a {INSPIRE_VALUE_AT_STAKE} that {INSPIRE_VALUE_EXPRESSION_NAME} does not "
                        f"give from the row's own numbers, so the App would show other {inspire_stake_noun(value)} when someone "
                        f"changes an assumption; write {INSPIRE_VALUE_AT_STAKE} with the arithmetic of {INSPIRE_VALUE_EXPRESSION_NAME}")
    if int(profile.get("unaccepted_flagged") or 0):
        problems.append(f"{inspire_plural(profile['unaccepted_flagged'], 'row is', 'rows are')} 'Act now' or 'Review', but the run passed accepted=False: this use "
                        "case acts on a prediction, so no record is actionable until a model passes acceptance; write every row "
                        "'Monitor' and pass nothing_to_act with the metrics that missed")
    cuts = inspire_share_cuts(value)
    if cuts and int(profile.get("off_share_tier") or 0):
        problems.append(f"{inspire_plural(profile['off_share_tier'], 'row has', 'rows have')} a priority_tier that does not follow the share rule: with "
                        f"{inspire_plural(value['passed'], 'row that passes', 'rows that pass')} the decision test, ranked 1 to {inspire_number_text(value['passed'], grouped=True)} first, a row whose "
                        f"data_quality_flag is 'ok' is 'Act now' at rank {cuts[0]} or better, 'Review' at rank {cuts[1]} or better, "
                        "and every other row is 'Monitor' (when every passing row is flagged, no more than the larger of "
                        f"{INSPIRE_FULL_SHARE_ACT_MIN} rows and {INSPIRE_FULL_SHARE_ACT_PERCENT}% of the passing rows are 'Act now')")
    if cuts and int(profile.get("ok_rows") or 0) < int(value["passed"]):
        problems.append(f"the run passed passed={value['passed']}, but only {inspire_plural(profile.get('ok_rows') or 0, 'row has', 'rows have')} a "
                        "data_quality_flag of 'ok'; pass the number of rows that pass the decision test")
    if int(profile.get("off_tier") or 0):
        act_now, review = inspire_value_amounts(value)
        problems.append(f"{inspire_plural(profile['off_tier'], 'row has', 'rows have')} a priority_tier that does not follow {'its' if profile['off_tier'] == 1 else 'their'} {INSPIRE_VALUE_AT_STAKE}: this "
                        f"use case acts on measured amounts, so a row whose data_quality_flag is 'ok' is 'Act now' at "
                        f"{inspire_money_text(act_now, inspire_value_unit(value))} or more, 'Review' at "
                        f"{inspire_money_text(review, inspire_value_unit(value))} or more"
                        + (f", in a segment flag_segments names ({', '.join(inspire_flag_segments(value))})" if inspire_flag_segments(value) else "")
                        + ", and every other row is 'Monitor'")
    if int(profile.get("no_use_case") or 0) or int(profile.get("use_cases") or 0) != 1:
        problems.append("every row must carry this use case's id in use_case_id, and only that id")
    if int(profile.get("no_run") or 0) or int(profile.get("runs") or 0) != 1:
        problems.append("every row must carry this run's one run_id")
    if int(profile.get("undated") or 0):
        problems.append(f"{inspire_plural(profile['undated'], 'row has', 'rows have')} no as_of_date; write the resolved as-of date on every row")
    if int(profile.get("bad_levels") or 0):
        problems.append(f"{inspire_plural(profile['bad_levels'], 'row has', 'rows have')} a {INSPIRE_CONFIDENCE} other than "
                        f"{', '.join(repr(level) for level in INSPIRE_CONFIDENCE_LEVELS)}; set it on every row with the evidence rule")
    if int(profile.get("bare_flagged") or 0):
        problems.append(f"{inspire_plural(profile['bare_flagged'], 'row is', 'rows are')} 'Act now' or 'Review' with no {INSPIRE_AI_FACTS}, the facts Inspire "
                        "explains each flagged record from; build the facts on every row")
    return problems

INSPIRE_WRITE_CONFLICT_RE = _inspire_re.compile(
    r"DELTA_CONCURRENT_\w+|DELTA_(METADATA|PROTOCOL)_CHANGED|CONCURRENT_(APPEND|DELETE|TRANSACTION|WRITE)|"
    r"Concurrent\w*Exception|(Metadata|Protocol)ChangedException|changed by a concurrent update", _inspire_re.I)
INSPIRE_WRITE_CONFLICT_WINDOW_SECONDS = 120.0
INSPIRE_WRITE_CONFLICT_BASE_SECONDS = 0.5
INSPIRE_WRITE_CONFLICT_MAX_WAIT_SECONDS = 10.0
INSPIRE_WRITE_WAIT_SECONDS = 2 * INSPIRE_WRITE_CONFLICT_WINDOW_SECONDS


class InspireWriteConflict(RuntimeError):
    attempts, seconds, error = 0, 0.0, ""


def inspire_write_conflict(what, attempts, seconds, error_text) -> InspireWriteConflict:
    conflict = InspireWriteConflict(f"{what} conflicted with other Inspire runs writing the same table at the same time, "
                                    f"{inspire_plural(attempts, 'attempt')} over {seconds:.0f}s; this is a conflict between runs, "
                                    f"not missing write access ({error_text})")
    conflict.attempts, conflict.seconds, conflict.error = attempts, seconds, error_text
    return conflict


def inspire_is_write_conflict(error) -> bool:
    return error is not None and bool(INSPIRE_WRITE_CONFLICT_RE.search(f"{type(error).__name__}: {error}"))


def inspire_write_conflict_attempts(window=INSPIRE_WRITE_CONFLICT_WINDOW_SECONDS, base=INSPIRE_WRITE_CONFLICT_BASE_SECONDS,
                                    cap=INSPIRE_WRITE_CONFLICT_MAX_WAIT_SECONDS) -> int:
    attempts, spent = 1, 0.0
    while True:
        wait = 1.5 * min(float(base) * (2 ** (attempts - 1)), float(cap))
        if spent + wait > float(window):
            return attempts
        spent += wait
        attempts += 1


def inspire_retry_write_conflicts(write, what, window=INSPIRE_WRITE_CONFLICT_WINDOW_SECONDS, base=INSPIRE_WRITE_CONFLICT_BASE_SECONDS,
                                  max_attempts=None, on_retry=None, sleep=None):
    import random
    import time
    attempts = max(1, int(max_attempts) if max_attempts else inspire_write_conflict_attempts(window, base))
    began = time.monotonic()
    for attempt in range(1, attempts + 1):
        try:
            return write()
        except Exception as error:
            if isinstance(error, InspireWriteConflict) or not inspire_is_write_conflict(error):
                raise
            text = inspire_redact_text(str(error), set(), "")[:300]
            if attempt >= attempts:
                raise inspire_write_conflict(what, attempts, time.monotonic() - began, text) from error
            ceiling = min(float(INSPIRE_WRITE_CONFLICT_MAX_WAIT_SECONDS), float(base) * (2 ** (attempt - 1)))
            wait = ceiling + random.random() * ceiling
            if on_retry is not None:
                on_retry(attempt, wait, text)
            (sleep or time.sleep)(wait)


def inspire_column_names(spark, result_table):
    names = set()
    for row in inspire_rows(spark.sql(f"DESCRIBE TABLE {inspire_table_name(result_table)}")):
        name = str(row.get("col_name") or "").strip()
        if not name or name.startswith("#"):
            break
        names.add(name.lower())
    return names


def inspire_add_missing_columns(spark, result_table, columns, existing=None, quote=True, attempts=None,
                                backoff_seconds=INSPIRE_WRITE_CONFLICT_BASE_SECONDS):
    existing = inspire_column_names(spark, result_table) if existing is None else existing
    added, failed = [], []
    for name, kind in columns:
        if str(name).lower() in existing:
            continue
        column = inspire_column(name) if quote else name

        def _add(name=name, column=column, kind=kind) -> bool:
            nonlocal existing
            try:
                spark.sql(f"ALTER TABLE {inspire_table_name(result_table)} ADD COLUMN {column} {kind}")
                return True
            except Exception as error:
                low = str(error).lower()
                try:
                    existing = inspire_column_names(spark, result_table)
                except Exception:
                    pass
                if str(name).lower() in existing or "already exists" in low or "duplicate" in low:
                    return False
                raise
        try:
            if inspire_retry_write_conflicts(_add, f"adding column {name} to {inspire_table_name(result_table)}",
                                             base=backoff_seconds, max_attempts=attempts):
                added.append(name)
            existing.add(str(name).lower())
        except Exception as error:
            failed.append((name, inspire_redact_text(str(getattr(error, "error", "") or error), set(), "")[:200]))
    return added, failed, existing


def inspire_add_required_columns(spark, result_table, columns, existing=None):
    added, failed, _existing = inspire_add_missing_columns(spark, result_table, columns, existing=existing)
    if failed:
        raise RuntimeError(f"Inspire could not add {inspire_plural(len(failed), 'column')} to {inspire_table_name(result_table)}: "
                           + "; ".join(f"{name} ({error})" for name, error in failed))
    return added


def inspire_type_change_sql(table, table_schema, run_schema, replace_edits):
    wanted = {str(name).lower(): kind for name, kind in run_schema}
    picked, changed = [], []
    for name, kind in table_schema:
        target = wanted.get(str(name).lower())
        column = inspire_column(name, "r")
        have, want = kind.replace(" ", "").lower(), (target or "").replace(" ", "").lower()
        if target is None or have == want:
            picked.append(column)
            continue
        if have.startswith("struct<") and ("<result:" in have or ",result:" in have) and want == "string":
            value = f"CAST({column}.result AS STRING)"
        elif have == "string" and want == "array<string>":
            value = f"IF({column} IS NULL, NULL, array({column}))"
        else:
            value = f"TRY_CAST({column} AS {target})"
        picked.append(f"{value} AS {inspire_column(name)}")
        changed.append(f"{name} ({kind} to {target})")
    if not changed:
        return "", []
    flag = inspire_sql_literal("true" if replace_edits else "false")
    return (f"CREATE OR REPLACE TABLE {inspire_table_name(table)} TBLPROPERTIES ({inspire_sql_literal(INSPIRE_REPLACE_EDITS_PROPERTY)} = "
            f"{flag}) AS SELECT {', '.join(picked)} FROM {inspire_table_name(table)} AS r", changed)

def inspire_history_merge_sql(table, stage, columns, key_names, use_case_id, replace_edits, ai=None, derived=None, now="current_timestamp()",
                              kept=(), kinds=None, stamp=True):
    target, source = inspire_table_name(table), inspire_table_name(stage)
    key = inspire_record_key_sql(key_names, "s")
    picked = inspire_stage_columns_sql(columns, derived, "s")
    reasoning = [name for name, _kind in inspire_ai_columns(ai)]
    same_facts = f"p.ai_facts_hash = {inspire_ai_hash_sql(ai['fingerprint'], 's')}" if ai else ""
    carried = ", ".join([inspire_column(c, "p") for c in INSPIRE_ACTION_NAMES]
                        + [f"CASE WHEN {same_facts} THEN {inspire_column(c, 'p')} END AS {inspire_column(c)}" for c in reasoning])
    close = f"SET valid_to = {now}, is_current = FALSE"
    latest = (f"SELECT * FROM {target} AS r WHERE r.use_case_id = {inspire_sql_literal(use_case_id)} QUALIFY ROW_NUMBER() "
              f"OVER (PARTITION BY r.record_key ORDER BY r.is_current DESC, r.valid_from DESC) = 1")
    tier = inspire_tier_column(columns)
    kinds = {str(name).lower(): kind for name, kind in (kinds or [])}

    def newest(alias):
        status, shown = inspire_column("action_status", alias), inspire_column(tier, alias) if tier else "NULL"
        values = [f"CASE WHEN {status} = {inspire_sql_literal(inspire_shell_facts.DISMISSED)} THEN 'Monitor' ELSE {shown} END" if c == tier else inspire_column(c, alias)
                  for c in columns]
        judged = inspire_column("run_tier", alias) if "run_tier" in (derived or {}) else shown
        measured = inspire_sql_list(inspire_shell_facts.MEASURED_STATUSES)
        outcome = (f"CASE WHEN {status} IN ({measured}) AND {judged} IN ({inspire_sql_list(INSPIRE_FLAGGED_TIERS)}) THEN 'Still flagged' "
                   f"WHEN {status} IN ({measured}) THEN 'Resolved' ELSE {inspire_column('action_outcome', alias)} END")
        return values, outcome

    values, outcome = newest("u")
    staged, staged_outcome = newest("q")
    actions = ", ".join(outcome if c == "action_outcome" else
                        f"CASE WHEN u._inspire_was_source IS NULL THEN {now} ELSE {inspire_column(c, 'u')} END" if c == INSPIRE_JOINED_AT
                        else "CAST(NULL AS STRING)" if c in INSPIRE_ACT_BATCH_NAMES else inspire_column(c, "u") for c in INSPIRE_ACTION_NAMES)
    stamps = [c for c in columns if str(c).lower() in INSPIRE_RUN_STAMPS]
    compared = [(value, c) for value, c in zip(staged, columns) if str(c).lower() not in INSPIRE_RUN_STAMPS]
    compared += [(staged_outcome, "action_outcome")] + [(inspire_column(c, "q"), c) for c in reasoning]
    was = {c: f"_inspire_was_{i}" for i, (_value, c) in enumerate(compared)}
    was.update({c: f"_inspire_kept_{i}" for i, c in enumerate(kept)})
    prior = ", ".join(f"{inspire_column(c, 'p')} AS {name}" for c, name in was.items())
    same = " AND ".join(["q._inspire_was_current", "q._inspire_was_source = 'run'"]
                        + [inspire_compared_sql(value, f"q.{was[c]}", kinds.get(str(c).lower())) for value, c in compared]
                        + [f"q.{was[c]} IS NULL" for c in kept])
    if ai and tier:
        same += (f" AND NOT (({staged[columns.index(tier)]}) IN ({inspire_sql_list(INSPIRE_FLAGGED_TIERS)}) AND "
                 f"nullif(trim(q.{INSPIRE_AI_FACTS}), '') IS NOT NULL AND q.ai_facts_hash IS NULL)")
    rows = (f"SELECT {key} AS _inspire_record_key, {picked}, {carried}, COALESCE(p.is_current, FALSE) AS _inspire_was_current, "
            f"p.change_source AS _inspire_was_source, COALESCE(p.is_current AND {inspire_replaceable_sql('p', replace_edits)}, FALSE) AS "
            f"_inspire_was_open, {prior} FROM {source} AS s LEFT JOIN ({latest}) AS p ON p.record_key = {key}")
    judged = f"SELECT q.*, {f'COALESCE({same}, FALSE)' if stamp else 'FALSE'} AS _inspire_same FROM ({rows}) AS q"
    restamp = ", ".join([f"{inspire_column(c)} = {inspire_column(c, 'u')}" for c in stamps]
                        + [f"valid_from = {now}", "changed_by = concat('run ', u.run_id)"])

    def replaceable(alias):
        return inspire_replaceable_sql(alias, replace_edits)

    restamped = f"WHEN MATCHED AND {replaceable('t')} AND u._inspire_same THEN UPDATE SET {restamp} " if stamp else ""
    return (f"MERGE INTO {target} AS t USING (WITH z AS ({judged}) SELECT z._inspire_record_key AS _inspire_key, z.* FROM z "
            f"UNION ALL SELECT NULL AS _inspire_key, z.* FROM z WHERE z._inspire_was_open AND NOT z._inspire_same) AS u "
            f"ON t.use_case_id = u.use_case_id AND t.record_key = u._inspire_key AND t.is_current "
            f"{restamped}WHEN MATCHED AND {replaceable('t')} THEN UPDATE {close} "
            f"WHEN NOT MATCHED THEN INSERT ({', '.join(inspire_column(c) for c in columns + list(INSPIRE_ACTION_NAMES) + reasoning)}, "
            f"record_key, valid_from, valid_to, is_current, change_source, changed_by) VALUES ({', '.join(values)}, {actions}, "
            + "".join(f"{inspire_column(c, 'u')}, " for c in reasoning) +
            f"u._inspire_record_key, {now}, NULL, TRUE, 'run', concat('run ', u.run_id)) "
            f"WHEN NOT MATCHED BY SOURCE AND t.use_case_id = {inspire_sql_literal(use_case_id)} AND t.is_current AND "
            f"{replaceable('t')} THEN UPDATE {close}")

def inspire_scope_of_sql(column, keys):
    scoped = sorted((run, found) for run, found in (keys or {}).items() if found)
    if not scoped:
        return ""
    return (f"CASE CAST({column} AS STRING) " + " ".join(f"WHEN {inspire_sql_literal(r)} THEN {inspire_sql_literal(k)}" for r, k in scoped)
            + " ELSE '' END")


def inspire_run_log_backfill_sql(result_table, tier, names, keys=None):
    rows = f"SELECT {inspire_run_log_rows_sql('t', tier, names)} FROM {inspire_table_name(result_table)} AS t WHERE t.change_source = 'run'"
    scope = inspire_scope_of_sql("b.run_id", keys)
    first = "first_seen" if scope else "first_run"
    seen = f", MIN_BY(b.run_id, b.ran_at) OVER (PARTITION BY b.record_key, {scope}) AS first_seen" if scope else ""
    within = f"PARTITION BY {inspire_scope_of_sql('run_id', keys)} " if scope else ""
    moved = f"r.prev IS NOT NULL AND n.{first} <> n.run_id AND w.record_key IS NOT NULL AND NOT (n.tier <=> w.tier)"
    return (f"INSERT INTO {inspire_table_name(inspire_run_log_table(result_table))} SELECT n.*{' EXCEPT (first_seen)' if scope else ''}, CASE WHEN "
            f"r.prev IS NULL THEN NULL WHEN n.{first} = n.run_id THEN 'new' WHEN {moved} THEN 'moved' END AS change, CASE WHEN {moved} THEN "
            f"n.tier END AS moved_to FROM (SELECT b.*, MIN_BY(b.run_id, b.ran_at) OVER (PARTITION BY b.record_key) AS first_run{seen} FROM "
            f"({rows}) AS b) AS n JOIN (SELECT run_id, LAG(run_id) OVER ({within}ORDER BY MIN(ran_at)) AS prev FROM ({rows}) AS g GROUP BY "
            f"run_id) AS r ON r.run_id = n.run_id LEFT JOIN (SELECT DISTINCT record_key, run_id, tier FROM ({rows}) AS p) AS w ON "
            f"w.record_key = n.record_key AND w.run_id = r.prev")

                f"n.record_key" if same else "")
    return (f"INSERT INTO {log} SELECT n.*, COALESCE(s.first_run, n.run_id) AS first_run, CASE WHEN c.runs = 0 THEN NULL WHEN "
            f"{seen}.record_key IS NULL THEN 'new' WHEN w.record_key IS NOT NULL AND NOT (n.tier <=> w.tier) THEN 'moved' END AS change, "
            f"CASE WHEN c.runs > 0 AND {seen}.record_key IS NOT NULL AND w.record_key IS NOT NULL AND NOT (n.tier <=> w.tier) THEN n.tier END "
            f"AS moved_to FROM (SELECT {inspire_run_log_rows_sql('t', tier, names)} FROM {inspire_table_name(table)} AS t WHERE "
            f"t.use_case_id = {inspire_sql_literal(use_case_id)} AND t.is_current AND t.change_source = 'run' AND CAST(t.run_id AS STRING) = "
            f"{run} AND t.valid_from = {now}) AS n CROSS JOIN (SELECT COUNT(*) AS runs FROM ({previous}) AS v0) AS c LEFT JOIN (SELECT "
            f"record_key, MIN_BY(first_run, ran_at) AS first_run FROM {log} WHERE CAST(run_id AS STRING) <> {run} GROUP BY record_key) AS s "
            f"ON s.record_key = n.record_key LEFT JOIN (SELECT DISTINCT p.record_key, p.tier FROM {log} AS p JOIN ({previous}) AS v ON "
            f"p.run_id = v.run_id) AS w ON w.record_key = n.record_key{compared}")


def inspire_run_log_properties(spark, table):
    properties = {str(r.get("key")): str(r.get("value")) for r in inspire_rows(spark.sql(f"SHOW TBLPROPERTIES {inspire_table_name(table)}"))}
    return {k: properties[k] for k in (INSPIRE_RUN_LOG_PROPERTY, INSPIRE_RUN_LOG_TIER_PROPERTY) if k in properties}


def inspire_run_log_properties_sql(result_table, properties):
    listed = ", ".join(f"{inspire_sql_literal(k)} = {inspire_sql_literal(v)}" for k, v in properties.items())
    return f"ALTER TABLE {inspire_table_name(result_table)} SET TBLPROPERTIES ({listed})"


def inspire_run_log_make(spark, result_table, schema, tier, backfill=True):
    spark.sql(inspire_run_log_create_sql(result_table, schema))
    if backfill:
        spark.sql(inspire_run_log_backfill_sql(result_table, tier, [name for name, _kind in schema],
                                               inspire_scope_keys(spark, result_table, "")))


def inspire_run_log_switch(spark, result_table, tier, out=print):
    log = inspire_run_log_table(result_table)
    spark.sql(inspire_run_log_properties_sql(result_table, {INSPIRE_RUN_LOG_PROPERTY: log, INSPIRE_RUN_LOG_TIER_PROPERTY: tier or ""}))
    out(f"Run log: {log} now lists the records of every run of {result_table}, so a run writes a new version of a record only when its "
        "values change")


def inspire_run_log_stamps(spark, table, tier):
    properties = inspire_run_log_properties(spark, table)
    if properties.get(INSPIRE_RUN_LOG_PROPERTY) != inspire_run_log_table(table):
        return False
    if properties.get(INSPIRE_RUN_LOG_TIER_PROPERTY, "") != (tier or ""):
        spark.sql(inspire_run_log_properties_sql(table, {INSPIRE_RUN_LOG_TIER_PROPERTY: tier or ""}))
    return True


def inspire_run_order(runs):
    first = {}
    for row in runs:
        key = row.get("run_id")
        first[key] = min(first.get(key, row["ran_us"]), row["ran_us"])
    return [key for key, _ran in sorted(first.items(), key=lambda item: item[1], reverse=True)]


def inspire_run_facts(spark, table, use_case_id, run_id, tier, stamp_us, out=print, keys=None, key=""):
    log, acted, target = inspire_run_log_table(table), inspire_acted_table(table), inspire_table_name(table)
    same = inspire_same_scope_sql("run_id", None if keys is None else dict(keys, **{str(run_id): key}), key)
    try:
        version = inspire_table_version(spark, table)
        schema = inspire_table_schema(spark, table) or []
        names = {str(name) for name, _kind in schema}
        scope = f"`use_case_id` = {inspire_sql_literal(use_case_id)}" if "use_case_id" in names else ""
        switched = inspire_run_log_properties(spark, table).get(INSPIRE_RUN_LOG_PROPERTY) == log
        history = {"log": inspire_table_name(log), "columns": [str(name) for name, _kind in schema]} if switched else None
        pinned = f"{target} VERSION AS OF {int(version)}"
        runs = [{"run_id": r.get("run_id"), "tier": r.get("tier"), "as_of_date": None if r.get("as_of_date") is None else str(r["as_of_date"])[:10],
                 "ran_us": int(r.get("ran_us") or 0), "n": int(r.get("n") or 0)}
                for r in inspire_rows(spark.sql(inspire_shell_facts.runs_sql(inspire_table_name(log), bool(tier), scope)))]
        order = inspire_run_order(runs)
        if same:
            within = " AND ".join(part for part in (scope, f"(CAST(run_id AS STRING) = {inspire_sql_literal(run_id)} OR (TRUE{same}))") if part)
            order = [r.get("run_id") for r in inspire_rows(spark.sql(
                f"SELECT CAST(run_id AS STRING) AS run_id FROM {inspire_table_name(inspire_run_log_table(table))} WHERE {within} GROUP BY run_id "
                "ORDER BY MIN(ran_at) DESC LIMIT 2"))]
        previous = order[1] if len(order) > 1 else None
        numbers = inspire_shell_facts.numbers_of(inspire_rows(spark.sql(inspire_shell_facts.snapshot_numbers_sql(
            target, names, tier, inspire_table_name(log), None if previous is None else inspire_sql_literal(previous), scope, version))))
        if previous is None:
            numbers["left_n"] = 0
        floored = inspire_shell_facts.floors_sql(f"(SELECT * FROM {pinned}{' WHERE ' + scope if scope else ''})", names)
        floors = inspire_shell_facts.floors_of(inspire_rows(spark.sql(floored))) if floored else {}
        label = "entity_label" if "entity_label" in names else None
        made = inspire_rows(spark.sql(f"CREATE OR REPLACE TABLE {inspire_table_name(inspire_acted_table(table))} CLUSTER BY (record_key) AS "
                                      + inspire_shell_facts.acted_versions_sql(pinned, names, label, scope, history)))
        spark.sql(f"CREATE OR REPLACE TABLE {inspire_table_name(inspire_closed_table(table))} AS "
                  + inspire_shell_facts.closed_rows_sql(inspire_table_name(inspire_acted_table(table)), names, label))
        spark.sql(f"CREATE OR REPLACE TABLE {inspire_table_name(inspire_closes_table(table))} CLUSTER BY (`_inspire_k`) AS "
                  + inspire_shell_facts.closes_rows_sql(inspire_table_name(inspire_acted_table(table)), names, label))
        actions = inspire_shell_facts.action_numbers_of(inspire_rows(spark.sql(inspire_shell_facts.action_counts_sql(
            inspire_table_name(inspire_acted_table(table))))))
        changes = inspire_changes_table(table) if switched and order else None
        if changes:
            spark.sql(f"CREATE OR REPLACE TABLE {inspire_table_name(inspire_changes_table(table))} CLUSTER BY (record_key) AS "
                      + inspire_shell_facts.latest_changes_sql(inspire_table_name(log), inspire_sql_literal(order[0]), scope))
        people = inspire_rows(spark.sql(
            f"SELECT CAST(record_key AS STRING) AS k, unix_micros(valid_from) AS v FROM {pinned} WHERE "
            f"{scope + ' AND ' if scope else ''}valid_from > timestamp_micros({int(stamp_us)}) AND change_source <> 'run' "
            f"LIMIT {inspire_shell_facts.PEOPLE_MAX + 1}"))
        facts = {"v": inspire_shell_facts.FACTS_VERSION, "log": log, "acted": acted, "closed": inspire_closed_table(table), "changes": changes,
                 "actions": actions, "tier": tier, "label": label, "version": version,
                 "stamp_us": int(stamp_us), "people": [[str(r.get("k")), int(r.get("v") or 0)] for r in people],
                 "latest": order[0] if order else None, "previous": previous, "switched": switched, "runs": runs, "numbers": numbers,
                 "columns": inspire_shell_facts.acted_columns(names, label), "scope": str(use_case_id) if scope else None, "names": sorted(names),
                 "floors": floors, "closes": inspire_shell_facts.closes_facts(inspire_closes_table(table))}
        inspire_ensure_summary(spark, table)
        spark.sql(inspire_ai_summary_merge_sql(table, use_case_id, run_id, {
            INSPIRE_RUN_FACTS: inspire_sql_literal(_inspire_json.dumps(facts, ensure_ascii=False, default=str))}))
    except Exception as error:
        out(f"Run facts: run {run_id} wrote its rows, but Inspire could not keep the run's counts for the App, so the App counts them "
            f"itself: {inspire_redact_text(str(error), set(), '')[:300]}")
        return None
    kept = int((made[0] if made else {}).get("num_inserted_rows") or 0)
    out(f"Run facts: kept the counts of run {run_id} for the App ({inspire_plural(len(runs), 'run and tier row')}, "
        f"{inspire_plural(numbers['current_n'], 'current record')}, {inspire_number_text(numbers['left_n'], grouped=True)} left since the run before, and "
        f"{inspire_plural(kept, 'version')} of the records people acted on in {acted})")
    return facts


def inspire_restore_table(spark, table, version):
    if version is None or inspire_table_version(spark, table) == version:
        return ""
    spark.sql(f"RESTORE TABLE {inspire_table_name(table)} TO VERSION AS OF {int(version)}")
    return f"{table} is back at its version {int(version)}"


def inspire_restore_run(spark, table, before, error, out=print):
    notes = []
    for name, restore in ((table, lambda version: inspire_restore_table(spark, table, version)),
                          (inspire_run_log_table(table), lambda version: inspire_restore_table(spark, inspire_run_log_table(table), version))):
        version = before.get(name)
        try:
            notes.append(restore(version))
        except Exception as failed:
            notes.append(f"{name} could not be put back at its version {int(version)} ({inspire_redact_text(str(failed), set(), '')[:200]}); run RESTORE TABLE {name} "
                         f"TO VERSION AS OF {int(version)}")
    notes = [note for note in notes if note]
    out(f"Business output: the run failed ({inspire_redact_text(str(error), set(), '')[:300]}), so Inspire put {table} back as it was before the run"
        + (f": {'; '.join(notes)}" if notes else ": nothing had changed yet"))


def inspire_action_counts_sql(table, stage, columns, key_names):
    target, source = inspire_table_name(table), inspire_table_name(stage)
    tier = inspire_tier_column(columns)
    dismissed, done = inspire_sql_literal(inspire_shell_facts.DISMISSED), inspire_sql_literal(inspire_shell_facts.DONE)
    kept_out = f"t.action_status = {dismissed} AND {inspire_column(tier, 's')} = 'Act now'" if tier else "FALSE"
    held = (f", COUNT_IF(t.held_back AND t.{inspire_column(tier)} = {inspire_sql_literal(INSPIRE_HELD_BACK)}) AS held_back"
            if tier and "held_back" in {str(c).lower() for c in columns} else "")
    return (f"SELECT COUNT_IF(t.action_status = {dismissed}) AS dismissed, COUNT_IF({kept_out}) AS kept_out, "
            f"COUNT_IF(t.action_status = {done} AND t.action_outcome = 'Resolved') AS resolved, "
            f"COUNT_IF(t.action_status = {done} AND t.action_outcome = 'Still flagged') AS still_flagged{held} FROM {target} AS t JOIN "
            f"{source} AS s ON t.use_case_id = s.use_case_id AND t.run_id = s.run_id AND t.record_key = "
            f"{inspire_record_key_sql(key_names, 's')} WHERE t.is_current AND t.change_source = 'run'")


def inspire_holdout_share(run=None):
    run = (globals().get("INSPIRE_RUN") or {}) if run is None else run
    try:
        share = float((run or {}).get("holdout_share") or 0)
    except (TypeError, ValueError):
        return 0.0
    return share if 0 < share <= INSPIRE_HOLDOUT_MAX_SHARE else 0.0


def inspire_held_back_sql(share, key_sql, alias):
    if not share:
        return "CAST(NULL AS BOOLEAN)"
    bucket = int(round(float(share) * INSPIRE_HOLDOUT_BUCKETS))
    return (f"crc32(concat(CAST({inspire_column('use_case_id', alias)} AS STRING), '|', {key_sql})) % {INSPIRE_HOLDOUT_BUCKETS} "
            f"< {bucket}")


def inspire_holdout_derived(columns, key_names, share, existing):
    tier = inspire_tier_column(columns)
    kept = {str(name).lower() for name, _kind in existing or []}
    if not tier or not (share or "held_back" in kept):
        return {}, {}
    held, shown = inspire_held_back_sql(share, inspire_record_key_sql(key_names, "s"), "s"), inspire_column(tier, "s")
    derived = {"held_back": held, "run_tier": f"CAST({shown} AS STRING)"}
    if share:
        derived[tier] = (f"CASE WHEN {held} AND {shown} IN ({inspire_sql_list(INSPIRE_FLAGGED_TIERS)}) THEN "
                         f"{inspire_sql_literal(INSPIRE_HELD_BACK)} ELSE {shown} END")
    return derived, dict(INSPIRE_HOLDOUT_COLUMNS)

        if str(row.get("key")) == INSPIRE_REPLACE_EDITS_PROPERTY:
            return str(row.get("value")).strip().lower() == "true"
    return False


# Widgets whose default changes what a run computes. Only these log a failed read; a failed read of any other widget keeps its default.
INSPIRE_WIDGETS_WITH_MEANING = (INSPIRE_GROUP_BY,)


def inspire_widget(name, default=""):
    applied = globals().get("INSPIRE_SETTINGS_APPLIED") or {}
    if name in applied:
        return str(applied[name])
    dbu = globals().get("dbutils")
    if dbu is None:
        return default
    try:
        return str(dbu.widgets.get(name)).strip()
    except Exception as error:
        if name not in INSPIRE_WIDGETS_WITH_MEANING:
            return default
        reason = inspire_redact_text(str(error), set(), "")[:300]
        print(f"[WIDGET-READ FAILED] {name}: {reason}", flush=True)
        try:
            proof = str(dbu.widgets.get(INSPIRE_GROUP_PROOF_MARK)).strip()
        except Exception:
            proof = ""
        if proof and name == INSPIRE_GROUP_BY:
            raise inspire_group_stop(f"the Grouping of this proof run could not be read ({reason}), so it would have grouped by {default}")
        return default

def inspire_training_sample_sql(keys):
    present = " AND ".join(f"{inspire_column(key)} IS NOT NULL" for key in keys)
    hashed = f"xxhash64({', '.join(f'CAST({inspire_column(key)} AS STRING)' for key in keys)})"
    return present, hashed


def inspire_training_sample(frame, key, spark_session=None, out=print):
    keys = [key] if isinstance(key, str) else [str(k) for k in key or []]
    if not keys or not all(str(k).strip() for k in keys):
        raise ValueError("inspire_training_sample(frame, key) needs the entity key column of the training rows, or a list of them")
    cap = inspire_preview_training_cap()
    if not cap or not inspire_is_preview(inspire_widget("run_profile", "full")):
        return frame
    spark_frame = hasattr(frame, "write")
    rows = frame if spark_frame else (spark_session if spark_session is not None else globals()["spark"]).createDataFrame(frame)
    have = {str(column).lower() for column in rows.columns}
    missing = [k for k in keys if k.lower() not in have]
    if missing:
        raise ValueError(f"inspire_training_sample: the training rows have no {', '.join(missing)} column; pass the entity key column "
                         "of the rows the model trains on")
    present, hashed = inspire_training_sample_sql(keys)
    found = inspire_rows(rows.where(present).selectExpr(f"{hashed} AS _inspire_hash").distinct().orderBy("_inspire_hash")
                         .limit(cap + 1).selectExpr("_inspire_hash", "row_number() OVER (ORDER BY _inspire_hash) AS _inspire_nth")
                         .selectExpr("COUNT(*) AS entities", f"MAX(CASE WHEN _inspire_nth <= {cap} THEN _inspire_hash END) AS cutoff"))[0]
    named, entities = ", ".join(keys), int(found.get("entities") or 0)
    if entities <= cap:
        out(f"Preview sample: the training rows hold {inspire_number_text(entities, grouped=True)} entities by {named}, no more than the {inspire_number_text(cap, grouped=True)} a preview trains on, so "
            "this preview trains on all of them")
        return frame
    globals()["INSPIRE_PREVIEW_SAMPLE"] = cap
    out(f"Preview sample: this preview trains on {inspire_plural(cap, 'entity', 'entities')} of the training rows, which hold more, picked by a stable hash of "
        f"{named} so every preview of the same data trains on the same ones; a full run trains on every training row")
    sampled = rows.where(f"{present} AND {hashed} <= {int(found['cutoff'])}")
    return sampled if spark_frame else sampled.toPandas()


def inspire_ai_owned(name):

def inspire_ai_columns(ai):

def inspire_ai_category_problems(categories):
    if not isinstance(categories, dict) or not 1 <= len(categories) <= 3:
        return ["AI_CATEGORIES must be a top-level dict of 1 to 3 categories, each keyed ai_cat_<lowercase letters, digits or _> "
                "and holding a meaning and 2 to 8 distinct values, such as {'ai_cat_driver': {'meaning': 'what moved the record', "
                "'values': ['Price', 'Volume', 'Mix']}}"]
    problems = []
    for name, spec in categories.items():
        if not isinstance(name, str) or _inspire_re.match(INSPIRE_AI_CATEGORY_NAME, name) is None:
            problems.append(f"the AI category {name!r} is not named ai_cat_<lowercase letters, digits or _>")
            continue
        spec = spec if isinstance(spec, dict) else {}
        values = spec.get("values")
        texts = [v.strip() for v in values] if isinstance(values, (list, tuple)) and all(isinstance(v, str) for v in values) else []
        if not str(spec.get("meaning") or "").strip():
            problems.append(f"{name} has no meaning")
        if not 2 <= len(texts) <= 8 or not all(texts) or len({t.lower() for t in texts}) != len(texts):
            problems.append(f"{name} needs 2 to 8 distinct, non-empty text values")
        if any(t.lower() == INSPIRE_AI_UNEXPLAINED.lower() for t in texts):
            problems.append(f"{name} uses {INSPIRE_AI_UNEXPLAINED!r} as a value, which the App keeps for records without reasoning")
        if any(len(t) > 60 for t in texts):
            problems.append(f"{name} has a value longer than 60 characters")
        if texts and all(t.lower() in INSPIRE_AI_LEVEL_WORDS for t in texts):
            problems.append(f"{name} sorts records by a level ({', '.join(texts)}), which is not a reason: the App shows "
                            f"{INSPIRE_CONFIDENCE} and the priority already, so its values must name what causes the record, "
                            "such as 'Price', 'Volume' or 'Mix'")
    return problems


def inspire_ai_prompt(categories, context, asks=""):
    lines = [str(context or "").strip()] + ([str(asks).strip()] if str(asks or "").strip() else []) + ["",
             INSPIRE_AI_USER_WINS, "",
             "You explain one record of this use case's results to the person who acts on it. The facts at the end are what is "
             "known about the record. Reply with one JSON object and nothing else, with these keys:",
             '- "explanation": 2 to 4 plain sentences: what stands out about this record and why it matters to the business. '
             "State the facts as facts. Never write about yourself, about the facts or this message, or about what is not known, "
             'and never use the word "I".',
             '- "confidence_reason": one plain sentence, in the same style, naming the facts behind the record\'s confidence '
             "level, which the facts end with; never change the level or give another one.",
             '- "next_action": one plain sentence, in the same style: the one concrete action to take on this record now.',
             '- "expected_result": one plain sentence, in the same style: what should happen if that action is taken, with a '
             "number from the facts when the facts give one.",
             '- "other_option": one plain sentence, in the same style: one other action to take on this record instead.',
             f'- "headline": at most {INSPIRE_AI_HEADLINE_WORDS} words, no full stop: what this record needs, in the plain words of the facts.',
             '- "what": one plain sentence, in the same style: what stands out about this record.',
             '- "why": one plain sentence, in the same style: why it matters to the business.',
             '- "do": one plain sentence, in the same style: the one action to take now.',
             "Use no number from the facts more than once across headline, what, why and do.",
             f'- "evidence": a JSON array of {INSPIRE_AI_EVIDENCE_SENTENCES[0]} to {INSPIRE_AI_EVIDENCE_SENTENCES[1]} short plain '
             "sentences, in the same style, each stating one fact behind this record with a number written exactly as the facts "
             "write it."]
    for name, spec in categories.items():
        lines.append(f'- "{name}": exactly one of {", ".join(_inspire_json.dumps(v) for v in spec["values"])}: {spec["meaning"]}.')
    return "\n".join(lines + ["", "Facts:", ""]).lstrip()


def inspire_ai_fingerprint(endpoint, prompt):
    return _inspire_hashlib.sha256(f"{INSPIRE_AI_PROMPT_VERSION}\n{endpoint}\n{prompt}".encode("utf-8")).hexdigest()[:16]


def inspire_ai_facts_sql(alias):
    return (f"concat(coalesce({inspire_column(INSPIRE_AI_FACTS, alias)}, ''), {inspire_sql_literal(chr(10) + 'Confidence level: ')}, "
            f"coalesce({inspire_column(INSPIRE_CONFIDENCE, alias)}, ''))")


def inspire_ai_hash_sql(fingerprint, alias):
    return f"sha2(concat({inspire_sql_literal(fingerprint)}, {inspire_ai_facts_sql(alias)}), 256)"


def inspire_ai_config(categories, endpoint, context, asks=""):
    problems = inspire_ai_category_problems(categories)
    if problems:
        raise ValueError("Inspire did not write the business output: " + "; ".join(problems))
    clean = {name: {"meaning": str(spec["meaning"]).strip(), "values": [v.strip() for v in spec["values"]]}
             for name, spec in categories.items()}
    endpoint = str(endpoint or "").strip()
    prompt = inspire_ai_prompt(clean, context, asks)
    brief = "\n".join(part for part in (str(context or "").strip(), str(asks or "").strip()) if part)
    return {"endpoint": endpoint, "categories": clean, "prompt": prompt, "fingerprint": inspire_ai_fingerprint(endpoint, prompt),
            "version": INSPIRE_AI_PROMPT_VERSION, "context": brief}


def inspire_resolved_model(endpoint, spark_session=None) -> str:
    """REQ-422: the callable name of a user-chosen model (the ai_endpoint widget): databricks-<model> is called as its Unity Gateway
    model service system.ai.<model> when that answers a one-token ask, and by its own name otherwise. Defined here because a
    generated notebook runs this source without the agent."""
    name = str(endpoint or "").strip()
    session = spark_session if spark_session is not None else globals().get("spark")
    if not name.startswith("databricks-") or name == "databricks-" or not callable(getattr(session, "sql", None)):
        return name
    service = "system.ai." + name[len("databricks-"):]
    try:
        reply = session.sql(f"SELECT ai_query({inspire_sql_literal(service)}, 'Reply with ok.', failOnError => false) AS r").collect()[0]["r"]
        return service if reply is not None and not reply["errorMessage"] else name
    except Exception:
        return name


def inspire_ai_run_config(categories=None, endpoint=None, context=None, spark_session=None):
    run = globals().get("INSPIRE_RUN") or {}
    return inspire_ai_config(globals().get("AI_CATEGORIES") if categories is None else categories,
                             inspire_resolved_model(inspire_widget("ai_endpoint") if endpoint is None else endpoint, spark_session),
                             run.get("ai_context", "") if context is None else context, run.get("ai_asks", ""))


def inspire_ai_duration(rows):
    seconds = max(0, int(rows)) / INSPIRE_AI_ROWS_PER_SECOND
    if seconds < 90:
        return f"about {max(1, round(seconds))} seconds"
    if seconds < 5400:
        return f"about {round(seconds / 60)} minutes"
    return f"about {round(seconds / 3600)} hours"


def inspire_ai_flagged_sql(tier, alias):
    return f"{inspire_column(tier, alias)} IN ({inspire_sql_list(INSPIRE_FLAGGED_TIERS)})"


def inspire_ai_need_sql(table, use_case_id, ai, tier):
    return (f"SELECT t.* FROM {inspire_table_name(table)} AS t WHERE t.use_case_id = {inspire_sql_literal(use_case_id)} AND "
            f"t.is_current AND t.change_source <> 'delete' AND {inspire_ai_flagged_sql(tier, 't')} AND "
            f"nullif(trim(t.{INSPIRE_AI_FACTS}), '') IS NOT NULL AND "
            f"(t.ai_facts_hash IS NULL OR t.ai_facts_hash <> {inspire_ai_hash_sql(ai['fingerprint'], 't')})")


def inspire_ai_work_table(table, suffix, part):
    return inspire_table_name(table, f"{suffix}_{part}")


def inspire_ai_work_sql(table, suffix, need, tier, rank, rows):
    order = (f"CASE {inspire_column(tier, 'n')} " + " ".join(f"WHEN {inspire_sql_literal(t)} THEN {i}" for i, t in
                                                            enumerate(INSPIRE_FLAGGED_TIERS)) + " END, "
             + (f"{inspire_column(rank, 'n')} ASC NULLS LAST, " if rank else "") + "n.record_key")
    picked = (f"n.use_case_id, n.record_key, n.{INSPIRE_AI_FACTS}, n.{INSPIRE_CONFIDENCE}, {inspire_column(tier, 'n')}"
              + (f", {inspire_column(rank, 'n')}" if rank else ""))
    return (f"CREATE TABLE {inspire_ai_work_table(table, suffix, 'work')} AS SELECT n.use_case_id, n.record_key, "
            f"n.{INSPIRE_AI_FACTS}, n.{INSPIRE_CONFIDENCE}, CAST(floor((ROW_NUMBER() OVER (ORDER BY {order}) - 1) / "
            f"{INSPIRE_AI_CHUNK_ROWS}) AS INT) AS ai_chunk FROM (SELECT {picked} FROM ({need}) AS n ORDER BY {order} "
            f"LIMIT {int(rows)}) AS n")


def inspire_ai_reply_sql(batch, ai, retry=False):
    asked = f"concat({inspire_sql_literal(ai['prompt'])}, {inspire_ai_facts_sql('w')}"
    if retry:
        asked += (f", {inspire_sql_literal(chr(10) * 2 + 'Your last reply was rejected (')}, w.ai_problem, "
                  f"{inspire_sql_literal('). Reply again with one JSON object that follows every rule above.')}")
    return (f"SELECT w.use_case_id, w.record_key, w.{INSPIRE_AI_FACTS}, w.{INSPIRE_CONFIDENCE}, w.ai_chunk, "
            f"ai_query({inspire_sql_literal(ai['endpoint'])}, {asked}), failOnError => false) AS ai_reply FROM ({batch}) AS w")


def inspire_ai_checks(ai):
    hedge = inspire_sql_literal(INSPIRE_AI_HEDGE)
    repr_text = inspire_sql_literal(INSPIRE_PYTHON_REPR)
    checks = [("endpoint error", "c.ai_error IS NOT NULL", "c.ai_error")]
    for key, column in INSPIRE_AI_TEXTS:
        label, text = key.replace("_", " "), inspire_column(column, "c")
        checks += [("unreadable reply" if key == "explanation" else f"no {label}", f"coalesce({text}, '') = ''",
                    "left(c.ai_raw, 300)" if key == "explanation" else "NULL"),
                   (f"hedged {label}", f"{text} RLIKE {hedge}", f"regexp_extract({text}, {hedge}, 0)"),
                   (f"{label} reads as a Python object dump", f"{text} RLIKE {repr_text}", f"regexp_extract({text}, {repr_text}, 0)")]
    short = {key: f"coalesce({inspire_column('ai_' + key, 'c')}, '')" for key in INSPIRE_AI_SHORT_KEYS}
    stops = inspire_sql_literal(INSPIRE_AI_SENTENCE_END)
    numbers = f"regexp_extract_all(concat_ws(' ', {', '.join(short.values())}), {inspire_sql_literal(INSPIRE_AI_NUMBER)}, 0)"
    checks.append((f"headline over {INSPIRE_AI_HEADLINE_WORDS} words", f"size(filter(split(trim({short['headline']}), {inspire_sql_literal(INSPIRE_AI_SPACES)}), w -> w <> '')) > {INSPIRE_AI_HEADLINE_WORDS}",
                   short["headline"]))
    checks += [(f"{key} is not one sentence", f"size(regexp_extract_all({short[key]}, {stops}, 0)) > 1", short[key]) for key in ("what", "why", "do")]
    checks.append(("number repeated across headline, what, why and do", f"size({numbers}) > size(array_distinct({numbers}))",
                   f"concat_ws(' | ', {', '.join(short.values())})"))
    low, high = INSPIRE_AI_EVIDENCE_SENTENCES
    sentences = "c.ai_evidence_list"
    unbacked = (f"NOT exists(regexp_extract_all(e, {inspire_sql_literal(INSPIRE_AI_NUMBER)}, 0), "
                f"n -> instr(c.{INSPIRE_AI_FACTS}, n) > 0)")
    checks += [(f"evidence not {low} to {high} sentences", f"{sentences} IS NULL OR size({sentences}) NOT BETWEEN {low} AND {high} "
                f"OR exists({sentences}, e -> coalesce(e, '') = '')", inspire_column(INSPIRE_AI_EVIDENCE, "c")),
               ("hedged evidence", f"exists({sentences}, e -> e RLIKE {hedge})",
                f"regexp_extract(array_join({sentences}, ' '), {hedge}, 0)"),
               ("evidence without a number from the facts", f"exists({sentences}, e -> {unbacked})",
                f"element_at(filter({sentences}, e -> {unbacked}), 1)"),
                ("evidence reads as a Python object dump", f"exists({sentences}, e -> e RLIKE {repr_text})", f"regexp_extract(array_join({sentences}, ' '), {repr_text}, 0)")]
    level = f"lower(coalesce({inspire_column(INSPIRE_CONFIDENCE, 'c')}, ''))"
    for key, column in INSPIRE_AI_TEXTS:
        label, text = key.replace("_", " "), f"coalesce({inspire_column(column, 'c')}, '')"
        named = f"regexp_extract({text}, {inspire_sql_literal(INSPIRE_AI_LEVEL_NAMED)}, 0)"
        said = f"lower(regexp_extract({named}, '(?i)(high|medium|low)', 1))"
        number = inspire_sql_literal(INSPIRE_AI_LEVEL_NUMBER)
        checks += [(f"{label} names another confidence level", f"{said} NOT IN ('', {level})", named),
                   (f"{label} gives the confidence as a number", f"{text} RLIKE {number}", f"regexp_extract({text}, {number}, 0)")]
    return checks + [(f"{name} outside its values", f"{inspire_column(name, 'c')} IS NULL", inspire_column(name + "_given", "c"))
                     for name in ai["categories"]]


def inspire_ai_checked_sql(table, suffix, ai):
    keys = tuple(key for key, _column in INSPIRE_AI_TEXTS) + tuple(ai["categories"])
    shape = inspire_sql_literal(", ".join(f"{key} STRING" for key in keys) + ", evidence ARRAY<STRING>")
    parsed = (f"SELECT a.use_case_id, a.record_key, a.{INSPIRE_AI_FACTS}, a.{INSPIRE_CONFIDENCE}, a.ai_chunk, "
              f"a.ai_reply.errorMessage AS ai_error, a.ai_reply.result AS ai_raw, "
              f"from_json(regexp_extract(a.ai_reply.result, '(?s)[{{].*[}}]', 0), {shape}) AS ai_json "
              f"FROM {inspire_ai_work_table(table, suffix, 'reply')} AS a")
    categories = []
    for name, spec in ai["categories"].items():
        given = inspire_column(name, "b.ai_json")
        picks = " ".join(f"WHEN {inspire_sql_literal(v.lower())} THEN {inspire_sql_literal(v)}" for v in spec["values"])
        categories.append(f"{given} AS {inspire_column(name + '_given')}, CASE lower(trim({given})) {picks} END AS {inspire_column(name)}")
    sentences = "transform(b.ai_json.evidence, e -> trim(e))"
    texts = ", ".join(f"trim(b.ai_json.{key}) AS {inspire_column(column)}" for key, column in INSPIRE_AI_TEXTS)
    fields = (f"SELECT b.use_case_id, b.record_key, b.{INSPIRE_AI_FACTS}, b.{INSPIRE_CONFIDENCE}, b.ai_chunk, b.ai_error, b.ai_raw, "
              f"{texts}, {sentences} AS ai_evidence_list, to_json({sentences}) AS {inspire_column(INSPIRE_AI_EVIDENCE)}, "
              f"{', '.join(categories)} FROM ({parsed}) AS b")
    checks = inspire_ai_checks(ai)
    kind = " ".join(f"WHEN {test} THEN {inspire_sql_literal(label)}" for label, test, _detail in checks)
    problem = " ".join(f"WHEN {test} THEN concat({inspire_sql_literal(label)}, coalesce(concat(': ', {detail}), ''))"
                       for label, test, detail in checks)
    return f"SELECT c.*, CASE {kind} END AS ai_problem_kind, CASE {problem} END AS ai_problem FROM ({fields}) AS c"


def inspire_ai_rows_merge_sql(table, suffix, passed, sets):
    return (f"MERGE INTO {inspire_table_name(table)} AS t USING (SELECT * FROM {inspire_ai_work_table(table, suffix, 'checked')} "
            f"WHERE ai_problem_kind IS {'NULL' if passed else 'NOT NULL'}) AS c "
            f"ON t.use_case_id = c.use_case_id AND t.record_key = c.record_key AND t.is_current AND "
            f"t.{INSPIRE_AI_FACTS} <=> c.{INSPIRE_AI_FACTS} AND t.{INSPIRE_CONFIDENCE} <=> c.{INSPIRE_CONFIDENCE} "
            f"WHEN MATCHED THEN UPDATE SET {sets}")


def inspire_ai_written_names(ai):
    return [name for name, _kind in inspire_ai_columns(ai) if name not in ("ai_facts_hash", INSPIRE_AI_CHECK)]


def inspire_ai_update_sql(table, suffix, ai):
    sets = ", ".join(f"{inspire_column(n)} = {inspire_column(n, 'c')}" for n in inspire_ai_written_names(ai))
    return inspire_ai_rows_merge_sql(table, suffix, True, f"{sets}, {inspire_column(INSPIRE_AI_CHECK)} = NULL, "
                                                          f"ai_facts_hash = {inspire_ai_hash_sql(ai['fingerprint'], 'c')}")


def inspire_ai_flag_sql(table, suffix, ai):
    sets = ", ".join(f"{inspire_column(n)} = NULL" for n in inspire_ai_written_names(ai))
    return inspire_ai_rows_merge_sql(table, suffix, False, f"{sets}, {inspire_column(INSPIRE_AI_CHECK)} = c.ai_problem_kind, "
                                                           "ai_facts_hash = NULL")


def inspire_ai_one_level_text(records, level):
    head = f"all {inspire_number_text(records, grouped=True)} 'Act now' and 'Review' records have the {INSPIRE_CONFIDENCE} {level!r}"
    if level == "Low":
        return (f"{head}: either the evidence is weak for every one of them, such as a model that tests poorly or a short history, "
                "or the rule that sets it reads too little evidence to tell them apart, so read their facts to see which")
    return (f"{head}, so the rule that sets it reads too little evidence to tell them apart, such as the distance from the "
            "threshold or baseline, the history length or the data completeness")


def inspire_ai_status_sql(table, use_case_id, ai, tier):
    same = f"t.ai_facts_hash = {inspire_ai_hash_sql(ai['fingerprint'], 't')}"
    return (f"SELECT COUNT(*) AS records, COUNT_IF({same}) AS explained, COUNT_IF(t.{INSPIRE_AI_CHECK} IS NOT NULL) AS flagged, "
            f"COUNT(DISTINCT t.{INSPIRE_CONFIDENCE}) AS levels, "
            f"MIN(t.{INSPIRE_CONFIDENCE}) AS level FROM {inspire_table_name(table)} AS t WHERE t.use_case_id = "
            f"{inspire_sql_literal(use_case_id)} AND t.is_current AND t.change_source <> 'delete' AND "
            f"{inspire_ai_flagged_sql(tier, 't')} AND nullif(trim(t.{INSPIRE_AI_FACTS}), '') IS NOT NULL")


def inspire_ensure_ai(spark, table, schema, ai, out=print):
    missing = inspire_history_missing([name for name, _kind in schema or []], inspire_ai_columns(ai))
    if not missing:
        return []
    added = inspire_add_required_columns(spark, table, missing, existing={str(name).lower() for name, _kind in schema or []})
    if added:
        out(f"AI reasoning: added {', '.join(added)} to {table}, where Inspire keeps the AI reasoning of "
            "each record; the rows already there hold NULL in them until they are explained")
    return added


def inspire_ai_explain(spark, table, use_case_id, ai, cap, out=print):
    columns = [name for name, _kind in inspire_table_schema(spark, table) or []]
    tier = inspire_tier_column(columns)
    rank = next((c for c in columns if str(c).lower() == "rank"), None)
    report = {"needed": 0, "explained": 0, "without": 0, "kinds": {}, "later": 0}
    if not tier:
        out(f"AI reasoning: {table} has no priority_tier column, so no record is flagged 'Act now' or 'Review' for the AI to explain")
        return report
    need = inspire_ai_need_sql(table, use_case_id, ai, tier)
    count = report["needed"] = int(inspire_rows(spark.sql(f"SELECT COUNT(*) AS n FROM ({need}) AS q"))[0].get("n") or 0)
    if not count:
        out(f"AI reasoning: every current 'Act now' and 'Review' record with facts already holds the reasoning of {ai['endpoint']} "
            "for those facts")
        return inspire_ai_tally(spark, table, use_case_id, ai, tier, report, out)
    rows = min(count, max(1, int(cap)))
    report["later"] = count - rows
    inspire_run_stage(INSPIRE_RUN_STAGES[3], rows_done=0, rows_total=rows, out=out)
    suffix, started = "_inspire_ai_" + _inspire_uuid.uuid4().hex[:12], _inspire_time.time()
    chunks = (rows + INSPIRE_AI_CHUNK_ROWS - 1) // INSPIRE_AI_CHUNK_ROWS
    out(f"AI reasoning: explaining {inspire_plural(rows, 'record')} flagged 'Act now' or 'Review' with {ai['endpoint']}, best ranked first, in "
        f"{inspire_plural(chunks, 'batch', 'batches')}, {inspire_ai_duration(rows)}"
        + (f"; the next runs explain the other {inspire_number_text(count - rows, grouped=True)}, at most {inspire_number_text(rows, grouped=True)} per run like this one" if rows < count else ""))
    try:
        spark.sql(inspire_ai_work_sql(table, suffix, need, tier, rank, rows))
        for chunk in range(chunks):
            batch = f"SELECT * FROM {inspire_ai_work_table(table, suffix, 'work')} WHERE ai_chunk = {chunk}"
            for attempt in ("first", "retry"):
                spark.sql(f"CREATE OR REPLACE TABLE {inspire_ai_work_table(table, suffix, 'reply')} AS "
                          f"{inspire_ai_reply_sql(batch, ai, retry=attempt == 'retry')}")
                spark.sql(f"CREATE OR REPLACE TABLE {inspire_ai_work_table(table, suffix, 'checked')} AS "
                          f"{inspire_ai_checked_sql(table, suffix, ai)}")
                inspire_sql_retrying(spark, inspire_ai_update_sql(table, suffix, ai), table, "Inspire saved AI reasoning to it",
                                     "saving the reasoning", out)
                tally = {str(r.get("kind") or ""): int(r.get("n") or 0) for r in inspire_rows(spark.sql(
                    f"SELECT coalesce(ai_problem_kind, '') AS kind, COUNT(*) AS n "
                    f"FROM {inspire_ai_work_table(table, suffix, 'checked')} GROUP BY 1"))}
                report["explained"] += tally.pop("", 0)
                if not tally:
                    break
                if attempt == "retry":
                    for kind, n in tally.items():
                        report["kinds"][kind] = report["kinds"].get(kind, 0) + n
                    inspire_sql_retrying(spark, inspire_ai_flag_sql(table, suffix, ai), table,
                                         "Inspire flagged the AI reasoning that failed its checks", "flagging those records", out)
                batch = f"SELECT * FROM {inspire_ai_work_table(table, suffix, 'checked')} WHERE ai_problem_kind IS NOT NULL"
    finally:
        for part in ("work", "reply", "checked"):
            spark.sql(f"DROP TABLE IF EXISTS {inspire_ai_work_table(table, suffix, part)}")
    report["without"] = sum(report["kinds"].values())
    out(f"AI reasoning: {inspire_plural(report['explained'], 'record')} explained by {ai['endpoint']} in "
        f"{round(_inspire_time.time() - started)} seconds"
        + (f"; {inspire_number_text(report['without'], grouped=True)} flagged in {INSPIRE_AI_CHECK}, with no AI reasoning shown, after failing Inspire's checks twice ("
           + ", ".join(f"{inspire_number_text(n, grouped=True)} {kind}" for kind, n in sorted(report["kinds"].items())) + ")" if report["without"] else ""))
    return inspire_ai_tally(spark, table, use_case_id, ai, tier, report, out)


def inspire_ai_tally(spark, table, use_case_id, ai, tier, report, out=print):
    status = inspire_rows(spark.sql(inspire_ai_status_sql(table, use_case_id, ai, tier)))[0]
    records, explained, flagged = (int(status.get(name) or 0) for name in ("records", "explained", "flagged"))
    one_level = status.get("level") if records >= INSPIRE_AI_ONE_LEVEL_ROWS and int(status.get("levels") or 0) == 1 else None
    report.update(records=records, current=explained, flagged=flagged, one_level=one_level,
                  missing=max(0, records - explained - flagged - int(report.get("later") or 0)))
    out(f"AI reasoning: of the {inspire_plural(records, 'current record')} flagged 'Act now' or 'Review' with facts, "
        f"{inspire_plural(explained, 'holds', 'hold')} AI reasoning"
        + (f", {inspire_number_text(flagged, grouped=True)} flagged in {INSPIRE_AI_CHECK}" if flagged else "")
        + (f", {inspire_plural(report['later'], 'waits', 'wait')} for the next runs" if report.get("later") else "")
        + (f", {inspire_plural(report['missing'], 'holds', 'hold')} neither" if report["missing"] else ""))
    if one_level:
        out(f"AI reasoning: warning: {inspire_ai_one_level_text(records, one_level)}")
    return report

def inspire_apply_setting(row, setting, scope, out=print, into=None):
    name = str(setting["name"])
    held_in = scope if into is None else into
    first_run = into is not None and name == INSPIRE_SCOPE_SETTINGS[0]
    text = row.get("text_value") if setting.get("kind") == "choice" else row.get("value")
    if first_run:
        text = row.get("value") if row.get("text_value") is None else row.get("text_value")
    text = inspire_setting_text(text)
    if first_run and text.strip() and text.strip().lower() not in INSPIRE_SCOPES:
        if name in held_in:
            held_in[name] = text
        return None
    problem = inspire_setting_problem(setting, text)
    if problem:
        out(f"Settings: the App holds a value for {name} outside what the run accepts ({problem}), so this run keeps the notebook's value")
        return None
    if name not in held_in or isinstance(held_in.get(name), bool):
        out(f"Settings: the App holds {text} for {name}, but this notebook does not read the {name} widget into a variable of that "
            f"name, so this run keeps the widget's value; {inspire_rebuild_action()} so the App's settings apply")
        return None
    held = held_in[name]
    if setting.get("kind") == "choice":
        chosen = text
    elif setting.get("kind") in ("integer", "days") or (isinstance(held, int) and float(text).is_integer()):
        chosen = int(float(text))
    else:
        chosen = float(text)
    applied = scope.setdefault("INSPIRE_SETTINGS_APPLIED", {})
    applied[name] = text
    if chosen == held:
        return None
    held_in[name] = chosen
    out(f"Settings: {name} is {text} for this run, as {row.get('changed_by') or 'someone'} set it in the App on "
        f"{inspire_iso_time(row.get('changed_at')) or 'an earlier day'} (the notebook's value is "
        f"{inspire_number_text(held) if inspire_number_constant(held) else held})")
    return chosen


def inspire_apply_capacity_setting(row, setting, scope, out=print):
    text = inspire_setting_text(row.get("value")).strip()
    problem = inspire_setting_problem(setting, text)
    if problem:
        out(f"Settings: the App holds a team capacity outside what the run accepts ({problem}), so this run keeps the notebook's value")
        return False
    scope[INSPIRE_CAPACITY_HELD] = int(float(text)) if text else None
    who, when = row.get("changed_by") or "someone", inspire_iso_time(row.get("changed_at")) or "an earlier day"
    out(f"Settings: team capacity is {inspire_plural(scope[INSPIRE_CAPACITY_HELD], 'record') + ' each run' if text else 'no cap'} for this "
        f"run, as {who} set it in the App on {when}")
    return True


def inspire_apply_assumptions(spark, table, use_case_id, scope=None, out=print, run=None):
    scope = globals() if scope is None else scope
    settings = inspire_run_settings(scope.get("INSPIRE_RUN") if run is None else run)
    formula = scope.get(INSPIRE_VALUE_FORMULA_NAME)
    named = {m.group(1).strip(): (m.group(2) or "number").strip()
             for m in _inspire_re.finditer(INSPIRE_VALUE_FORMULA_PLACEHOLDER, formula)} if isinstance(formula, str) else {}
    if not settings and not any(inspire_number_constant(scope.get(name)) for name in named):
        return {}
    if spark is None or not str(table or "").strip() or not str(use_case_id or "").strip():
        return {}
    where, stake = inspire_assumptions_table(table), inspire_stake_label(inspire_impact(scope.get("INSPIRE_RUN")))
    try:
        inspire_ensure_assumptions(spark, table)
        rows = list(inspire_assumptions_latest(spark, table, use_case_id).values())
    except Exception as error:
        out(f"{stake}: Inspire could not read the assumptions people changed in the App from {where}, so this run uses the "
            f"notebook's own values: {inspire_redact_text(str(error), set(), '')[:300]}")
        return {}
    applied = {}
    live = bool(((scope.get("INSPIRE_RUN") if run is None else run) or {}).get("settings_read"))
    for row in rows:
        name, number = str(row.get("name") or ""), row.get("value")
        if name in INSPIRE_GROUP_SQL_SETTINGS or name in INSPIRE_SCOPE_SQL_SETTINGS or name in INSPIRE_APP_ONLY_SETTINGS:
            continue
        if name == INSPIRE_GROUP_BY and inspire_widget(INSPIRE_GROUP_PROOF_MARK):
            continue
        if name == INSPIRE_CAPACITY_SETTING and name in settings:
            if inspire_apply_capacity_setting(row, settings[name], scope, out):
                applied[name] = scope[INSPIRE_CAPACITY_HELD]
            continue
        if name in settings and not live and name not in INSPIRE_SCOPE_SETTINGS:
            out(f"Settings: the App holds a value for {name}, but this notebook was built before the App's settings, so this run "
                f"keeps the widget's value; {inspire_rebuild_action()} so the App's settings apply")
            continue
        if name in settings:
            chosen = inspire_apply_setting(row, settings[name], scope, out,
                                           into=scope.get("INSPIRE_SCOPE_VALUES") if name in INSPIRE_SCOPE_SETTINGS else None)
            if chosen is not None:
                applied[name] = chosen
            continue
        held = scope.get(name)
        if not inspire_number_constant(held):
            out(f"{stake}: the App holds a value for {name}, which this notebook does not define as a number, so this run "
                "leaves it out")
            continue
        try:
            number = float(number)
        except (TypeError, ValueError):
            number = float("nan")
        if number != number or number in (float("inf"), float("-inf")):
            out(f"{stake}: the App holds {row.get('value')!r} for {name}, which is not a number, so this run keeps {held}")
            continue
        declared = scope.get(INSPIRE_VALUE_BOUNDS_NAME)
        pair = inspire_declared_range(declared.get(name)) if isinstance(declared, dict) else None
        problem = inspire_setting_problem(inspire_assumption_setting(name, named.get(name, "number"), *(pair or (None, None))),
                                          inspire_number_text(number))
        if problem:
            out(f"{stake}: the App holds a value for {name} outside what the run accepts ({problem}), so this run keeps {held}")
            continue
        chosen = int(number) if isinstance(held, int) and number.is_integer() else number
        if chosen != held:
            scope[name] = chosen
            applied[name] = chosen
            out(f"{stake}: {name} is {inspire_number_text(chosen)} for this run, as {row.get('changed_by') or 'someone'} set "
                f"it in the App on {inspire_iso_time(row.get('changed_at')) or 'an earlier day'} (the notebook's value is "
                f"{inspire_number_text(held)})")
    return applied


def inspire_read_names(scope):
    module, todo, seen = scope.get("__name__"), [], set()
    for value in list(scope.values()):
        if isinstance(value, _inspire_types.FunctionType) and value.__module__ == module and not value.__name__.startswith("inspire_"):
            todo.append(value.__code__)
    while todo:
        code = todo.pop()
        seen.update(code.co_names)
        todo.extend(item for item in code.co_consts if hasattr(item, "co_names"))
    return seen


def inspire_unread_settings(scope, run):
    if not (run or {}).get("settings_read"):
        return []
    formula = scope.get(INSPIRE_VALUE_FORMULA_NAME)
    named = [m.group(1).strip() for m in _inspire_re.finditer(INSPIRE_VALUE_FORMULA_PLACEHOLDER, formula)] if isinstance(formula, str) else []
    declared = ([name for name in inspire_run_settings(run) if name not in INSPIRE_SCOPE_SETTINGS and name != INSPIRE_CAPACITY_SETTING]
                + [name for name in named if inspire_number_constant(scope.get(name))])
    read = inspire_read_names(scope)
    return [name for name in dict.fromkeys(declared) if name not in read]


def inspire_check_settings_read(run, trigger, scope=None, out=print):
    scope = globals() if scope is None else scope
    unread = inspire_unread_settings(scope, run)
    if not unread:
        return []
    text = (f"Settings: no step of this notebook reads {', '.join(unread)}, so a value an owner saves for "
            f"{'it' if len(unread) == 1 else 'them'} in the App would change nothing; read each one, by its own name, inside the "
            "step it controls, when that step runs")
    if trigger == INSPIRE_RUN_TRIGGERS[2]:
        raise RuntimeError(text)
    out(f"{text}. This run goes on with the notebook's values.")
    return unread


def inspire_now_micros():

def inspire_run_log_start(execution, table=None):

def inspire_run_stage_sql(table, marker):
    return (f"ALTER TABLE {inspire_table_name(inspire_ai_summary_table(table))} SET TBLPROPERTIES "
            f"({inspire_sql_literal(INSPIRE_RUN_STAGE_PROPERTY)} = {inspire_sql_literal(_inspire_json.dumps(marker, sort_keys=True))})")


def inspire_run_marker(log, name=None, rows_done=None, rows_total=None, state="running", error=None):
    if name:
        index = INSPIRE_RUN_STAGES.index(name) + 1
    else:
        index = len(INSPIRE_RUN_STAGES) if state == "done" else log["stages"][-1]["index"]
    return {"run": int(log["started_us"]), "index": index, "total": len(INSPIRE_RUN_STAGES), "stages": list(INSPIRE_RUN_STAGES),
            "rows_done": rows_done, "rows_total": rows_total, "at": inspire_now_micros(), "state": state,
            "reason": "" if error is None else inspire_run_stop_reason(error), "cancelled": isinstance(error, KeyboardInterrupt)}


def inspire_error_line(error):
    try:
        known = "\n".join(sorted(globals().get("INSPIRE_SCHEMA_NAMES") or ()))
        return (inspire_redact_text(str(error), set(), known).strip().splitlines() or [""])[0].strip()
    except Exception:
        return ""


INSPIRE_RUN_CANCELLED = "The run was cancelled."


def inspire_run_stop_reason(error):
    if isinstance(error, KeyboardInterrupt):
        return INSPIRE_RUN_CANCELLED
    if type(error).__name__ == "InspireOutcomeMissing" and error.__cause__ is not None:
        error = error.__cause__
    first = inspire_error_line(error)
    return f"The run stopped on {type(error).__name__}{': ' + first if first else ''}"[:INSPIRE_RUN_REASON_CHARS]


def inspire_run_stage_write(log, marker, session, out=print):
    try:
        if not log.get("summary_ready"):
            inspire_ensure_summary(session, log["table"])
            log["summary_ready"] = True
        session.sql(inspire_run_stage_sql(log["table"], marker))
    except Exception as error:
        if marker["state"] == "running":
            out(f"Run progress: Inspire could not record step {marker['index']} of {marker['total']} for the App, so the App shows "
                f"less of this run's progress; the run goes on: {inspire_redact_text(str(error), set(), '')[:300]}")
        else:
            out(f"Run progress: Inspire could not record that this run {'finished' if marker['state'] == 'done' else 'stopped'} for "
                f"the App, so the App says it has no news of the run once its steps take longer than they used to: "
                f"{inspire_redact_text(str(error), set(), '')[:300]}")
        return False
    return True


def inspire_run_stage(name=None, rows_done=None, rows_total=None, spark_session=None, out=print, state="running", error=None, log=None):
    log = log if log is not None else globals().get("INSPIRE_RUN_LOG")
    session = spark_session if spark_session is not None else globals().get("spark")
    if not log or not log.get("table") or session is None:
        return None
    if state == "running":
        if name not in INSPIRE_RUN_STAGES or any(s["stage"] == name for s in log["stages"]):
            return None
    elif state not in INSPIRE_RUN_ENDS or name is not None or not log.get("stages") or log.get("end_state"):
        return None
    marker = inspire_run_marker(log, name, rows_done, rows_total, state, error)
    if state == "running":
        log["stages"].append({"stage": name, "index": marker["index"], "at": marker["at"], "rows_total": rows_total})
    else:
        log["end_state"] = state
    return marker if inspire_run_stage_write(log, marker, session, out) else None


def inspire_segments(run):
    shown = []
    for item in (run or {}).get("segments") or []:
        entry = {"column": str(item["column"]), "values": [str(v) for v in item.get("values") or []]}
        op = str(item.get("op") or "=").lower()
        if op not in INSPIRE_SEGMENT_OPS:
            entry["op"] = op
        shown.append(entry)
    return shown


def inspire_segment_sql(item):
    column, op = inspire_column(item["column"]), str(item.get("op") or "=").lower()
    values = [str(v) for v in item.get("values") or []]
    if op in ("!=", "not in"):
        return f"(CAST({column} AS STRING) NOT IN ({inspire_sql_list(values)}) OR {column} IS NULL)"
    if op in ("=", "in"):
        return f"CAST({column} AS STRING) IN ({inspire_sql_list(values)})"
    typed = [v if _inspire_re.fullmatch(r"-?[0-9]+", v) else inspire_sql_literal(v) for v in values]
    return f"{column} BETWEEN {typed[0]} AND {typed[1]}" if op == "between" else f"{column} {op} {typed[0]}"


def inspire_table_rows_sql(read_table, segments, scoped=""):
    name = ".".join(inspire_table_parts(read_table)).lower()
    picked = [inspire_segment_sql(s) for s in segments or [] if ".".join(inspire_table_parts(s["table"])).lower() == name]
    picked += [scoped] if scoped else []
    where = f" WHERE {' AND '.join(picked)}" if picked else ""
    return f"SELECT COUNT(*) AS n FROM {inspire_table_name(read_table)}{where}"

INSPIRE_RUN_SCOPE_WORDS = ((r"\b(records?|rows?)\b", "records", "scope_records"), (r"\bdays?\b", "recent", "scope_days"))


def inspire_setting_text(value):

        return inspire_number_text(value)
    return "" if value is None else str(value)


def inspire_recorded_run_scope(values, declared=None):
    recorded = values if isinstance(values, dict) else {}
    was = inspire_setting_text(recorded.get("run_scope")).strip()
    if not was:
        return recorded, False
    if was.lower() in INSPIRE_SCOPES:
        return (recorded if was.lower() == recorded["run_scope"] else dict(recorded, run_scope=was.lower())), False
    if was.lower() == INSPIRE_GROUP_CUSTOM:
        return dict(recorded, run_scope=INSPIRE_GROUP_CUSTOM), False
    count = int(was) if _inspire_re.fullmatch(r"[0-9]+", was) else None
    legacy = isinstance(declared, dict) and declared.get("kind") != "choice"
    words = " ".join(inspire_setting_text(declared.get(key)).strip() for key in ("label", "help", "unit")).lower() if legacy else ""
    meant = [w for w in INSPIRE_RUN_SCOPE_WORDS if _inspire_re.search(w[0], words)] if count is not None and count >= 1 else []
    if len(meant) == 1:
        return dict(recorded, run_scope=meant[0][1], **{meant[0][2]: str(count)}), False
    return dict(recorded, run_scope=""), True

def inspire_scope_groups_of(text, finest=""):
    raw = str(text or "").strip()
    listed = None
    if raw.startswith("["):
        try:
            listed = _inspire_json.loads(raw)
        except ValueError:
            listed = None

    def parts_of(group):
        if not isinstance(group, list) or not group:
            return None
        parts = [dict({"column": str(p.get("column") or "").strip(), "value": "" if p.get("value") is None else str(p.get("value"))},
                      **({"join": dict(p["join"])} if isinstance(p.get("join"), dict) else {}))
                 for p in group if isinstance(p, dict)]
        return parts if len(parts) == len(group) and all(p["column"] for p in parts) else None

    groups = [parts_of(g) for g in listed] if isinstance(listed, list) else []
    if groups and all(groups):
        return groups
    seen = []
    for part in raw.split(","):
        value = part.strip()
        if value and value.lower() not in [s.lower() for s in seen]:
            seen.append(value)
    return [[{"column": str(finest or "").strip(), "value": v}] for v in seen if v.lower() != INSPIRE_SETTING_ALL]


def inspire_scope_group_text(group):
    return " and ".join(f"{p['column']} = {p['value']}" if p.get("column") else p["value"] for p in group or [])


def inspire_whole_groups(sizes, target):
    goal = int(target) if str(target or "").strip().isdigit() else INSPIRE_SCOPE_DEFAULTS["scope_records"]
    eligible = sorted(((str(s["value"]), int(s.get("records") or 0)) for s in sizes or []
                       if isinstance(s, dict) and s.get("value") is not None and int(s.get("records") or 0) > 0),
                      key=lambda g: (-g[1], g[0]))
    fits = [g for g in eligible if g[1] <= goal]
    order = [fits[0], fits[(len(fits) - 1) // 2], fits[-1]] + fits if fits else []
    picked, total = [], 0
    for group in order:
        if len(picked) >= INSPIRE_WHOLE_GROUPS_MAX:
            break
        if group not in picked and total + group[1] <= goal:
            picked.append(group)
            total += group[1]
    for group in order:
        if len(picked) >= min(INSPIRE_WHOLE_GROUPS_MIN, len(fits)):
            break
        if group not in picked:
            picked.append(group)
    if not picked and eligible:
        picked.append(eligible[-1])
    return [{"value": value, "records": records} for value, records in sorted(picked, key=eligible.index)]


def inspire_scope_choice(run=None, scope=None, out=print):
    scope = globals() if scope is None else scope
    run = (scope.get("INSPIRE_RUN") or {}) if run is None else run
    spec = dict((run or {}).get("scope") or {})
    table = ".".join(inspire_table_parts(spec["table"])) if spec.get("table") else ""
    values = inspire_run_scope_values(run, scope.get("INSPIRE_SCOPE_VALUES") or {}, out)
    default = str(spec.get("default") or "").strip().lower()
    default = inspire_scope_default(spec) if default in ("", INSPIRE_SCOPES[0]) else default
    mode = str(values.get("run_scope", default) or "").strip().lower() or default
    if mode == INSPIRE_GROUP_CUSTOM:
        mode = default if default in INSPIRE_SCOPES else INSPIRE_SCOPES[0]
        out(INSPIRE_CUSTOM_SQL_PLAIN.format(part="First run", plain=f"the First run choice {mode}"))
    if mode not in INSPIRE_SCOPES:
        raise ValueError(f"First run is {mode!r}, which is not one of {', '.join(INSPIRE_SCOPES)}; choose one of them on the Run page")
    chosen = {"scope": mode, "table": table}
    if mode == INSPIRE_SCOPES[0]:
        return chosen
    if not table:
        raise ValueError("First run needs the use case's main table, which this notebook does not name; choose all on the Run page")
    if mode == "records":
        technique = str(spec.get("technique") or "this technique")
        records = int(values.get("scope_records", INSPIRE_SCOPE_DEFAULTS["scope_records"]))
        first = dict(chosen, records=records, key=str(spec.get("key_column") or ""))
        if inspire_standalone(technique):
            return first
        if (inspire_grouping_applies(technique) and INSPIRE_GROUP_BY in scope
                and inspire_group_by_mode(inspire_group_by_plain(scope[INSPIRE_GROUP_BY]))[0] == INSPIRE_SETTING_ALL):
            return dict(first, one_group=True)
        try:
            column = inspire_scope_column(spec, scope)
        except ValueError:
            column = ""
        column = column or str(spec.get("group_column") or "")
        return dict(first, column=column, whole_groups=True, technique=technique) if column else dict(first, cut=technique)
    if mode == "recent":
        column = str(spec.get("time_column") or "")
        if not column:
            raise ValueError(f"First run on a recent period needs the date column of {table}, which this use case does not name; "
                             "choose groups or all on the Run page")
        period, via = str(spec.get("period") or "day").lower(), spec.get("time_via")
        if via:
            via = {part: str(via.get(part) or "").strip() if isinstance(via, dict) else "" for part in ("table", "key", "date")}
            if not all(via.values()):
                raise ValueError(f"First run on a recent period reads the dates of {table} through a date table, which this use case "
                                 "names only in part; choose groups or all on the Run page")
            via["table"] = ".".join(inspire_table_parts(via["table"]))
        return dict(chosen, column=column, days=int(values.get("scope_days", INSPIRE_SCOPE_DEFAULTS["scope_days"])),
                    period=period if period in INSPIRE_SCOPE_PERIODS else "day", time_series=bool(spec.get("time_series")),
                    **({"via": via} if via else {}))
    named = str(values.get("scope_groups") or "").strip()
    listed = inspire_scope_groups_of(named) if named.startswith("[") else []
    if listed and listed[0][0]["column"]:
        return dict(chosen, column="", groups=listed, values=[inspire_scope_group_text(g) for g in listed])
    column = inspire_scope_column(spec, scope)
    if not column:
        raise ValueError(f"First run on groups needs a column of {table} that groups its records, and this use case has none; "
                         "choose a recent period or all on the Run page, or set Data groups")
    return dict(chosen, column=column, values=[g[0]["value"] for g in inspire_scope_groups_of(named, column)])


def inspire_scoped_from(table) -> str:
    """The FROM text that reads a table through its Table Scope in this run: the scope view when it is ready, else the scope's own
    subquery of the columns and rows it keeps, else the table itself. A First run suggestion never reads rows outside the scope."""
    scope = inspire_table_scope(".".join(inspire_table_parts(table)))
    if not scope:
        return inspire_table_name(table)
    if scope.get("ready") and scope.get("view"):
        return inspire_table_name(scope["view"])
    return scope["source"]


def inspire_scope_groups_sql(source_table, column, minimum=0, limit=INSPIRE_SUGGESTED_GROUPS):
    named = inspire_column(column)
    having = f"HAVING COUNT(*) >= {int(minimum)} " if minimum else ""
    return (f"SELECT CAST({named} AS STRING) AS g, COUNT(*) AS n FROM {inspire_scoped_from(source_table)} WHERE {named} IS NOT NULL "
            f"GROUP BY CAST({named} AS STRING) {having}ORDER BY n DESC, g LIMIT {int(limit)}")


def inspire_scope_recent_sql(source_table, chosen):
    named, period, via = inspire_column(chosen["column"]), str(chosen.get("period") or "day").upper(), chosen.get("via")
    dated = (f"SELECT CAST(r.{inspire_column(via['date'])} AS DATE) AS d FROM {inspire_scoped_from(source_table)} m JOIN "
             f"{inspire_scoped_from(via['table'])} r ON m.{named} = r.{inspire_column(via['key'])} WHERE m.{named} IS NOT NULL" if via else
             f"SELECT CAST({named} AS DATE) AS d FROM {inspire_scoped_from(source_table)} WHERE {named} IS NOT NULL")
    return (f"WITH dated AS ({dated}) "
            f"SELECT DATE_SUB(MAX(d), {max(int(chosen['days']), 1) - 1}) AS by_days, MAX(d) AS newest, (SELECT MIN(p) FROM (SELECT DISTINCT "
            f"CAST(DATE_TRUNC('{period}', d) AS DATE) AS p FROM dated ORDER BY p DESC LIMIT {INSPIRE_GROUP_MIN_PERIODS}) AS q) AS by_periods "
            "FROM dated")


def inspire_scope_part_keys_sql(join, value):
    return (f"SELECT {inspire_column(INSPIRE_GROUP_JOIN_KEY)} AS k FROM ({join['sql']}) AS g WHERE "
            f"CAST({inspire_column(join['column'])} AS STRING) = {inspire_sql_literal(value)} ORDER BY k LIMIT {INSPIRE_SCOPE_VIA_KEYS_MAX + 1}")


def inspire_scope_via_keys_sql(chosen):
    via = chosen["via"]
    date = f"CAST({inspire_column(via['date'])} AS DATE)"
    return (f"SELECT DISTINCT CAST({inspire_column(via['key'])} AS STRING) AS k FROM {inspire_scoped_from(via['table'])} WHERE {date} >= DATE "
            f"{inspire_sql_literal(chosen['since'])} AND {date} <= DATE {inspire_sql_literal(chosen['newest'])} AND "
            f"{inspire_column(via['key'])} IS NOT NULL ORDER BY k LIMIT {INSPIRE_SCOPE_VIA_KEYS_MAX + 1}")


class InspireScopeEmpty(ValueError):
    pass


def inspire_scope_available(spark, chosen) -> int:
    """The records the source table holds under its Table Scope. A First run on records reads up to its goal, so the text names min(goal, this)."""
    rows = inspire_rows(spark.sql(f"SELECT COUNT(*) AS n FROM {inspire_scoped_from(chosen['table'])}"))
    return int((rows or [{}])[0].get("n") or 0)


def inspire_scope_resolve(spark, chosen, run=None):
    if chosen["scope"] == "records" and not chosen.get("whole_groups"):
        chosen = dict(chosen, available=inspire_scope_available(spark, chosen))
    if chosen["scope"] == "records" and chosen.get("whole_groups"):
        sizes = [{"value": r.get("g"), "records": int(r.get("n") or 0)}
                 for r in inspire_rows(spark.sql(inspire_scope_groups_sql(chosen["table"], chosen["column"], INSPIRE_GROUP_MIN_RECORDS, INSPIRE_GROUP_MAX_GROUPS))) if r.get("g") is not None]
        picked = inspire_whole_groups(sizes, chosen["records"])
        if not picked:
            return dict({k: v for k, v in chosen.items() if k not in ("column", "whole_groups", "technique")},
                        cut=chosen.get("technique"), cut_column=chosen["column"], available=inspire_scope_available(spark, chosen))
        return dict(chosen, values=[g["value"] for g in picked], group_records={g["value"]: g["records"] for g in picked})
    if chosen["scope"] == "groups" and chosen.get("groups"):
        resolved = []
        for group in chosen["groups"]:
            parts = []
            for part in group:
                if part.get("join") or "." in part["column"]:
                    join = inspire_group_join(part["column"], part.get("join"))
                    if join is None and not part.get("join"):
                        raise ValueError(f"First run group {inspire_scope_group_text(group)} names {part['column']}, a column of another "
                                         "table, but carries no join to that table, so the records have no "
                                         f"{part['column']} to choose; choose the groups again on the Run page")
                    if join is None:
                        raise ValueError(f"First run group {inspire_scope_group_text(group)} names {part['column']} through a join that "
                                         "is named only in part; choose the groups again on the Run page")
                    keys = [str(r.get("k")) for r in inspire_rows(spark.sql(inspire_scope_part_keys_sql(join, part["value"])))
                            if r.get("k") is not None]
                    if not keys or len(keys) > INSPIRE_SCOPE_VIA_KEYS_MAX:
                        raise ValueError(f"First run group {inspire_scope_group_text(group)} reads {part['column']} through {join['table']}, "
                                         "and " + (f"no {join['ref_column']} there has {part['column']} {part['value']}" if not keys else
                                                   f"more than {INSPIRE_SCOPE_VIA_KEYS_MAX:,} values of {join['ref_column']} have it")
                                         + "; choose other groups on the Run page")
                    part = dict(part, keys=keys)
                parts.append(part)
            resolved.append(parts)
        return dict(chosen, groups=resolved)
    if chosen["scope"] == "groups" and not chosen.get("values"):
        found = [str(r.get("g")) for r in inspire_rows(spark.sql(inspire_scope_groups_sql(chosen["table"], chosen["column"])))
                 if r.get("g") is not None]
        if not found:
            raise InspireScopeEmpty(f"First run on groups found no value of {chosen['column']} in {chosen['table']}; choose all on the Run page")
        chosen = dict(chosen, values=found, suggested=True)
    if chosen["scope"] == "recent":
        row = (inspire_rows(spark.sql(inspire_scope_recent_sql(chosen["table"], chosen))) or [{}])[0]
        by_days, by_periods = row.get("by_days"), row.get("by_periods")
        if by_days is None:
            raise InspireScopeEmpty(f"First run on a recent period found no date in {chosen['column']} of {chosen['table']}; choose all on the "
                             "Run page")
        since = min(str(by_days)[:10], str(by_periods)[:10]) if chosen.get("time_series") and by_periods is not None else str(by_days)[:10]
        chosen = dict(chosen, since=since, widened=since < str(by_days)[:10])
        if chosen.get("via"):
            via = chosen["via"]
            chosen = dict(chosen, newest=str(row.get("newest") or by_days)[:10])
            keys = [str(r.get("k")) for r in inspire_rows(spark.sql(inspire_scope_via_keys_sql(chosen))) if r.get("k") is not None]
            if not keys or len(keys) > INSPIRE_SCOPE_VIA_KEYS_MAX:
                raise ValueError(f"First run on a recent period reads {chosen['column']} through {via['table']}.{via['date']}, and "
                                 + (f"no {via['key']} there falls since {since}" if not keys else
                                    f"more than {INSPIRE_SCOPE_VIA_KEYS_MAX:,} values of {via['key']} fall since {since}")
                                 + "; choose fewer days, groups or all on the Run page")
            chosen = dict(chosen, keys=keys)
    return chosen


def inspire_scope_group_where(groups, names):
    terms = []
    for group in groups:
        parts = []
        for part in group:
            column = str(part["join"]["key"] if part.get("join") else part["column"]).lower()
            if column not in names:
                return ""
            named = inspire_column(names[column])
            parts.append(f"CAST({named} AS STRING) IN ({inspire_sql_list(part['keys'])})" if part.get("join") else
                         f"CAST({named} AS STRING) = {inspire_sql_literal(part['value'])}")
        terms.append("(" + " AND ".join(parts) + ")")
    return " OR ".join(terms) if len(terms) == 1 else "(" + " OR ".join(terms) + ")"


def inspire_scope_where(chosen, columns):
    names = {str(c).lower(): str(c) for c in columns or []}
    column = str((chosen or {}).get("column") or "").lower()
    if (chosen or {}).get("scope") == "groups" and chosen.get("groups"):
        return inspire_scope_group_where(chosen["groups"], names)
    whole_groups = (chosen or {}).get("scope") == "records" and chosen.get("whole_groups")
    if ((chosen or {}).get("scope") not in ("groups", "recent") and not whole_groups) or column not in names:
        return ""
    named = inspire_column(names[column])
    if chosen["scope"] in ("groups", "records"):
        return f"CAST({named} AS STRING) IN ({inspire_sql_list(chosen['values'])})"
    if chosen.get("via"):
        return f"CAST({named} AS STRING) IN ({inspire_sql_list(chosen['keys'])})"
    return f"CAST({named} AS DATE) >= DATE {inspire_sql_literal(chosen['since'])}"


def inspire_scope_text(chosen):
    def plural(n, one):
        return inspire_plural(n, one, grouped=True)

    mode = (chosen or {}).get("scope") or INSPIRE_SCOPES[0]
    shown = min(chosen["records"], chosen["available"]) if (chosen or {}).get("available") is not None else (chosen or {}).get("records")
    if mode == "groups" and chosen.get("groups"):
        return f"the {plural(len(chosen['groups']), 'group')} {'; '.join(inspire_scope_group_text(g) for g in chosen['groups'])}"
    if mode == "groups":
        return f"the {plural(len(chosen['values']), 'group')} {', '.join(chosen['values'])} of {chosen['column']}"
    if mode == "records" and chosen.get("whole_groups"):
        sizes = chosen.get("group_records") or {}
        named = "; ".join(f"{chosen['column']} = {v}" + (f", {plural(sizes[v], 'record')}" if v in sizes else "")
                          for v in chosen.get("values") or [])
        total = sum(sizes.get(v, 0) for v in chosen.get("values") or [])
        if total > chosen["records"] and len(chosen.get("values") or []) > 1:
            return (f"{plural(total, 'record')} of {chosen['table']}, in {plural(len(chosen['values']), 'whole group')}, so the "
                    f"First run sees more than one group and no group is cut: {named}")
        if total > chosen["records"]:
            return (f"{plural(total, 'record')} of {chosen['table']}, all of {chosen['column']} = {chosen['values'][0]}, the smallest "
                    f"group, since every group holds more than {plural(chosen['records'], 'record')}")
        if sizes and chosen.get("values") and all(v in sizes for v in chosen["values"]):
            return (f"{plural(total, 'record')} of {chosen['table']}, in "
                    f"{plural(len(chosen['values']), 'whole group')} so no group is cut: {named}")
        return (f"about {plural(chosen['records'], 'record')} of {chosen['table']}, in "
                f"{plural(len(chosen.get('values') or []), 'whole group')} so no group is cut: {named}")
    if mode == "records" and chosen.get("one_group"):
        return f"{plural(shown, 'record')} of {chosen['table']} {INSPIRE_SPREAD_WORDS}, the whole data being one group"
    if mode == "records" and chosen.get("cut"):
        held = (f"{chosen['table']} has no group of {chosen['cut_column']} with {plural(INSPIRE_GROUP_MIN_RECORDS, 'record')} or more"
                if chosen.get("cut_column") else f"{chosen['table']} has no column to take whole groups of")
        return (f"{plural(shown, 'record')} of {chosen['table']} {INSPIRE_SPREAD_WORDS}; {chosen['cut']} reads records "
                f"together and {held}, so a group may be cut in this First run")
    if mode == "recent" and chosen.get("via"):
        via = chosen["via"]
        return f"the records of {chosen['column']} whose {via['date']} in {via['table']} is since {chosen['since']}"
    if mode == "recent":
        return f"the records of {chosen['column']} since {chosen['since']}"
    if mode == "records":
        return f"{plural(shown, 'record')} of {chosen['table']} {INSPIRE_SPREAD_WORDS}"
    return "all the records"


def inspire_scope_key(chosen, grouping=""):
    mode = (chosen or {}).get("scope") or INSPIRE_SCOPES[0]
    defined = {"scope": mode, "table": chosen.get("table"), "column": chosen.get("column"), "days": chosen.get("days"),
               "values": list(chosen.get("values") or []) if mode == "groups" or chosen.get("whole_groups") else None,
               "records": chosen.get("records")}
    if chosen.get("via"):
        defined["via"] = dict(chosen["via"])
    if chosen.get("groups"):
        defined["groups"] = [[{k: p[k] for k in ("column", "value", "join") if k in p} for p in g] for g in chosen["groups"]]
    key = mode if mode == INSPIRE_SCOPES[0] else (
        f"{mode}:" + _inspire_hashlib.sha256(_inspire_json.dumps(defined, sort_keys=True).encode("utf-8")).hexdigest()[:16])
    return f"{key}|grouping:{str(grouping)[:16]}" if grouping else key


def inspire_scope_record(chosen, grouping=""):
    mode = (chosen or {}).get("scope") or INSPIRE_SCOPES[0]
    entry = {"scope": mode, "key": inspire_scope_key(chosen, grouping), "text": inspire_scope_text(chosen)}
    if mode == "groups":
        entry.update(table=chosen["table"], column=chosen["column"], values=list(chosen["values"]), suggested=bool(chosen.get("suggested")))
        if chosen.get("groups"):
            entry["groups"] = [[{k: p[k] for k in ("column", "value", "join") if k in p} for p in g] for g in chosen["groups"]]
    elif mode == "recent":
        entry.update(table=chosen["table"], column=chosen["column"], op=">=", values=[chosen["since"]], days=chosen["days"],
                     widened=bool(chosen.get("widened")), **({"via": dict(chosen["via"])} if chosen.get("via") else {}))
    elif mode == "records":
        entry.update(table=chosen["table"], records=chosen["records"])
        if chosen.get("whole_groups"):
            entry.update(column=chosen["column"], values=list(chosen.get("values") or []), whole_groups=True)
        elif chosen.get("cut"):
            entry["cut"] = True
    return entry


INSPIRE_SCOPE_KEY_PATTERN = '"key": "([^"]*)"'


def inspire_run_scope_key(scope=None):
    scope = globals() if scope is None else scope
    log = scope.get("INSPIRE_RUN_LOG") or {}
    if not (scope.get("INSPIRE_RUN") or {}).get("scope") or not log.get("scope"):
        return ""
    return inspire_scope_key(log["scope"], (scope.get("INSPIRE_RUN_GROUPING") or {}).get("hash"))


def inspire_scope_keys_sql(result_table):
    return (f"SELECT CAST(run_id AS STRING) AS run_id, MAX(regexp_extract(run_segments, {inspire_sql_literal(INSPIRE_SCOPE_KEY_PATTERN)}, "
            f"1)) AS scope_key FROM {inspire_table_name(inspire_ai_summary_table(result_table))} GROUP BY CAST(run_id AS STRING)")


def inspire_scope_keys(spark, result_table, key, out=print):
    try:
        return {str(r.get("run_id")): str(r.get("scope_key") or "") for r in inspire_rows(spark.sql(inspire_scope_keys_sql(result_table)))}
    except Exception as error:
        if key:
            out(f"Run log: Inspire could not read which earlier runs of {result_table} covered the same records as this one, so this run marks "
                f"no record as new or moved and the App compares it with no earlier run: {inspire_redact_text(str(error), set(), '')[:300]}")
        return None


def inspire_same_scope_sql(column, keys, key):
    if keys is None:
        return " AND FALSE" if key else ""
    if key:
        same = sorted(run for run, found in keys.items() if found == key)
        return f" AND CAST({column} AS STRING) IN ({', '.join(inspire_sql_literal(r) for r in same)})" if same else " AND FALSE"
    other = sorted(run for run, found in keys.items() if found)
    return f" AND CAST({column} AS STRING) NOT IN ({', '.join(inspire_sql_literal(r) for r in other)})" if other else ""


INSPIRE_SCOPE_DONE = ("First run: an earlier run already covered the First run, so this run covers all the records in the Data "
                      "groups you picked. Choose a First run on the Run page to run one again")


def inspire_scope_asked_sql(result_table, use_case_id):
    return (f"SELECT unix_micros(MAX(changed_at)) AS us FROM {inspire_table_name(inspire_assumptions_table(result_table))} "
            f"WHERE use_case_id = {inspire_sql_literal(use_case_id)} AND name = {inspire_sql_literal(INSPIRE_SCOPE_SETTINGS[0])}")


def inspire_scope_runs_sql(result_table, since_us):
    return (f"SELECT COUNT(*) AS n, COALESCE(SUM(CASE WHEN run_trigger = {inspire_sql_literal(INSPIRE_RUN_TRIGGERS[2])} THEN 0 ELSE 1 END), 0) "
            f"AS later FROM (SELECT run_id, MAX(run_trigger) AS run_trigger FROM {inspire_table_name(inspire_ai_summary_table(result_table))} "
            f"WHERE run_started_at IS NOT NULL AND unix_micros(run_started_at) > {int(since_us)} GROUP BY run_id)")


def inspire_scope_asked_us(spark, result_table, use_case_id):
    if not str(use_case_id or "").strip():
        return 0
    try:
        rows = inspire_rows(spark.sql(inspire_scope_asked_sql(result_table, use_case_id)))
    except Exception:
        return 0
    return int((rows[0] if rows else {}).get("us") or 0)


def inspire_scope_done(spark, run, trigger, result_table, out=print):
    if spark is None or not str(result_table or "").strip():
        return False
    table = ".".join(inspire_table_parts(result_table))
    since = max(int(((run or {}).get("scope") or {}).get("asked_us") or 0),
                inspire_scope_asked_us(spark, table, (run or {}).get("use_case_id")))
    try:
        rows = inspire_rows(spark.sql(inspire_scope_runs_sql(table, since)))
    except Exception as error:
        if "TABLE_OR_VIEW_NOT_FOUND" not in str(error) and "cannot be found" not in str(error):
            out(f"First run: Inspire could not read the earlier runs of {table}, so this run keeps the First run: "
                f"{inspire_redact_text(str(error), set(), '')[:300]}")
        return False
    held = rows[0] if rows else {}
    return int(held.get("later" if trigger == INSPIRE_RUN_TRIGGERS[2] else "n") or 0) > 0


def inspire_scope_start(spark, run=None, scope=None, out=print):
    scope = globals() if scope is None else scope
    run = (scope.get("INSPIRE_RUN") or {}) if run is None else run
    if not (run or {}).get("scope"):
        return None
    held = scope.get("INSPIRE_RUN_LOG") or {}
    done = inspire_scope_done(spark, run, held.get("trigger"), held.get("table") or run.get("result_table"), out)
    try:
        if done:
            out(INSPIRE_SCOPE_DONE)
        chosen = inspire_scope_choice(run, dict(scope, INSPIRE_SCOPE_VALUES=dict(
            scope.get("INSPIRE_SCOPE_VALUES") or {}, run_scope=INSPIRE_SCOPES[0])) if done else scope, out)
        chosen = inspire_scope_resolve(spark, chosen, run) if chosen["scope"] != INSPIRE_SCOPES[0] else chosen
    except InspireScopeEmpty as error:
        reason = _inspire_re.sub(r"[;,]\s*choose\b.*$", "", str(error)).strip().rstrip(".")
        spec = dict((run or {}).get("scope") or {})
        chosen = {"scope": INSPIRE_SCOPES[0], "table": ".".join(inspire_table_parts(spec["table"])) if spec.get("table") else "",
                  "fallback": reason}
        out(f"{reason}, so this run covers all records instead. Choose the First run again on the Run page to narrow it")
    scope["INSPIRE_SCOPE"] = chosen
    log = scope.get("INSPIRE_RUN_LOG")
    if log is not None:
        log["scope"] = chosen
    widened = (f", widened to keep the {INSPIRE_GROUP_MIN_PERIODS} most recent {chosen.get('period')} periods"
               if chosen.get("widened") else "")
    out(f"First run: this run covers {inspire_scope_text(chosen)}{widened}")
    return chosen


def inspire_spread_hash(frame) -> str:
    kinds = dict(frame.dtypes)
    usable = [c for c in frame.columns if not str(kinds.get(c, "")).lower().startswith("map")]
    return f"xxhash64({', '.join(f'CAST({inspire_column(c)} AS STRING)' for c in usable)})" if usable else ""


def _inspire_spread_expr(spread):
    from pyspark.sql import functions as _inspire_f
    return _inspire_f.expr(spread)


def inspire_sample_covers(frame, limit) -> bool:
    return frame.count() <= int(limit)


def inspire_read_source(table, spark=None, purpose=INSPIRE_READ_PURPOSES[0]):
    if purpose not in INSPIRE_READ_PURPOSES:
        raise ValueError(f"inspire_read_source takes purpose={INSPIRE_READ_PURPOSES[0]!r} for the records the run scores or "
                         f"purpose={INSPIRE_READ_PURPOSES[1]!r} for the rows a model trains on, not {purpose!r}")
    session = spark if spark is not None else globals().get("spark")
    name = ".".join(inspire_table_parts(table))
    scope = inspire_table_scope(name)
    if scope and scope.get("ready") and scope.get("view"):
        frame = session.table(inspire_table_name(scope["view"]))
    else:
        frame = session.table(inspire_table_name(name))
        if scope and scope["where"]:
            frame = frame.where(scope["where"])
    inspire_register_schema(frame)
    log = globals().get("INSPIRE_RUN_LOG")
    if purpose == INSPIRE_READ_PURPOSES[1]:
        if log is not None:
            log.setdefault("training_reads", {})[name.lower()] = _scoped_read_entry(scope, {"where": "", "limit": None})
        return inspire_scope_projection(frame, scope)
    chosen = globals().get("INSPIRE_SCOPE") or {"scope": INSPIRE_SCOPES[0]}
    records = bool(chosen.get("table")) and name.lower() == ".".join(inspire_table_parts(chosen["table"])).lower()
    where = inspire_scope_where(chosen, frame.columns) if records else ""
    if where:
        frame = frame.where(where)
    limit = (chosen.get("records") if records and chosen["scope"] == "records" and not chosen.get("whole_groups") else None)
    if limit and inspire_sample_covers(frame, limit):
        limit = None
    if limit:
        spread = inspire_spread_hash(frame)
        frame = (frame.orderBy(_inspire_spread_expr(spread)) if spread else frame).limit(int(limit))
    if log is not None:
        log.setdefault("reads", {})[name.lower()] = _scoped_read_entry(scope, {"where": where, "limit": int(limit) if limit else None})
    return inspire_scope_projection(frame, scope)


def inspire_iso_time(stamp):
    if stamp is None:
        return None
    if hasattr(stamp, "isoformat"):
        return stamp.isoformat(timespec="seconds") + ("Z" if getattr(stamp, "tzinfo", None) is None else "")
    return str(stamp)


def inspire_table_updated_at(spark, read_table):
    try:
        rows = inspire_rows(spark.sql(f"DESCRIBE HISTORY {inspire_table_name(read_table)} LIMIT 1"))
    except Exception:
        catalog, schema, name = inspire_table_parts(read_table)
        rows = inspire_rows(spark.sql(f"SELECT last_altered AS timestamp FROM {inspire_column(catalog)}.information_schema.tables "
                                      f"WHERE lower(table_schema) = lower({inspire_sql_literal(schema)}) AND lower(table_name) = "
                                      f"lower({inspire_sql_literal(name)})"))
    return inspire_iso_time(rows[0].get("timestamp")) if rows else None


def inspire_run_tables(spark, run, out=print):
    log = globals().get("INSPIRE_RUN_LOG") or {}
    found, reads, trained = [], log.get("reads") or {}, log.get("training_reads") or {}
    for table in (run or {}).get("source_tables") or []:
        name = ".".join(inspire_table_parts(table))
        read = reads.get(name.lower()) or {}
        entry = {"table": name, "rows": None, "rows_total": None, "scoped": bool(read.get("where") or read.get("limit") or read.get("columns")),
                 "read_by_helper": name.lower() in reads or name.lower() in trained, "updated_at": None}
        if name.lower() in trained:
            entry.update(training_read=True, training_scoped=False, training_rows=None)
        try:
            entry["rows_total"] = int(inspire_rows(spark.sql(inspire_table_rows_sql(name, run.get("segments"))))[0].get("n") or 0)
            if name.lower() in trained:
                entry["training_rows"] = entry["rows_total"]
            counted = (int(inspire_rows(spark.sql(inspire_table_rows_sql(name, run.get("segments"), read.get("where"))))[0].get("n") or 0)
                       if read.get("where") else entry["rows_total"])
            entry["rows"] = min(counted, int(read["limit"])) if read.get("limit") else counted
        except Exception as error:
            out(f"Run log: Inspire could not count the rows of {name} this run covered: {inspire_redact_text(str(error), set(), '')[:300]}")
        try:
            entry["updated_at"] = inspire_table_updated_at(spark, name)
        except Exception as error:
            out(f"Run log: Inspire could not read when {name} last changed: {inspire_redact_text(str(error), set(), '')[:300]}")
        found.append(entry)
    return found


def inspire_save_run_log(spark, table, use_case_id, run_id, out=print):
    run = globals().get("INSPIRE_RUN") or {}
    log = globals().get("INSPIRE_RUN_LOG") or {"started_us": None, "trigger": inspire_run_trigger("here"), "written": []}
    try:
        tables = inspire_run_tables(spark, run, out)
        values = {"run_started_at": f"timestamp_micros({int(log['started_us'])})" if log.get("started_us") else "NULL",
                  "run_trigger": inspire_sql_literal(log["trigger"]),
                  "run_segments": inspire_sql_literal(_inspire_json.dumps(inspire_segments(run) + (
                      [inspire_scope_record(log.get("scope"), (globals().get("INSPIRE_RUN_GROUPING") or {}).get("hash"))]
                      if (run or {}).get("scope") else []), ensure_ascii=False)),
                  "run_tables": inspire_sql_literal(_inspire_json.dumps(tables, ensure_ascii=False)),
                  "run_grouping": (inspire_sql_literal(_inspire_json.dumps(globals()["INSPIRE_RUN_GROUPING"], ensure_ascii=False))
                                   if globals().get("INSPIRE_RUN_GROUPING") is not None else "NULL")}
        inspire_ensure_summary(spark, table)
        spark.sql(inspire_ai_summary_merge_sql(table, use_case_id, run_id, values))
    except Exception as error:
        out(f"Run log: run {run_id} wrote its rows, but Inspire could not record when it ran and what it read in "
            f"{inspire_ai_summary_table(table)}: {inspire_redact_text(str(error), set(), '')[:300]}")
        return None
    log["written"].append((table, use_case_id, run_id))
    slices = inspire_segments(run)
    started = {"schedule": "the schedule", "refresh": "a refresh", "build": "the build"}[log["trigger"]]
    out(f"Run log: run {run_id} was started by {started} and read {inspire_plural(len(tables), 'table')} "
        + (f"for {'; '.join(s['column'] + ' ' + s.get('op', 'in') + ' ' + ', '.join(s['values']) for s in slices)}" if slices
           else "with all their data") + (f", its First run covering {inspire_scope_text(log['scope'])}" if log.get("scope") else "")
        + (f"; {inspire_training_text(log)}" if log.get("training_reads") else ""))
    return values


def inspire_run_log_end(log, spark_session=None, out=print, error=None):
    if globals().get("INSPIRE_RUN_LOG") is log:
        globals().pop("INSPIRE_RUN_LOG")
    session = spark_session if spark_session is not None else globals().get("spark")
    if log:
        inspire_run_stage(state="done" if error is None else "stopped", error=error, spark_session=session, out=out, log=log)
    if not log or not log.get("written"):
        return None
    ended_us = inspire_now_micros()
    ended = f"timestamp_micros({ended_us})"
    fields = {"run_ended_at": ended}
    if error is None and log.get("stages"):
        fields["run_stages"] = inspire_sql_literal(_inspire_json.dumps({"stages": log["stages"], "ended": ended_us}, sort_keys=True))
    for table, use_case_id, run_id in log["written"]:
        try:
            session.sql(inspire_ai_summary_merge_sql(table, use_case_id, run_id, fields))
        except Exception as error:
            out(f"Run log: Inspire could not record when run {run_id} ended in {inspire_ai_summary_table(table)}: {inspire_redact_text(str(error), set(), '')[:300]}")
    return ended

def inspire_outcome_sql(table, use_case_id, columns, value):
    named = {str(c).lower(): inspire_column(c, "t") for c in columns}
    tier = named["priority_tier"]
    picks = [f"COUNT_IF({tier} = 'Act now') AS act_now_records", f"COUNT_IF({tier} = 'Review') AS review_records",
             "COUNT(*) AS records"]
    if "data_quality_flag" in named:
        flag = named["data_quality_flag"]
        picks += [f"COUNT_IF(coalesce({flag}, '') <> 'ok') AS flagged_records", f"mode(nullif({flag}, 'ok')) AS top_flag"]
    if "held_back" in named:
        picks.append(f"COUNT_IF({tier} = {inspire_sql_literal(INSPIRE_HELD_BACK)}) AS held_records")
    if value and INSPIRE_VALUE_AT_STAKE in named:
        picks.append(f"MAX({named[INSPIRE_VALUE_AT_STAKE]}) AS top_stake")
    read = [named[n] for n in ("priority_tier", "as_of_date", "data_quality_flag", "held_back") if n in named]
    if value and INSPIRE_VALUE_AT_STAKE in named:
        read.append(named[INSPIRE_VALUE_AT_STAKE])
    rows = (f"SELECT {', '.join(read)} FROM {inspire_table_name(table)} AS t WHERE t.use_case_id = "
            f"{inspire_sql_literal(use_case_id)} AND t.is_current AND t.change_source <> 'delete'")
    if "as_of_date" in named:
        rows = inspire_shell_facts.run_rows_source(rows, "as_of_date")
    return f"SELECT {', '.join(picks)} FROM ({rows}) AS t"


def inspire_run_total_sql(table, use_case_id, columns, value):
    """The run's value at stake through the App's money rule (shell_facts.run_total_query): the flagged current records' amounts in
    closed months, so undated and future rows are left out. A run with no value contract, no stake or no as_of_date has no total: the
    query returns NULL, which is what the run records."""
    named = {str(c).lower(): inspire_column(c, "t") for c in columns}
    if not value or INSPIRE_VALUE_AT_STAKE not in named or "as_of_date" not in named or "priority_tier" not in named:
        return "SELECT CAST(NULL AS DOUBLE) AS value"
    source = (f"SELECT {named[INSPIRE_VALUE_AT_STAKE]} AS value_at_stake, {named['as_of_date']} AS as_of_date "
              f"FROM {inspire_table_name(table)} AS t WHERE t.use_case_id = {inspire_sql_literal(use_case_id)} "
              f"AND t.is_current AND t.change_source <> 'delete' AND {named['priority_tier']} IN ({inspire_sql_list(INSPIRE_FLAGGED_TIERS)})")
    return inspire_shell_facts.run_total_query(source)


INSPIRE_NOTHING_TO_ACT_FLAGGED = "Every record has a data quality issue"
INSPIRE_NOTHING_TO_ACT_EMPTY = "The last run wrote no current records."
INSPIRE_NOTHING_TO_ACT_UNTESTED = "No record passed this use case's decision test in the last run."
INSPIRE_NOTHING_TO_ACT_UNACCEPTED = "No model passed this use case's acceptance check in the last run."
INSPIRE_THRESHOLD_PREFIXES = ("min_", "max_")
INSPIRE_THRESHOLD_LABELS = {}


def inspire_missed_text(missed, accepted):
    if missed is None:
        return ""
    if accepted is not False:
        raise ValueError("inspire_write_result takes missed= only with accepted=False: the thresholds no model passed")
    if not isinstance(missed, dict) or not missed:
        raise ValueError("inspire_write_result takes missed={<threshold widget>: <measured value>}, such as missed={'min_auc': 0.58}, "
                         f"not {missed!r}")
    parts = []
    for name, measured in missed.items():
        label = INSPIRE_THRESHOLD_LABELS.get(str(name))
        if not label:
            raise ValueError(f"inspire_write_result takes missed= keyed by a threshold widget, such as "
                             f"{', '.join(sorted(INSPIRE_THRESHOLD_LABELS)[:4])}, not {name!r}")
        try:
            number = None if isinstance(measured, (str, bytes, bool)) or measured is None else float(measured)
        except (TypeError, ValueError):
            number = None
        if number is None or not _inspire_math.isfinite(number):
            guide = "; a criterion that does not apply goes in not_applicable=[...], never in missed=" if measured is None else ""
            raise ValueError(f"inspire_write_result takes missed={{{str(name)!r}: <measured value>}} with a number, not {measured!r}{guide}")
        try:
            needs = float(inspire_widget(str(name), ""))
        except ValueError:
            raise ValueError(f"missed= names {name}, but this notebook has no {name} widget holding a number, so Inspire cannot "
                             "show the bar the model missed") from None
        if (number >= needs) if str(name).startswith("min_") else (number <= needs):
            raise ValueError(f"missed= says the model missed {name}, but it measured {inspire_number_text(number)} and the bar is "
                             f"{inspire_number_text(needs)}, which it meets")
        parts.append(f"{label} is {inspire_number_text(needs)}, and this run measured {inspire_number_text(number)}")
    return f"{INSPIRE_NOTHING_TO_ACT_UNACCEPTED.rstrip('.')}: {'; '.join(parts)}."


def inspire_nothing_to_act(found, value, given):
    if int(found.get("act_now_records") or 0) + int(found.get("review_records") or 0):
        return None
    records, flagged = int(found.get("records") or 0), int(found.get("flagged_records") or 0)
    if records and flagged >= records:
        top = found.get("top_flag")
        return INSPIRE_NOTHING_TO_ACT_FLAGGED + (f", most often: {top}." if top else ".")
    if records and int(found.get("held_records") or 0):
        return (f"Every record the last run flagged is held back to measure the impact of this use case "
                f"({inspire_number_text(int(found['held_records']), grouped=True)} of them), so none is on the to fix list.")
    if str(given or "").strip():
        return str(given).strip()
    if not records:
        return INSPIRE_NOTHING_TO_ACT_EMPTY
    if value and value["basis"] == "predicted" and value.get("accepted") is False:
        return INSPIRE_NOTHING_TO_ACT_UNACCEPTED
    review = inspire_value_amounts(value)[1]
    if value and value["basis"] == "measured" and review is not None:
        top = found.get("top_stake")
        return (f"No record has {inspire_money_text(review, inspire_value_unit(value))} or more at stake, the amount this use case "
                "reviews from" + (f"; the most at stake on one record is {inspire_money_text(top, inspire_value_unit(value))}."
                                  if top is not None else "."))
    return INSPIRE_NOTHING_TO_ACT_UNTESTED


def inspire_not_reliable(found, accepted, given):
    records, flagged = int(found.get("records") or 0), int(found.get("flagged_records") or 0)
    if accepted is False:
        text = str(given or "").strip() or INSPIRE_NOTHING_TO_ACT_UNACCEPTED
        if _inspire_re.search(r"[0-9]", text):
            return text
        return (f"{text.rstrip('.')}; the run did not say which threshold the model missed or what it measured, so "
                f"{inspire_rebuild_action()} so it names them.")
    if records and flagged >= records:
        top = found.get("top_flag")
        return INSPIRE_NOTHING_TO_ACT_FLAGGED + (f", most often: {top}." if top else ".")
    return None


def inspire_scope_narrowed(log):
    chosen = (log or {}).get("scope") or {}
    return chosen.get("scope") not in (None, "", INSPIRE_SCOPES[0]) and any(
        read.get("where") or read.get("limit") for read in ((log or {}).get("reads") or {}).values())


def inspire_small_groups_text(grouping):
    record = grouping if isinstance(grouping, dict) else {}
    named = record.get("group_by")
    if record.get("mode") != "columns" or not isinstance(named, list) or not named:
        return ""
    minimum = int(record.get("minimum") or INSPIRE_GROUP_MIN_RECORDS)
    levels = [level for level in record.get("levels") or [] if isinstance(level, dict)]
    fell = sum(int(level.get("records") or 0) for level in levels if level.get("level") == INSPIRE_SETTING_ALL)
    kept = [level for level in levels if level.get("level") != INSPIRE_SETTING_ALL]
    kept_records, groups = sum(int(level.get("records") or 0) for level in kept), sum(int(level.get("groups") or 0) for level in kept)
    by = ", ".join(str(name) for name in named)
    if fell and fell >= INSPIRE_SMALL_GROUP_SHARE * (fell + kept_records):
        unit = "period" if record.get("time_column") else "record"
        return (f"the Grouping by {by} leaves {inspire_number_text(fell, grouped=True)} of {inspire_plural(fell + kept_records, 'record')} "
                f"in groups of fewer than {inspire_plural(minimum, unit)}, so {'it is' if fell == 1 else 'they are'} compared with the whole data "
                f"instead of {'its' if fell == 1 else 'their'} own group")
    if groups and not record.get("time_column") and kept_records < INSPIRE_SMALL_GROUP_FACTOR * minimum * groups:
        return (f"the Grouping by {by} splits the records into {inspire_plural(groups, 'group')} of about "
                f"{inspire_plural(round(kept_records / groups), 'record')} each, too few for each group to learn from")
    return ""


def inspire_note_value_fact(name, scoped, whole, effect, log=None):
    """Record one distinct-value count the readiness step measured: `scoped` in the records this run scores, `whole` in the whole table.
    Returns the plain line the App shows when that count is why the run is not reliable."""
    log = (globals().get("INSPIRE_RUN_LOG") or {}) if log is None else log
    text = f"{name} holds {inspire_plural(int(whole), 'distinct value')} in the data, so {effect} cannot be measured"
    if isinstance(log, dict):
        log.setdefault("value_facts", []).append({"name": str(name), "scoped": int(scoped), "whole": int(whole), "text": text})
    return text

def inspire_first_run_left_out(chosen):
    """One plain clause for a model that learned from a First run: the records it left out, and how to run on all of them."""
    chosen = chosen or {}
    records, available = int(chosen.get("records") or 0), chosen.get("available")
    full = "choose Run on everything on the Run page to use all of them"
    if chosen.get("scope") != "records" or not records or available is None:
        return f"the First run covers only {inspire_scope_text(chosen)}, so the model learned from those records alone; {full}"
    available = int(available)
    if available <= records:
        return ""
    return (f"the First run left out {inspire_number_text(available - records, grouped=True)} of "
            f"{inspire_plural(available, 'record')}, so the model learned from only {inspire_plural(records, 'record')}; {full}")


def inspire_unreliable_cause(accepted=None, log=None, grouping=None, run=None):
    log = (globals().get("INSPIRE_RUN_LOG") or {}) if log is None else log
    grouping = globals().get("INSPIRE_RUN_GROUPING") if grouping is None else grouping
    run = (globals().get("INSPIRE_RUN") or {}) if run is None else run
    narrowed, causes = inspire_scope_narrowed(log), []
    facts = [fact for fact in (log.get("value_facts") or []) if isinstance(fact, dict) and fact.get("whole") is not None]
    causes.extend(fact["text"] for fact in facts if int(fact["whole"]) < 2 and fact.get("text"))
    moved = [fact for fact in facts if fact.get("scoped") is not None and int(fact["scoped"]) < 2 <= int(fact["whole"])]
    scope_decides = bool(moved) if facts else True
    if narrowed and scope_decides and ((run or {}).get("trains_model") or accepted is not None) and not log.get("training_reads"):
        left = inspire_first_run_left_out(log["scope"])
        if left:
            causes.append(left)
    small = inspire_small_groups_text(grouping)
    if small:
        causes.append(small + (" in the First run scope" if narrowed else "") + "; change the Grouping on the Run page to fewer "
                      "columns or all" + (", or choose a wider First run there" if narrowed else ""))
    return f"{INSPIRE_CAUSE_LEAD} {'; and '.join(causes)}." if causes else ""


def inspire_with_cause(reason, accepted=None):
    text = " ".join(str(reason or "").split())
    cause = inspire_unreliable_cause(accepted) if text else ""
    return f"{text.rstrip('.')}. {cause}" if cause and cause not in text else text

def inspire_capacity_seats_sql(stage, columns, key_names, share=0.0, dismissed_in=None, use_case_id=""):
    named = {str(c).lower(): c for c in columns}
    tier = inspire_column(inspire_tier_column(columns), "s")
    key = inspire_record_key_sql(key_names, "s")
    order = (f"{inspire_column(named['rank'], 's')} ASC NULLS LAST, " if "rank" in named else "") + f"{key} ASC"
    skip = [f"coalesce({inspire_held_back_sql(share, key, 's')}, FALSE)"] if share else []
    joined = ""
    if dismissed_in:
        joined = (f" LEFT JOIN (SELECT DISTINCT record_key FROM {inspire_table_name(dismissed_in)} WHERE use_case_id = "
                  f"{inspire_sql_literal(use_case_id)} AND {INSPIRE_CURRENT_ROWS} AND action_status = {inspire_sql_literal(inspire_shell_facts.DISMISSED)}) AS d "
                  f"ON d.record_key = {key}")
        skip.append("d.record_key IS NOT NULL")
    kept = f" AND NOT ({' OR '.join(skip)})" if skip else ""
    return (f"SELECT {key} AS k, ROW_NUMBER() OVER (ORDER BY {order}) AS seat FROM {inspire_table_name(stage)} AS s{joined} "
            f"WHERE {tier} = 'Act now'{kept}")


def inspire_capacity_merge_sql(stage, columns, key_names, cap, seats):
    tier = inspire_column(inspire_tier_column(columns))
    return (f"MERGE INTO {inspire_table_name(stage)} AS t USING (SELECT k FROM ({seats}) AS q WHERE seat > {int(cap)}) AS c "
            f"ON {inspire_record_key_sql(key_names, 't')} = c.k WHEN MATCHED THEN UPDATE SET {tier} = 'Review'")


def inspire_apply_capacity(spark, table, stage, columns, key_names, share, existing, use_case_id, cap, out=print):
    if not cap or not inspire_tier_column(columns):
        return None
    kept = {str(name).lower() for name, _kind in existing or []}
    dismissed_in = table if {"record_key", "is_current", "change_source", "action_status"} <= kept else None
    seats = inspire_capacity_seats_sql(stage, columns, key_names, share, dismissed_in, use_case_id)
    passed = int((inspire_rows(spark.sql(f"SELECT COUNT(*) AS seats FROM ({seats}) AS q")) or [{}])[0].get("seats") or 0)
    over = max(0, passed - int(cap))
    if over:
        spark.sql(inspire_capacity_merge_sql(stage, columns, key_names, cap, seats))
        out(f"Team capacity: {inspire_plural(passed, 'record')} passed the Act now test and the team acts on {inspire_number_text(cap, grouped=True)} each run, so the top "
            f"{inspire_plural(cap, 'record')} by rank {'stays' if cap == 1 else 'stay'} Act now and the other {inspire_plural(over, 'record moves', 'records move')} "
            f"to Review, where the App says the capacity holds {'it' if over == 1 else 'them'} back")
    else:
        out(f"Team capacity: {inspire_plural(passed, 'record')} passed the Act now test, within the {inspire_number_text(cap, grouped=True)} the team acts on each run")
    return {"limit": int(cap), "over": over}

def inspire_stake_amount(raw):
    try:
        amount = float(raw)
    except (TypeError, ValueError):
        return None
    return amount if amount == amount and amount not in (float("inf"), float("-inf")) else None


def inspire_record_stake(spark, table, use_case_id, run_id, found, value, out=print):
    if not value:
        return
    try:
        database = inspire_widget("02_inspire_database")
        session = inspire_session_id(inspire_widget("14_session_id") or (globals().get("INSPIRE_RUN") or {}).get("session_id") or "")
        if not database or not session:
            out(f"Value ledger: run {run_id} is not recorded, since the run has no Inspire database or session id")
            return
        money = value.get("kind", "money") == "money"
        unit = inspire_value_unit(value)
        row = inspire_shell_facts.ledger_stake_row(session, table, use_case_id, run_id, value.get("kind", "money"),
                                                   unit if money else "", "" if money else unit,
                                                   inspire_stake_amount(found.get(INSPIRE_VALUE_AT_STAKE)))
        ledger = inspire_shell_facts.ledger_table(database)
        spark.sql(inspire_shell_facts.ledger_create_sql(ledger))
        spark.sql(inspire_shell_facts.ledger_insert_sql(ledger, [row]))
    except Exception as error:
        out(f"Value ledger: run {run_id} is not recorded: {inspire_redact_text(str(error), set(), '')[:300]}")


def inspire_save_outcome(spark, table, use_case_id, run_id, columns, value, given, out=print, formula=None, expression=None,
                         accepted=None, capacity=None):
    if "priority_tier" not in {str(c).lower() for c in columns}:
        return None
    try:
        found = (inspire_rows(spark.sql(inspire_outcome_sql(table, use_case_id, columns, value))) or [{}])[0]
        found[INSPIRE_VALUE_AT_STAKE] = (inspire_rows(spark.sql(inspire_run_total_sql(table, use_case_id, columns, value))) or [{}])[0].get("value")
        reason = inspire_nothing_to_act(found, value, given)
        unreliable = inspire_not_reliable(found, accepted, given)
        unreliable = inspire_with_cause(unreliable, accepted) if unreliable else unreliable
        values = {name: str(int(found.get(name) or 0)) for name in ("act_now_records", "review_records", "records")}
        values["nothing_to_act"] = "NULL" if reason is None else inspire_sql_literal(reason)
        if value:
            act_now, review = inspire_value_amounts(value)
            values.update({INSPIRE_VALUE_AT_STAKE: inspire_sql_number(found.get(INSPIRE_VALUE_AT_STAKE)),
                           "currency": inspire_sql_literal(value["currency"]), "decision_basis": inspire_sql_literal(value["basis"]),
                           "act_now_amount": inspire_sql_number(act_now), "review_amount": inspire_sql_number(review)},
                          **inspire_value_columns_sql(value, formula, expression))
        if value:
            values.update(impact_kind=inspire_sql_literal(value.get("kind", "money")), impact_unit=inspire_sql_literal(inspire_value_unit(value)))
        if capacity:
            values.update(act_now_capacity=str(int(capacity["limit"])), over_capacity_records=str(int(capacity["over"])))
        if value and value.get("basis") == "predicted" and value.get("passed") is not None:
            shares = [inspire_amount((value.get("shares") or {}).get(name)) for name in INSPIRE_SHARE_NAMES]
            values.update(passing_records=str(int(value["passed"])), act_now_share=inspire_sql_number(shares[0]),
                          review_share=inspire_sql_number(shares[1]))
        if unreliable:
            values["not_reliable"] = inspire_sql_literal(unreliable)
        inspire_ensure_summary(spark, table)
        spark.sql(inspire_ai_summary_merge_sql(table, use_case_id, run_id, values))
    except Exception as error:
        out(f"Business output: run {run_id} wrote its rows, but Inspire could not record how many of them need action in "
            f"{inspire_ai_summary_table(table)}: {inspire_redact_text(str(error), set(), '')[:300]}")
        return None
    inspire_record_stake(spark, table, use_case_id, run_id, found, value, out)
    counts = (f"{inspire_number_text(values['act_now_records'], grouped=True)} Act now and "
              f"{inspire_number_text(values['review_records'], grouped=True)} Review of {inspire_plural(values['records'], 'current record')}")
    if unreliable:
        out(f"Business output: {counts}, so the App shows the run as not reliable yet, and its alert and weekly email say so too: "
            f"{unreliable}")
    elif reason is None:
        ignored = " (the nothing_to_act reason the run gave was left out, since records need action)" if str(given or "").strip() else ""
        out(f"Business output: {counts}{ignored}")
    else:
        source = "the run's own reason" if str(given or "").strip() else "Inspire's reason, since the run gave no nothing_to_act"
        out(f"Business output: {counts}, so the App shows nothing to act on yet, with {source}: {reason}")
    return reason


def inspire_ai_save_config(spark, table, use_case_id, run_id, ai):
    config = dict(ai, hedge=INSPIRE_AI_HEDGE, rows_per_second=INSPIRE_AI_ROWS_PER_SECOND, unexplained=INSPIRE_AI_UNEXPLAINED)
    inspire_ensure_summary(spark, table)
    spark.sql(inspire_ai_summary_merge_sql(table, use_case_id, run_id, {
        "ai_endpoint": inspire_sql_literal(ai["endpoint"]),
        "ai_config": inspire_sql_literal(_inspire_json.dumps(config, ensure_ascii=False, sort_keys=True))}))


def inspire_ai_checks_json(report):
    found = {name: report.get(name) for name in ("records", "current", "flagged", "later", "missing", "one_level", "error")}
    return _inspire_json.dumps(dict(found, kinds=dict(sorted((report.get("kinds") or {}).items()))), ensure_ascii=False, sort_keys=True)


def inspire_ai_save_checks(spark, table, use_case_id, run_id, report, out=print):
    try:
        inspire_ensure_summary(spark, table)
        spark.sql(inspire_ai_summary_merge_sql(table, use_case_id, run_id,
                                               {"ai_checks": inspire_sql_literal(inspire_ai_checks_json(report))}))
    except Exception as error:
        out(f"AI reasoning: Inspire could not record the AI checks of run {run_id} in {inspire_ai_summary_table(table)}: "
            f"{inspire_redact_text(str(error), set(), '')[:300]}")


def inspire_ai_after_merge(spark, table, use_case_id, run_id, ai, cap, out=print):
    if not ai:
        return None
    try:
        inspire_ai_save_config(spark, table, use_case_id, run_id, ai)
    except Exception as error:
        out(f"AI reasoning: Inspire could not record the AI settings of run {run_id} in {inspire_ai_summary_table(table)}: "
            f"{inspire_redact_text(str(error), set(), '')[:300]}")
    if not ai["endpoint"]:
        out("AI reasoning: the ai_endpoint widget is empty, so no record was explained")
        inspire_ai_save_checks(spark, table, use_case_id, run_id, {"error": "the ai_endpoint widget is empty"}, out)
        return None
    try:
        report = inspire_ai_explain(spark, table, use_case_id, ai, cap, out=out)
    except Exception as error:
        out(f"AI reasoning: run {run_id} wrote its rows, but Inspire could not explain them with {ai['endpoint']}: "
            f"{inspire_redact_text(str(error), set(), '')[:300]}")
        inspire_ai_save_checks(spark, table, use_case_id, run_id, {"error": f"Inspire could not explain the records with "
                                                                            f"{ai['endpoint']}: {inspire_redact_text(str(error), set(), '')[:300]}"}, out)
        return None
    inspire_ai_save_checks(spark, table, use_case_id, run_id, report, out)
    return report


def inspire_ai_summary_prompt(context):
    return "\n".join([str(context or "").strip(), "",
                      INSPIRE_AI_USER_WINS, "",
                      "Write the summary of this run of the use case for the people who act on its results: 3 to 5 plain "
                      "sentences on how ready the run is, how many records need action now and why, what changed against "
                      "earlier runs, and what to do first. The " + INSPIRE_SUMMARY_DECISION_FACT + " fact is Inspire's final "
                      "count: the records that need action now are its Act now ones, and when it gives why nothing is on the "
                      "to fix list, that is why. State the facts as facts. Reply with the summary text only, with no "
                      "heading, list or JSON. Never write about yourself, about the facts or this message, or about what is not "
                      'known, and never use the word "I".', "", "Facts:", ""]).lstrip()


def inspire_summary_decision_fact(session, table, use_case_id, run_id, out=print):
    try:
        row = (inspire_rows(session.sql(
            f"SELECT act_now_records, review_records, records, nothing_to_act "
            f"FROM {inspire_table_name(inspire_ai_summary_table(table))} WHERE use_case_id = {inspire_sql_literal(use_case_id)} AND run_id = {inspire_sql_literal(run_id)}")) or [{}])[0]
    except Exception as error:
        out(f"AI summary: Inspire could not read how many records of run {run_id} need action, so the summary is written without "
            f"that count: {inspire_redact_text(str(error), set(), '')[:300]}")
        return ""
    if row.get("records") is None:
        return ""
    fact = (f"{INSPIRE_SUMMARY_DECISION_FACT}: {inspire_number_text(int(row.get('act_now_records') or 0), grouped=True)} Act now and "
            f"{inspire_number_text(int(row.get('review_records') or 0), grouped=True)} Review of {inspire_plural(int(row['records']), 'current record')}")
    reason = str(row.get("nothing_to_act") or "").strip()
    return fact + (f"; nothing is on the to fix list, because: {reason}" if reason else "")


def inspire_data_age_fact(as_of_date, today=None):
    try:
        day = inspire_parse_time(str(as_of_date or "").strip()[:10]).date()
    except ValueError:
        return ""
    days = ((today or _inspire_datetime.date.today()) - day).days
    if days <= 0:
        return ""
    return f"The run is as of {day.isoformat()}, {inspire_number_text(days, grouped=True)} day{'' if days == 1 else 's'} before the day it ran."


def inspire_cutoff_day(as_of_date) -> str:
    try:
        return inspire_parse_time(str(as_of_date or "").strip()[:10]).date().isoformat()
    except ValueError:
        return ""


def inspire_cutoff_tables() -> list:
    """The tables the as-of cut-off applies to: each (table, time column) the run reads through a date column, in the run's order."""
    out = []
    for item in (globals().get("INSPIRE_RUN") or {}).get("cut_off_tables") or []:
        table, column = (str(item.get("table") or "").strip(), str(item.get("column") or "").strip()) if isinstance(item, dict) else ("", "")
        if table and column:
            out.append((table, column))
    return out


def inspire_cutoff_counts(spark, as_of_date, out=print) -> list:
    """(table, rows dated on the as-of day) for each table the cut-off applies to. A count that cannot be read is None.
    Empty when no table is read through a time column, or the as-of date is unreadable: then nothing is left out by date."""
    day = inspire_cutoff_day(as_of_date)
    counts = []
    for table, column in (inspire_cutoff_tables() if day else []):
        try:
            counted = inspire_rows(spark.sql(f"SELECT COUNT(*) AS n FROM {inspire_table_name(table)} WHERE CAST({inspire_column(column)} AS DATE) "
                                             f"= CAST({inspire_sql_literal(day)} AS DATE)"))[0]
            counts.append((table, int(counted.get("n") or 0)))
        except Exception as error:
            out(f"Summary: Inspire could not count the rows on {day} that the cut-off leaves out of {table}: {inspire_redact_text(str(error), set(), '')[:300]}")
            counts.append((table, None))
    return counts


def inspire_cutoff_fact(as_of_date, counts=()) -> str:
    """The as-of cut-off sentence. One table: its rows dated on the day, as one total. Several: each table's count. No table: no sentence."""
    day = inspire_cutoff_day(as_of_date)
    if not day or not counts:
        return ""
    if len(counts) == 1:
        _table, left_out = counts[0]
        if left_out is None:
            return f"Reads rows dated before {day}; rows on {day} are left out because that day may be incomplete."
        return f"Reads rows dated before {day}; {inspire_plural(left_out, 'row')} on {day} {'is' if left_out == 1 else 'are'} left out because that day may be incomplete."
    pieces = [f"{table} ({inspire_plural(left_out, 'row') if left_out is not None else 'count unavailable'})" for table, left_out in counts]
    listed = ", ".join(pieces[:-1]) + " and " + pieces[-1]
    return f"Reads rows dated before {day}; rows on {day} are left out of {listed} because that day may be incomplete."


def inspire_deterministic_summary(as_of_date, counts=()) -> str:
    return " ".join(part for part in (inspire_data_age_fact(as_of_date), inspire_cutoff_fact(as_of_date, counts)) if part)


def inspire_store_summary(session, table, run_id, as_of_date, summary, endpoint):
    run = globals().get("INSPIRE_RUN") or {}
    inspire_ensure_summary(session, table)
    inspire_require_no_python_repr_text(summary, f"the summary of run {run_id}")
    session.sql(inspire_ai_summary_merge_sql(table, str(run.get("use_case_id") or ""), str(run_id), {
        "summary": inspire_sql_literal(summary), "as_of_date": f"try_cast({inspire_sql_literal(as_of_date)} AS DATE)",
        "ai_endpoint": inspire_sql_literal(endpoint)}))


def inspire_write_summary(facts, result_table, run_id, as_of_date, spark_session=None, endpoint=None, context=None, out=print):
    session = spark_session if spark_session is not None else globals()["spark"]
    table = ".".join(inspire_table_parts(result_table))
    inspire_run_stage(INSPIRE_RUN_STAGES[4], out=out)
    text = facts if isinstance(facts, str) else _inspire_json.dumps(facts, default=str, ensure_ascii=False, sort_keys=True)
    if not str(text or "").strip():
        raise ValueError("inspire_write_summary needs the facts of the run's summary")
    age = inspire_data_age_fact(as_of_date)
    text = f"{text}\n{age}" if age else text
    trained = inspire_training_text()
    text = f"{text}\nTraining and scoring: {trained}." if trained else text
    endpoint = inspire_resolved_model(str(inspire_widget("ai_endpoint") if endpoint is None else endpoint).strip(), session)
    deterministic = inspire_deterministic_summary(as_of_date, inspire_cutoff_counts(session, as_of_date, out))
    if not endpoint:
        out("AI summary: the ai_endpoint widget is empty, so the run has no AI summary")
        if deterministic:
            inspire_store_summary(session, table, run_id, as_of_date, deterministic, "")
        return deterministic or None
    run = globals().get("INSPIRE_RUN") or {}
    decided = inspire_summary_decision_fact(session, table, str(run.get("use_case_id") or ""), str(run_id), out)
    prompt = (inspire_ai_summary_prompt(run.get("ai_context", "") if context is None else context)
              + (f"{decided}\n" if decided else "") + str(text))
    summary, problem = None, ""
    for _attempt in (1, 2):
        asked = prompt + (f"\n\nYour last reply was rejected ({problem}). Write it again following every rule above." if problem else "")
        reply = (inspire_rows(session.sql(f"SELECT ai_query({inspire_sql_literal(endpoint)}, {inspire_sql_literal(asked)}, "
                                          "failOnError => false) AS r")) or [{}])[0].get("r") or {}
        result, error = str(reply["result"] or "").strip() if reply else "", reply["errorMessage"] if reply else "no reply"
        hedged = _inspire_re.search(INSPIRE_AI_HEDGE, result)
        problem = (f"endpoint error: {error}" if error else "empty reply" if not result else
                   f"it wrote about yourself or the facts ('{hedged.group(0)}')" if hedged else "it wrote a Python object dump" if _inspire_re.search(INSPIRE_PYTHON_REPR, result) else "")
        if not problem:
            summary = result
            break
    if summary is None:
        out(f"AI summary: run {run_id} has no AI summary, since {endpoint} gave none Inspire could use after one retry: {problem[:300]}")
        if deterministic:
            inspire_store_summary(session, table, run_id, as_of_date, deterministic, endpoint)
        return deterministic or None
    stored = f"{summary}\n{deterministic}" if deterministic else summary
    inspire_store_summary(session, table, run_id, as_of_date, stored, endpoint)
    out(f"AI summary: wrote the summary of run {run_id} to {inspire_ai_summary_table(table)}")
    return summary


def inspire_sql_retrying(spark, sql, table, doing, again, out=print):
    attempts = inspire_write_conflict_attempts()

    def _again(attempt, _wait, _error):
        out(f"Business output: {table} changed while {doing}, such as an edit in the App, so {again} runs again "
            f"(attempt {attempt + 1} of {attempts})")
    return inspire_retry_write_conflicts(lambda: inspire_rows(spark.sql(sql)), f"{doing} on {table}", max_attempts=attempts,
                                         on_retry=_again)


def inspire_no_data_reason(reason) -> bool:
    return bool(INSPIRE_NO_DATA_RE.search(" ".join(str(reason or "").split())))


def inspire_create_empty_result(spark, table, stage, columns, out=print):
    now = inspire_now_sql(inspire_now_us(spark))
    spark.sql(inspire_history_create_sql(table, stage, columns, ("entity_id",), None, {}, now))
    tier = inspire_tier_column(columns)
    inspire_run_log_make(spark, table, inspire_table_schema(spark, table) or [], tier, backfill=False)
    inspire_run_log_switch(spark, table, tier, out)
    inspire_share_outputs(spark, table, out)
    out(f"Business output: created {table} with no rows yet, so the App and the Genie Agent can read it: {INSPIRE_NO_DATA_YET}")


def inspire_create_empty_core_result(spark, table, out=print):
    if inspire_table_schema(spark, table) is not None:
        return False
    catalog, schema, name = inspire_table_parts(table)
    stage = f"{catalog}.{schema}.{name}_inspire_stage"
    picked = ", ".join(f"CAST(NULL AS {kind}) AS {inspire_column(column)}" for column, kind in INSPIRE_CORE_COLUMNS)
    spark.sql(f"CREATE OR REPLACE TABLE {inspire_owned_table(stage)} AS SELECT {picked} WHERE FALSE")
    try:
        inspire_create_empty_result(spark, table, stage, [column for column, _kind in INSPIRE_CORE_COLUMNS], out)
    finally:
        spark.sql(f"DROP TABLE IF EXISTS {inspire_owned_table(stage)}")
    return True


def inspire_merge_run(spark, table, stage, out=print, ai=None, ai_cap=INSPIRE_AI_AUTO_ROWS, value=None, nothing_to_act=None, holdout=None,
                      accepted=None, capacity=None):
    holdout = inspire_holdout_share() if holdout is None else holdout
    capacity = inspire_act_now_capacity() if capacity is None else capacity
    owned = set(INSPIRE_HISTORY_NAMES) | set(INSPIRE_ACTION_NAMES) | set(INSPIRE_HOLDOUT_NAMES) | set(INSPIRE_KEPT_NAMES)
    staged = inspire_table_schema(spark, stage) or []
    reasoning = {str(name).lower() for name, _kind in staged if inspire_ai_owned(name)} if ai else set()
    given = {str(name).lower() for name, _kind in staged if str(name).lower() == INSPIRE_VALUE_FORMULA} if value else set()
    schema = [(name, kind) for name, kind in staged if str(name).lower() not in owned | reasoning | given]
    ignored = [name for name, _kind in staged if str(name).lower() in INSPIRE_ACTION_NAMES]
    if reasoning:
        out(f"AI reasoning: left out {', '.join(sorted(reasoning))} from the run's rows: Inspire writes them from each row's "
            f"{INSPIRE_AI_FACTS}")
    if given:
        out(f"{inspire_stake_label(value)}: left out {INSPIRE_VALUE_FORMULA} from the run's rows: Inspire writes it from "
            f"{INSPIRE_VALUE_FORMULA_NAME} and each row's numbers")
    columns = [name for name, _kind in schema]
    wanted = INSPIRE_RUN_KEY_COLUMNS + ((INSPIRE_CONFIDENCE,) if ai else ()) + ((INSPIRE_VALUE_AT_STAKE,) if value else ())
    absent = [name for name in wanted if name not in {str(c).lower() for c in columns}]
    if absent:
        raise ValueError(f"Inspire did not write the business output: the rows have no {', '.join(absent)} column, and every row needs it")
    formula, formula_problems = inspire_value_formula(value, schema) if value else (None, [])
    expression, expression_problems = inspire_value_expression(value, schema, formula) if formula else (None, [])
    expression_problems = expression_problems + (inspire_value_bounds(value, formula) if formula else [])
    profile = inspire_rows(spark.sql(inspire_run_profile_sql(stage, columns, value, formula, expression)))[0]
    rows = int(profile.get("row_count") or 0)
    inspire_run_stage(INSPIRE_RUN_STAGES[2], rows_done=0, rows_total=rows, out=out)
    if not rows:
        if inspire_table_schema(spark, table) is None:
            inspire_create_empty_result(spark, table, stage, columns, out)
            return 0
        out(f"Business output: the run produced no rows, so {table} was left as it was")
        return 0
    problems = (inspire_value_problems(value) + formula_problems + expression_problems if value else []) + inspire_run_problems(profile, value)
    id_unique, label_unique = bool(profile.get("id_unique")), bool(profile.get("label_unique"))
    key_names = inspire_record_key_rule(columns, id_unique, label_unique)
    if not problems and not ((key_names == ("entity_id",) and id_unique) or (key_names == ("entity_label",) and label_unique)):
        repeats = inspire_rows(spark.sql(inspire_key_repeats_sql(stage, key_names)))
        if repeats:
            shown = ", ".join(f"{r.get('record_key')!r} ({inspire_plural(r.get('row_count'), 'row')})" for r in repeats)
            problems.append(f"rows share the record key {' with '.join(key_names)} or leave it empty ({shown}); every row needs its own "
                            "entity_id, or its own entity_id with segment_or_item")
    if problems:
        raise ValueError("Inspire did not write the business output: " + "; ".join(problems))
    stake_total = profile.get("stake_total")
    try:
        spec = inspire_value_formula_spec(value, formula, {"exposure": float(stake_total), "period_days": None}) \
            if value and formula and stake_total is not None else None
        priced = inspire_value_calculate(spec) if spec else None
        if priced and not priced["ok"]:
            out(f"Value spec: not priced ({', '.join(priced['problems'])})")
        elif priced:
            base_row = (priced.get("scenarios") or {}).get("base") or {}
            gross, change = base_row.get("gross_annual"), base_row.get("change")
            shown = (f"the base gross over a year is {inspire_money_short(gross)}" if gross is not None
                     else f"the base change is {inspire_money_short(change)} {priced['unit'] or ''}".rstrip() if change is not None
                     else "no base value")
            out("Value spec: " + shown + (f" (flags: {', '.join(priced['flags'])})" if priced["flags"] else ""))
    except Exception as error:
        out(f"Value spec: not priced ({type(error).__name__}: {str(error)[:200]})")
    derived = {INSPIRE_VALUE_FORMULA: inspire_value_formula_sql(formula, value, "s")} if formula else {}
    columns, schema = columns + list(derived), schema + [(name, "STRING") for name in derived]
    use_case_id, run_id, keyed = str(profile["use_case_id"]), str(profile["run_id"]), " with ".join(key_names)
    if ignored:
        out(f"Business output: left out {', '.join(ignored)} from the run's rows: people set them in the App, and each run carries "
            "them forward from the record's current version")
    existing = inspire_table_schema(spark, table)
    if accepted is False and inspire_tier_column(columns):
        spark.sql(inspire_unaccepted_sql(stage, columns))
        out("Not reliable yet: the run passed accepted=False, so Inspire writes its 'Act now' and 'Review' rows as 'Monitor'")
    seated = None if accepted is False else inspire_apply_capacity(spark, table, stage, columns, key_names, holdout, existing,
                                                                   use_case_id, capacity, out)
    held, held_kinds = inspire_holdout_derived(columns, key_names, holdout, existing)
    derived = dict(derived, **held)
    added = [name for name in held if name not in columns]
    columns, schema = columns + added, schema + [(name, held_kinds[name]) for name in added]
    if holdout and held:
        out(f"Holdout: {inspire_number_text(round(holdout * 100, 1))}% of the records, picked by their record key, are held back: a "
            f"flagged one is '{INSPIRE_HELD_BACK}' instead of on the to fix list, so the App can compare them with the records people act on")
    tier = inspire_tier_column(columns)
    if existing is None:
        now_us = inspire_now_us(spark)
        now = inspire_now_sql(now_us)
        spark.sql(inspire_history_create_sql(table, stage, columns, key_names, ai, derived, now))
        created = inspire_table_schema(spark, table) or []
        inspire_run_log_make(spark, table, created, tier, backfill=False)
        inspire_run_log_switch(spark, table, tier, out)
        spark.sql(inspire_run_log_insert_sql(table, use_case_id, run_id, now, tier, [name for name, _kind in created]))
        out(f"Business output: created {table} with the {inspire_plural(rows, 'row')} of run {run_id} as the current version of each record, "
            f"keyed by {keyed}")
        inspire_save_outcome(spark, table, use_case_id, run_id, columns, value, nothing_to_act, out, formula, expression, accepted, seated)
        inspire_save_run_log(spark, table, use_case_id, run_id, out)
        inspire_save_kpi(spark, table, use_case_id, run_id, out=out)
        inspire_ai_after_merge(spark, table, use_case_id, run_id, ai, ai_cap, out)
        inspire_run_facts(spark, table, use_case_id, run_id, tier, now_us, out)
        inspire_share_outputs(spark, table, out)
        inspire_cluster_run(spark, table, out)
        return rows
    log = inspire_run_log_table(table)
    has_log = inspire_table_schema(spark, log) is not None
    before = {table: inspire_table_version(spark, table), log: inspire_table_version(spark, log) if has_log else None}
    try:
        replace_edits = inspire_replace_edits(spark, table)
        retype, changed = inspire_type_change_sql(table, existing, schema, replace_edits)
        if retype:
            held = inspire_run_log_properties(spark, table)
            spark.sql(retype)
            if held:
                spark.sql(inspire_run_log_properties_sql(table, held))
            out(f"Business output: converted {', '.join(changed)} in {table} to the types this run writes")
        added = inspire_add_required_columns(spark, table, schema, existing={str(name).lower() for name, _kind in existing})
        if added:
            out(f"Business output: added {', '.join(added)} to {table}; the rows already there hold NULL in them")
        if ai:
            inspire_ensure_ai(spark, table, existing, ai, out=out)
        unwritten = [name for name, _kind in existing if str(name).lower() not in {str(c).lower() for c in columns}
                     and str(name).lower() not in owned and not inspire_ai_owned(name)]
        if unwritten:
            out(f"Business output: kept {', '.join(unwritten)} in {table}, which this run does not write, so its rows hold NULL there")
        now_schema = inspire_table_schema(spark, table) or []
        written = {str(c).lower() for c in list(columns) + list(INSPIRE_HISTORY_NAMES) + list(INSPIRE_ACTION_NAMES)
                   + [name for name, _kind in inspire_ai_columns(ai)]}
        left_null = [name for name, _kind in now_schema if str(name).lower() not in written]
        stamped = inspire_run_log_stamps(spark, table, tier)
        now_us = inspire_now_us(spark)
        now = inspire_now_sql(now_us)
        merge = inspire_history_merge_sql(table, stage, columns, key_names, use_case_id, replace_edits, ai, derived, now, left_null,
                                          now_schema, stamped)
        merged = inspire_sql_retrying(spark, merge, table, f"run {run_id} merged into it", "the merge", out)
        scope_key = inspire_run_scope_key()
        scope_keys = inspire_scope_keys(spark, table, scope_key, out) if stamped or has_log else None
        logged = ((inspire_rows(spark.sql(inspire_run_log_insert_sql(table, use_case_id, run_id, now, tier,
                                                                     [name for name, _kind in now_schema], scope_keys, scope_key))) or [{}])[0]
                  if stamped or has_log else {})
        kept = "" if replace_edits else inspire_mark_kept(spark, table, stage, key_names, use_case_id, run_id, out)
    except Exception as error:
        inspire_restore_run(spark, table, before, error, out)
        raise
    counts = merged[0] if merged else {}
    inserted, updated = int(counts.get("num_inserted_rows", rows) or 0), int(counts.get("num_updated_rows", 0) or 0)
    listed = int(logged.get("num_inserted_rows", logged.get("num_affected_rows", inserted)) or 0)
    restamped = max(listed - inserted, 0)
    wrote = (f"wrote {inspire_plural(inserted, 'new version')} to {table} for the records that are new or changed and kept "
             f"{inspire_plural(restamped, 'record')} unchanged, closing "
             f"{inspire_plural(max(updated - restamped, 0), 'earlier one')}" if stamped else
             f"wrote {inspire_plural(inserted, 'version')} to {table}, one for each of its records, closing "
             f"{inspire_plural(updated, 'earlier one')}")
    out(f"Business output: run {run_id} {wrote}, keyed by {keyed}; "
        + ("runs replace the records people changed in the App, as the use case's settings ask" if replace_edits else kept)
        + f"; {inspire_action_summary(spark, table, stage, columns, key_names)}"
        + (f"; the table as it was before this run is its version {before[table]}, which RESTORE TABLE {table} TO VERSION AS OF "
           f"{before[table]} brings back" if before[table] is not None else ""))
    inspire_save_outcome(spark, table, use_case_id, run_id, columns, value, nothing_to_act, out, formula, expression, accepted, seated)
    inspire_save_run_log(spark, table, use_case_id, run_id, out)
    inspire_save_kpi(spark, table, use_case_id, run_id, out=out)
    inspire_ai_after_merge(spark, table, use_case_id, run_id, ai, ai_cap, out)
    if stamped or has_log:
        inspire_run_facts(spark, table, use_case_id, run_id, tier, now_us, out, scope_keys, scope_key)
    inspire_share_outputs(spark, table, out)
    inspire_cluster_run(spark, table, out)
    return rows


def _implement_table_parts(table: str):
    parts = [p.strip().strip("`") for p in str(table or "").split(".")]
    return parts if len(parts) == 3 and all(parts) else None


def inspire_existing_sql(count: int) -> str:
    return ("SELECT lower(table_name) AS t FROM system.information_schema.tables WHERE table_catalog = lower(:c) "
            f"AND table_schema = lower(:s) AND lower(table_name) IN ({', '.join(f':t{i}' for i in range(count))})")


def inspire_share_reads_sql() -> str:
    return ("SELECT lower(table_name) AS t, grantee AS p FROM system.information_schema.table_privileges WHERE table_catalog = lower(:c) "
            "AND table_schema = lower(:s) AND privilege_type = 'SELECT'")


def inspire_principal_ok(principal) -> bool:
    return bool(principal) and "`" not in principal and principal.isprintable()


def inspire_share_plan(table: str, existing, grants, reads) -> list:
    parts = _implement_table_parts(table)
    if not parts:
        return []
    held = {}
    for row in grants or ():
        principal = str(row.get("p") or "").strip()
        if inspire_principal_ok(principal):
            held.setdefault(str(row.get("t") or "").lower(), set()).add(principal)
    readers = sorted(held.get(parts[2].lower(), set()))
    have = {str(n).lower() for n in existing or ()}
    out = []
    for name in reads:
        mine = _implement_table_parts(name)
        if mine == parts:
            continue
        if mine and mine[2].lower() in have and mine[:2] == parts[:2]:
            out += [f"GRANT SELECT ON TABLE {inspire_table_name(name)} TO {inspire_quote_principal(p)}" for p in readers if p not in held.get(mine[2].lower(), set())]
    return out


def inspire_share_outputs(spark, table, out=print, run=None, readers=(), failed=None) -> list:
    run = (globals().get("INSPIRE_RUN") or {}) if run is None else run
    reads = [str(name) for name in (run or {}).get("app_reads") or []]
    parts = _implement_table_parts(table)
    if not parts or not reads:
        return []
    names = [(_implement_table_parts(n) or ["", "", ""])[2] for n in reads]
    try:
        marks = {f"t{i}": str(n).lower() for i, n in enumerate(names)}
        existing = [r.get("t") for r in inspire_rows(spark.sql(inspire_existing_sql(len(names)), args={"c": parts[0], "s": parts[1], **marks}))]
        grants = (inspire_rows(spark.sql(inspire_share_reads_sql(), args={"c": parts[0], "s": parts[1]}))
                  + [{"t": parts[2].lower(), "p": str(p or "").strip()} for p in readers or ()])
    except Exception as error:
        said = (f"App reads: Inspire could not check who reads {table}, so a table this run made may stay closed to the App until the "
                f"next run: {inspire_redact_text(str(error), set(), '')[:300]}")
        out(said)
        if failed is not None:
            failed.append(said)
        return []
    missed = []
    plan = inspire_share_plan(table, existing, grants, reads)
    for statement in plan:
        try:
            spark.sql(statement)
        except Exception as error:
            missed.append(f"{statement} failed: {inspire_redact_text(str(error), set(), '')[:200]}")
    if failed is not None:
        failed.extend(missed)
    if plan:
        out(f"[APP-READS FIRED] App reads: whoever reads {table} now reads the tables this run made beside it "
            f"({inspire_plural(len(plan) - len(missed), 'grant')} made" + (f"; {inspire_number_text(len(missed), grouped=True)} failed: {'; '.join(missed)[:600]}" if missed else "") + ")")
    return plan


def inspire_cluster_run(spark, table, out):
    try:
        spark.sql(f"OPTIMIZE {inspire_table_name(table)}")
    except Exception as error:
        out(f"Layout: Inspire could not cluster the versions this run wrote to {table}, so the App's first records read more files "
            f"until a later run clusters them: {inspire_redact_text(str(error), set(), '')[:300]}")
        return
    out(f"Layout: clustered the versions this run wrote to {table}, so the App's first records read only the files that hold them")


def inspire_mark_kept(spark, table, stage, key_names, use_case_id, run_id, out=print):
    inspire_sql_retrying(spark, inspire_kept_marks_sql(table, stage, key_names, use_case_id), table,
                         f"run {run_id} marked the records it keeps", "the marking", out)
    found = (inspire_rows(spark.sql(inspire_kept_edits_sql(table, use_case_id))) or [{}])[0]
    kept, missing = int(found.get("kept") or 0), int(found.get("missing") or 0)
    return (f"kept {inspire_plural(kept, 'record')} people changed in the App, since runs keep those changes"
            + (f"; {inspire_number_text(missing, grouped=True)} of them are not in this run, so the App tags them '{INSPIRE_NOT_IN_RUN}'" if missing else ""))


def inspire_action_summary(spark, table, stage, columns, key_names):
    try:
        found = (inspire_rows(spark.sql(inspire_action_counts_sql(table, stage, columns, key_names))) or [{}])[0]
    except Exception as error:
        return f"the actions people took in the App were carried forward, but Inspire could not count them: {inspire_redact_text(str(error), set(), '')[:300]}"
    dismissed, kept_out, resolved, flagged = (int(found.get(name) or 0) for name in ("dismissed", "kept_out", "resolved", "still_flagged"))
    held = (f"; held back {inspire_plural(int(found.get('held_back') or 0), 'flagged record')} from the to fix list to measure the impact"
            if "held_back" in found else "")
    return (f"kept {inspire_plural(dismissed, 'dismissed record')} in Monitor ({inspire_number_text(kept_out, grouped=True)} out of Act now), and wrote "
            f"{inspire_plural(resolved + flagged, 'outcome')} for records people marked Done ({inspire_number_text(resolved, grouped=True)} Resolved, {inspire_number_text(flagged, grouped=True)} Still flagged){held}")

    "min_pr_auc_lift": "No positive outcome in this data; the PR-AUC lift over the base rate is not measured.",
    "min_c_index": "No comparable event pairs in this data; the concordance index is not measured.",
    "min_precision_lift": "No popular item was a hit in the holdout; the precision lift over popularity is not measured.",
    "min_silhouette": "Fewer than two segments in this data; the silhouette score is not measured.",
    "min_bootstrap_ari": "Fewer than two segments in this data; the bootstrap adjusted Rand index is not measured.",
    "min_segment_share": "No segment in this data; the smallest segment share is not measured.",
    "max_mase": "The seasonal naive forecast has no error in this data; MASE is not measured.",
    "max_ai_error_rate": "No rows were sent to the AI function; the AI error rate is not measured.",
}


def inspire_not_applicable_text(names):
    if names is None:
        return ""
    if isinstance(names, str) or not isinstance(names, (list, tuple, set, frozenset)):
        raise ValueError(f"inspire_write_result takes not_applicable=[<threshold widgets>], such as not_applicable=['min_topk_stability'], "
                         f"not {names!r}")
    named = [str(name) for name in names]
    unknown = [name for name in named if name not in INSPIRE_NOT_APPLICABLE_NOTES]
    if unknown:
        raise ValueError(f"inspire_write_result takes not_applicable= threshold widgets from "
                         f"{', '.join(sorted(INSPIRE_NOT_APPLICABLE_NOTES))}, not {unknown[0]!r}")
    return " ".join(dict.fromkeys(INSPIRE_NOT_APPLICABLE_NOTES[name] for name in named))


def inspire_ratio(numerator, denominator):
    """A rate that is undefined when its denominator is 0 (no candidate alerts, no positives, no popular hit, no rows) is None:
    not applicable, never 0 and never NaN. A NaN in either number raises, because that is a failed measure, not a zero."""
    top, bottom = float(numerator), float(denominator)
    if not (_inspire_math.isfinite(top) and _inspire_math.isfinite(bottom)):
        raise ValueError(f"inspire_ratio takes finite numbers, not {numerator!r} over {denominator!r}: a NaN is a failed measure, not a zero")
    return None if bottom == 0 else top / bottom


def inspire_topk_stability(first, second, k):
    """Share of the first fit's top k alerts that the second fit also puts in its top k, from 0 to 1. k is the smallest of the alert cap
    and the two fits' alert counts. With k 0 there are no candidate alerts, so the agreement is not applicable: None."""
    top = int(k)
    inspire_record_denominator("candidate_alerts", max(top, 0))
    if top <= 0:
        return None
    return inspire_ratio(len(set(list(first)[:top]) & set(list(second)[:top])), top)


def inspire_backtest_origins(periods, horizon, season, folds=3) -> list:
    """The training lengths of the rolling-origin backtest folds the history fits, latest fold first, at most folds of them. Fold k tests on
    the horizon periods that end k horizons before the newest period and trains on every period before them; a fold needs at least two
    seasonal cycles of training. Folds do not overlap, so fewer than folds means this history cannot be backtested that many times."""
    periods, horizon, season, folds = (int(v) for v in (periods, horizon, season, folds))
    if min(periods, horizon, season, folds) < 1:
        raise ValueError(f"inspire_backtest_origins takes periods, horizon, season and folds of 1 or more, not {periods!r}, "
                         f"{horizon!r}, {season!r}, {folds!r}")
    origins = []
    for k in range(folds):
        train = periods - horizon * (k + 1)
        if train < 2 * season:
            break
        origins.append(train)
    return origins


def inspire_backtest_shortfall(origins, folds, periods, horizon, season) -> str:
    """Why the forecast cannot be checked on folds backtest folds, as the plain sentence the run reports: empty when the history fits them all."""
    fit = len(origins or [])
    if fit >= int(folds):
        return ""
    return (f"the history fits {fit} of {folds} backtest folds: each series has {periods} periods, and a fold trains on {2 * int(season)} periods "
            f"(two seasonal cycles) before testing on the next {horizon}, so the forecast cannot be checked on {folds} folds yet")


def inspire_accepted(measured):
    """The acceptance verdict over the applicable criteria. measured maps each threshold widget (min_ or max_) to the number the run
    measured, or to None when that criterion does not apply: None is excluded from acceptance, never 0, never NaN. Returns
    (accepted, missed), where missed holds only the applicable criteria the run did not meet, as {widget: measured}, or None."""
    if not isinstance(measured, dict) or not measured:
        raise ValueError(f"inspire_accepted takes {{<threshold widget>: <measured value or None>}}, such as "
                         f"{{'min_topk_stability': 0.7}}, not {measured!r}")
    missed = {}
    for name, value in measured.items():
        if not str(name).startswith(("min_", "max_")):
            raise ValueError(f"inspire_accepted takes threshold widgets named min_ or max_, not {name!r}")
        if value is None:
            continue
        try:
            needs = float(inspire_widget(str(name), ""))
        except ValueError:
            raise ValueError(f"inspire_accepted needs a {name} widget holding a number, so Inspire cannot show the bar") from None
        if isinstance(value, (str, bytes, bool)) or not _inspire_math.isfinite(float(value)):
            raise ValueError(f"inspire_accepted takes a number, or None when the criterion does not apply, not {value!r}")
        number = float(value)
        if (number >= needs) if str(name).startswith("min_") else (number <= needs):
            continue
        missed[str(name)] = number
    return (not missed), (missed or None)


INSPIRE_DENOMINATOR_RULES = {
    "min_topk_stability": ("candidate_alerts", 1), "min_auc": ("positives", 1), "min_auc_gain": ("positives", 1),
    "min_pr_auc_lift": ("positives", 1), "min_c_index": ("comparable_pairs", 1), "min_precision_lift": ("popular_hits", 1),
    "min_silhouette": ("segments", 2), "min_bootstrap_ari": ("segments", 2), "min_segment_share": ("segments", 2),
    "max_mase": ("naive_error", 1e-12), "max_ai_error_rate": ("ai_rows", 1),
}
INSPIRE_DENOMINATORS = {}


def inspire_record_denominator(key, count):
    """Records, for this run, the count a criterion's denominator came to. The helper that measures the criterion records it (top-k
    stability records its k); the write judges the criterion by this count and never by the number Genie wrote."""
    if key not in {rule[0] for rule in INSPIRE_DENOMINATOR_RULES.values()}:
        raise ValueError(f"inspire_record_denominator takes one of {', '.join(sorted({r[0] for r in INSPIRE_DENOMINATOR_RULES.values()}))}, "
                         f"not {key!r}")
    value = float(count)
    if not _inspire_math.isfinite(value) or value < 0:
        raise ValueError(f"inspire_record_denominator takes a finite count of 0 or more, not {count!r}")
    INSPIRE_DENOMINATORS[key] = value
    return count


def inspire_recorded_denominator(name):
    rule = INSPIRE_DENOMINATOR_RULES.get(str(name))
    return INSPIRE_DENOMINATORS.get(rule[0]) if rule else None


def inspire_apply_denominators(accepted, missed, not_applicable):
    """Judges each registry criterion by the denominator this run recorded, never by the number written. A criterion whose recorded
    denominator is below its minimum does not apply: it leaves missed and joins not_applicable, even when Genie wrote 0 or NaN. One
    above its minimum keeps its failure. A criterion with no recorded denominator keeps its failure, so the run stays not reliable.
    Claiming not applicable where the denominator is not recorded, or says it applies, raises. Returns (accepted, missed, not_applicable)."""
    claimed = [str(name) for name in (not_applicable or [])]
    inspire_not_applicable_text(claimed)
    for name in claimed:
        count, minimum = inspire_recorded_denominator(name), INSPIRE_DENOMINATOR_RULES[name][1]
        if count is None:
            raise ValueError(f"not_applicable= lists {name}, but its denominator is not recorded this run, so Inspire cannot show the "
                             "criterion does not apply: the helper that measures it must record its denominator")
        if count >= minimum:
            raise ValueError(f"not_applicable= lists {name}, but the denominator recorded for it is {inspire_number_text(count)}, so the "
                             "criterion applies and must be measured")
    if accepted is not False or not missed:
        return accepted, missed, (claimed or None)
    named, kept = list(claimed), {}
    for name, value in missed.items():
        count = inspire_recorded_denominator(name)
        if count is not None and count < INSPIRE_DENOMINATOR_RULES[str(name)][1]:
            named.append(str(name))
        else:
            kept[name] = value
    return (False if kept else True), (kept or None), (named or None)


def inspire_write_result(frame, result_table, spark_session=None, out=print, nothing_to_act=None, accepted=None, passed=None,
                         missed=None, not_applicable=None, why=None):
    if accepted is not None and not isinstance(accepted, bool):
        raise ValueError(f"inspire_write_result takes accepted=True or accepted=False, not {accepted!r}: True when the model whose "
                         "rows the run writes passed acceptance")
    accepted, missed, not_applicable = inspire_apply_denominators(accepted, missed, not_applicable)
    not_applicable_text = inspire_not_applicable_text(not_applicable)
    missed_text = inspire_missed_text(missed, accepted)
    if missed_text and why:
        missed_text = f"{missed_text.rstrip('.')}. {str(why)[:1].upper()}{str(why)[1:]}"
    nothing_to_act = missed_text or nothing_to_act or not_applicable_text
    if passed is not None and (isinstance(passed, bool) or not isinstance(passed, int) or passed < 0):
        raise ValueError(f"inspire_write_result takes passed=<a whole number of 0 or more>, not {passed!r}: how many rows pass the "
                         "decision test")
    session = spark_session if spark_session is not None else globals()["spark"]
    if not hasattr(frame, "write"):
        frame = session.createDataFrame(frame)
    inspire_require_plain_columns(frame, f"the result frame for {result_table}")
    inspire_require_no_python_repr(frame, f"the result frame for {result_table}")
    for field in frame.schema.fields:
        if field.dataType.typeName() in ("void", "null"):
            frame = frame.withColumn(field.name, frame[field.name].cast("string"))
    catalog, schema, name = inspire_table_parts(result_table)
    table, stage = f"{catalog}.{schema}.{name}", f"{catalog}.{schema}.{name}_inspire_stage"
    names = {f.name.lower() for f in frame.schema.fields}
    ai = inspire_ai_run_config(spark_session=session) if INSPIRE_AI_FACTS in names else None
    value = inspire_value_run_config(names)
    if value is not None:
        value["accepted"] = accepted
        value["passed"] = passed
        value[INSPIRE_PREVIEW_SAMPLE_COLUMN] = inspire_preview_sample_entities()
    preview = inspire_is_preview(inspire_widget("run_profile", "full"))
    frame.write.mode("overwrite").option("overwriteSchema", "true").saveAsTable(inspire_table_name(stage))
    try:
        return inspire_merge_run(session, table, stage, out=out, ai=ai,
                                 ai_cap=INSPIRE_AI_PREVIEW_ROWS if preview else INSPIRE_AI_AUTO_ROWS, value=value,
                                 nothing_to_act=nothing_to_act, accepted=accepted)
    finally:
        session.sql(f"DROP TABLE IF EXISTS {inspire_table_name(stage)}")


def inspire_require_no_python_repr(frame, what):
    """REQ-249: a text value that reads as a Python or pandas object dump never reaches a results table. Genie's code formatted an object
    (a method or a Series) instead of its value, so the run fails here before the write and Genie repairs the code."""
    strings = [f.name for f in frame.schema.fields if f.dataType.typeName() == "string"]
    if not strings:
        return
    lit = inspire_sql_literal(INSPIRE_PYTHON_REPR)
    row = frame.selectExpr(*[f"max(CAST({inspire_column(name)} RLIKE {lit} AS INT)) AS c{i}" for i, name in enumerate(strings)]).first()
    odd = [name for i, name in enumerate(strings) if row[i]]
    if odd:
        raise ValueError(f"{what} has {inspire_plural(len(odd), 'column')} whose text {INSPIRE_REPR_FAULT_TEXT}, not a value: "
                         f"{', '.join(odd[:5])}{' and more' if len(odd) > 5 else ''}. Build each text column from the values it names "
                         "(call the method, or format the value), then write again")


def inspire_require_no_python_repr_text(text, what):
    """REQ-249: the same refusal for one string Python already holds (a summary), before it is merged. No Spark frame is built for it."""
    if _inspire_re.search(INSPIRE_PYTHON_REPR, str(text or "")):
        raise ValueError(f"{what} {INSPIRE_REPR_FAULT_TEXT}, not a value: write the text from the values it names (call the method, "
                         "or format the value), then write again")


def inspire_require_plain_columns(frame, what):
    odd = [f.name for f in frame.schema.fields if not inspire_shell_facts.IDENT.match(str(f.name))]
    if odd:
        raise ValueError(f"{what} has {inspire_plural(len(odd), 'column')} the App cannot read or write by name: "
                         f"{', '.join(repr(n) for n in odd[:5])}{' and more' if len(odd) > 5 else ''}. Select each one with AS and a "
                         "name of letters A to Z, digits and underscores that starts with a letter or underscore, up to 63 "
                         "characters, such as AS order_date for a column named Order Date")


def inspire_write_companion(frame, result_table, suffix, columns, spark_session=None, out=print):
    if not _inspire_re.fullmatch(r"__[a-z][a-z_]{0,40}", str(suffix or "")):
        raise ValueError(f"inspire_write_companion takes the suffix of a companion table, such as '__cases', not {suffix!r}")
    wanted = [str(c) for c in (columns or ())]
    if not wanted:
        raise ValueError("inspire_write_companion takes the columns the companion table must hold, as the instruction lists them")
    session = spark_session if spark_session is not None else globals()["spark"]
    if not hasattr(frame, "write"):
        frame = session.createDataFrame(frame)
    catalog, schema, name = inspire_table_parts(result_table)
    table = inspire_shell_facts.companion(result_table, suffix)
    inspire_require_plain_columns(frame, f"the companion frame for {table}")
    inspire_require_no_python_repr(frame, f"the companion frame for {table}")
    have = {f.name.lower() for f in frame.schema.fields}
    missing = [c for c in wanted if c.lower() not in have]
    if missing:
        raise ValueError(f"the companion table {table} needs the columns {', '.join(missing)}, which the frame does not hold; the App "
                         "page reads them by these names")
    for field in frame.schema.fields:
        if field.dataType.typeName() in ("void", "null"):
            frame = frame.withColumn(field.name, frame[field.name].cast("string"))
    rows = frame.count()
    if not rows:
        raise ValueError(f"the companion table {table} holds no rows; when the data cannot fill it, skip it and say why")
    frame.write.mode("overwrite").option("overwriteSchema", "true").saveAsTable(inspire_table_name(table))
    out(f"[COMPANION-WRITE FIRED] {table}: {inspire_plural(rows, 'row')}")
    inspire_share_outputs(session, f"{catalog}.{schema}.{name}", out)
    return rows


def inspire_record_stop(run, outcome, spark_session=None, out=print):
    if not isinstance(outcome, dict) or outcome.get("status") != "stopped":
        return None
    session = spark_session if spark_session is not None else globals().get("spark")
    table = inspire_widget("result_table", "") or str((run or {}).get("result_table") or "")
    use_case_id = str((run or {}).get("use_case_id") or "")
    if session is None or not table or not use_case_id:
        return None
    reason = inspire_with_cause(" ".join(str(outcome.get("stop_reason") or "").split()) or INSPIRE_NOT_RELIABLE_STOPPED)
    run_id = str(outcome.get("run_id") or "").strip() or f"stopped-{_inspire_uuid.uuid4()}"
    if inspire_no_data_reason(reason):
        return inspire_record_no_data(session, table, use_case_id, run_id, reason, out)
    text = inspire_sql_literal(reason)
    try:
        inspire_ensure_summary(session, table)
        latest = inspire_rows(session.sql(
            f"SELECT {', '.join(inspire_column(name) for name in INSPIRE_STOP_CARRIED)} FROM "
            f"{inspire_table_name(inspire_ai_summary_table(table))} WHERE use_case_id = {inspire_sql_literal(use_case_id)} "
            "ORDER BY updated_at DESC NULLS LAST LIMIT 1"))
        carried = {name: inspire_sql_number(value) if isinstance(value, (int, float)) and not isinstance(value, bool)
                   else inspire_sql_literal(value) for name, value in (latest[0] if latest else {}).items()
                   if name in INSPIRE_STOP_CARRIED and value is not None}
        session.sql(inspire_ai_summary_merge_sql(table, use_case_id, run_id, dict(carried, not_reliable=text, nothing_to_act=text)))
    except Exception as error:
        out(f"Not reliable yet: run {run_id} stopped, but Inspire could not record why in {inspire_ai_summary_table(table)}: "
            f"{inspire_redact_text(str(error), set(), '')[:300]}")
        return None
    out(f"Not reliable yet: run {run_id} stopped before it wrote any record, so the App, the alert and the weekly email show the "
        f"use case as not reliable yet: {reason}")
    return reason


def inspire_record_no_data(session, table, use_case_id, run_id, reason, out=print):
    try:
        inspire_create_empty_core_result(session, table, out)
        inspire_ensure_summary(session, table)
        session.sql(inspire_ai_summary_merge_sql(table, use_case_id, run_id, {"nothing_to_act": inspire_sql_literal(INSPIRE_NO_DATA_YET)}))
    except Exception as error:
        out(f"{INSPIRE_NO_DATA_YET} Run {run_id} found no rows to read, but Inspire could not record it in "
            f"{inspire_ai_summary_table(table)}: {inspire_redact_text(str(error), set(), '')[:300]}")
        return None
    out(f"{INSPIRE_NO_DATA_YET} Run {run_id} found no rows to read ({reason}), so the App, the alert and the weekly email show "
        "the use case as waiting for data")
    return INSPIRE_NO_DATA_YET

# The one value calculator and spec builder: the agent execs this source and every generated notebook inlines it,
# so the card estimate and the run price a spec with the same code.
def inspire_money_short(amount) -> str:
    amount = float(amount)
    for size, suffix in ((1e9, "B"), (1e6, "M"), (1e3, "K")):
        if abs(amount) >= size:
            return f"{amount / size:,.1f}".rstrip("0").rstrip(".") + suffix
    return f"{amount:,.0f}"


INSPIRE_VALUE_MECHANISM_BUCKET = {"revenue": "cash", "cost": "cash", "working_capital": "cash", "risk": "cash",
                                  "capacity": "capacity", "non_financial": "non_financial"}
INSPIRE_VALUE_SOURCES = ("measured", "owner", "library", "exploratory")
INSPIRE_VALUE_TIERS = ("S", "M", "L")
INSPIRE_VALUE_SCENARIOS = ("low", "base", "high")


def _inspire_value_num(item) -> bool:
    return isinstance(item, (int, float)) and not isinstance(item, bool) and item == item and abs(item) != float("inf")


def _inspire_value_problems(spec: dict) -> tuple:
    problems, ranges, tiers = [], {}, None
    mechanism, stock = spec.get("mechanism"), spec.get("flow_or_stock")
    bucket = INSPIRE_VALUE_MECHANISM_BUCKET.get(mechanism)
    if bucket is None:
        problems.append("mechanism_unknown")
    if stock not in ("flow", "stock"):
        problems.append("flow_or_stock_unknown")
    horizon = spec.get("horizon_years")
    if not (_inspire_value_num(horizon) and float(horizon).is_integer() and 1 <= horizon <= 10):
        problems.append("horizon_invalid")
    measured = spec.get("measured") or {}
    if not (_inspire_value_num(measured.get("exposure")) and measured["exposure"] >= 0):
        problems.append("exposure_invalid")
    unit = measured.get("unit")
    if bucket in ("capacity", "non_financial") and not (isinstance(unit, str) and unit.strip()):
        problems.append("unit_missing")
    if mechanism == "working_capital" and stock == "flow":
        problems.append("working_capital_needs_stock")
    sourced = spec.get("sourced") or {}
    names = ["effect", "adoption", "realization"] + (["margin"] if mechanism == "revenue" else []) \
        + (["carrying_cost_rate"] if mechanism == "working_capital" else [])
    for name in names:
        item = sourced.get(name)
        if item is None:
            if name == "effect":
                problems.append("effect_missing")
            continue
        if item.get("source") not in INSPIRE_VALUE_SOURCES:
            problems.append(f"{name}_source_missing")
        trio = [item.get(key) for key in INSPIRE_VALUE_SCENARIOS]
        if not all(_inspire_value_num(part) for part in trio):
            problems.append(f"{name}_missing")
        elif not all(0 <= part <= 1 for part in trio):
            problems.append(f"{name}_out_of_range")
        elif not trio[0] <= trio[1] <= trio[2]:
            problems.append(f"{name}_order")
        else:
            ranges[name] = dict(zip(INSPIRE_VALUE_SCENARIOS, trio))
    cost = spec.get("cost")
    if cost is not None:
        held = (cost or {}).get("tiers") or {}
        valid = cost.get("source") in INSPIRE_VALUE_SOURCES and all(
            isinstance(held.get(tier), dict) and all(_inspire_value_num(held[tier].get(part)) and held[tier][part] >= 0
                                                     for part in ("implementation", "recurring_per_year"))
            for tier in INSPIRE_VALUE_TIERS)
        if valid:
            tiers = {tier: (held[tier]["implementation"], held[tier]["recurring_per_year"]) for tier in INSPIRE_VALUE_TIERS}
        else:
            problems.append("cost_invalid")
    return problems, ranges, tiers, bucket


INSPIRE_VALUE_RANK = {"low": 0, "medium": 1, "high": 2}
INSPIRE_VALUE_SOURCE_CONFIDENCE = {"measured": "high", "owner": "medium", "library": "medium", "exploratory": "low"}
INSPIRE_VALUE_COST_LABEL = {"measured": "measured", "owner": "assumption", "library": "assumption", "exploratory": "exploratory"}
INSPIRE_VALUE_LIB_PARAMETERS = ("effect", "adoption", "realization", "margin", "carrying_cost_rate")


def _inspire_value_lib_confidence(spec: dict, ranges: dict, flags: list, has_cost: bool) -> tuple:
    """Baseline, effect and cost confidence, and the gate over them. A gate, never a multiplier: no number reads this."""
    lowest = lambda levels: min(levels, key=INSPIRE_VALUE_RANK.get)
    sourced = spec.get("sourced") or {}
    levels = [INSPIRE_VALUE_SOURCE_CONFIDENCE[sourced[name]["source"]] for name in ranges if name in sourced]
    levels += ["low" for name in ("adoption", "realization") if f"{name}_assumed_full" in flags]
    held = (spec.get("measured") or {}).get("confidence")
    baseline = held if held in INSPIRE_VALUE_RANK else "low" if "period_missing" in flags else "high"
    cost = spec.get("cost") if has_cost else None
    confidence = {"baseline": baseline, "effect": lowest(levels), "cost": INSPIRE_VALUE_SOURCE_CONFIDENCE[cost["source"]] if cost else None}
    level = lowest([confidence["baseline"], confidence["effect"], confidence["cost"] or "low"])
    gate = {"level": level, "headline": level != "low",
            "weakest": sorted(name for name, value in confidence.items() if (value or "low") == level)}
    return confidence, gate, ({"source": cost["source"], "label": INSPIRE_VALUE_COST_LABEL[cost["source"]], "editable": True,
                               "tiers": cost["tiers"]} if cost else None)


def inspire_value_calculate(spec: dict) -> dict:
    """The one value calculator: joint low/base/high scenarios, gross and net per S/M/L cost tier, break-even effect and the 1% line.
    Pure: reads only the spec. Cash, capacity and non-financial values stay in separate buckets and are never added."""
    spec = spec if isinstance(spec, dict) else {}
    problems, ranges, tiers, bucket = _inspire_value_problems(spec)
    if problems:
        return {"ok": False, "problems": sorted(problems), "flags": [], "bucket": bucket, "unit": None, "period": None,
                "annual_exposure": None, "balance": None, "one_percent": None, "scenarios": None, "break_even_effect": None,
                "confidence": None, "gate": None, "cost": None}
    mechanism, stock, horizon = spec["mechanism"], spec["flow_or_stock"] == "stock", int(spec["horizon_years"])
    measured, exposure = spec["measured"], float(spec["measured"]["exposure"])
    flags = []
    for name in ("adoption", "realization"):
        if name not in ranges:
            ranges[name] = {key: 1.0 for key in INSPIRE_VALUE_SCENARIOS}
            flags.append(f"{name}_assumed_full")
    cash = bucket == "cash"
    currency = spec.get("currency")
    has_currency = isinstance(currency, str) and _inspire_re.fullmatch(r"[A-Z]{3}", currency) is not None
    if cash and not has_currency:
        flags.append("currency_missing")
    if mechanism == "revenue" and "margin" not in ranges:
        flags.append("margin_missing")
    if mechanism == "working_capital" and "carrying_cost_rate" not in ranges:
        flags.append("carrying_cost_missing")
    if tiers is None:
        flags.append("cost_missing")
    unit = (currency if has_currency else None) if cash else measured.get("unit")
    annual = balance = None
    days = measured.get("period_days")
    if stock:
        balance, period = exposure, "balance"
        if mechanism != "working_capital":
            flags.append("stock_not_annual")
    elif _inspire_value_num(days) and days > 0:
        annual, period = exposure * 365 / days, "year"
    else:
        period = "data period"
        flags.append("period_missing")
    amount = annual if annual is not None else balance if balance is not None else exposure
    one_percent = None if cash and not has_currency else {
        "value": 0.01 * amount, "unit": unit, "per": period,
        "meaning": f"1% of the observed {'balance' if stock else 'exposure'} per {period}, shown as a size reference, not a forecast"}
    priced = cash and has_currency and not {"margin_missing", "carrying_cost_missing", "stock_not_annual", "period_missing"} & set(flags)
    scenarios = {}
    for key in INSPIRE_VALUE_SCENARIOS:
        share = ranges["effect"][key] * ranges["adoption"][key] * ranges["realization"][key]
        row = {"change": None, "gross_annual": None, "gross_horizon": None, "one_off_release": None, "net": None}
        if mechanism == "working_capital":
            row["one_off_release"] = balance * share if has_currency else None
            if priced:
                row["gross_annual"] = balance * share * ranges["carrying_cost_rate"][key]
        elif "stock_not_annual" not in flags:
            row["change"] = amount * share
            if priced:
                row["gross_annual"] = row["change"] * (ranges["margin"][key] if mechanism == "revenue" else 1.0)
        if row["gross_annual"] is not None:
            row["gross_horizon"] = row["gross_annual"] * horizon
            if tiers:
                row["net"] = {tier: row["gross_horizon"] - tiers[tier][0] - tiers[tier][1] * horizon for tier in INSPIRE_VALUE_TIERS}
        scenarios[key] = row
    break_even = None
    if priced and tiers:
        multiplier = ranges["margin"]["base"] if mechanism == "revenue" else \
            ranges["carrying_cost_rate"]["base"] if mechanism == "working_capital" else 1.0
        divisor = amount * ranges["adoption"]["base"] * ranges["realization"]["base"] * multiplier * horizon
        break_even = {tier: ((tiers[tier][0] + tiers[tier][1] * horizon) / divisor if divisor > 0 else None) for tier in INSPIRE_VALUE_TIERS}
    if scenarios["base"]["net"] and any(net < 0 for net in scenarios["base"]["net"].values()):
        flags.append("net_negative")
    confidence, gate, cost = _inspire_value_lib_confidence(spec, ranges, flags, tiers is not None)
    return {"ok": True, "problems": [], "flags": sorted(flags), "bucket": bucket, "unit": unit, "period": period,
            "annual_exposure": annual, "balance": balance, "one_percent": one_percent, "scenarios": scenarios,
            "break_even_effect": break_even, "confidence": confidence, "gate": gate, "cost": cost}


def _inspire_value_proposal_numbers(node, path: str = "") -> list:
    """Dotted paths of every int/float in an LLM answer (a bool is not a number)."""
    if isinstance(node, bool):
        return []
    if isinstance(node, (int, float)):
        return [path]
    if isinstance(node, dict):
        return [found for key, inner in node.items() for found in _inspire_value_proposal_numbers(inner, f"{path}.{key}" if path else str(key))]
    if isinstance(node, list):
        return [found for index, inner in enumerate(node) for found in _inspire_value_proposal_numbers(inner, f"{path}.{index}" if path else str(index))]
    return []


def inspire_value_proposal_problems(proposal, columns=None, entries=None) -> list:
    """Sorted problem codes of an LLM value proposal. Any number anywhere in the answer rejects it: the model names, code measures."""
    if not isinstance(proposal, dict):
        return ["proposal_not_object"]
    problems = {f"llm_number:{path}" for path in _inspire_value_proposal_numbers(proposal)}
    mechanism, flow, unit = proposal.get("mechanism"), proposal.get("flow_or_stock"), proposal.get("unit")
    bucket = INSPIRE_VALUE_MECHANISM_BUCKET.get(mechanism) if isinstance(mechanism, str) else None
    if bucket is None:
        problems.add("mechanism_unknown")
    if flow not in ("flow", "stock"):
        problems.add("flow_or_stock_unknown")
    if bucket in ("capacity", "non_financial") and not (isinstance(unit, str) and unit.strip()):
        problems.add("unit_missing")
    sql = proposal.get("baseline_sql")
    if not (isinstance(sql, str) and _inspire_re.fullmatch(INSPIRE_KPI_QUERY, sql)):
        problems.add("baseline_sql_not_select")
    elif columns is not None:
        tables = {str(column).rsplit(".", 1)[0].lower() for column in columns}
        if not any(_inspire_re.search(r"(?<![\w.])" + _inspire_re.escape(table) + r"(?![\w])", sql.lower()) for table in tables):
            problems.add("baseline_sql_no_table")
    named = proposal.get("columns")
    if columns is not None and isinstance(named, dict):
        known = {str(column).lower() for column in columns}
        problems.update(f"column_unknown:{slot}" for slot, column in named.items() if isinstance(column, str) and column.lower() not in known)
    for item in proposal.get("assumptions") if isinstance(proposal.get("assumptions"), list) else []:
        if not isinstance(item, dict):
            continue
        if isinstance(item.get("slot"), str) and item["slot"] not in INSPIRE_VALUE_LIB_PARAMETERS:
            problems.add(f"slot_unknown:{item['slot']}")
        if entries is not None and isinstance(item.get("entry"), str) and item["entry"] not in entries:
            problems.add(f"assumption_unknown:{item['entry']}")
    return sorted(problems)


def inspire_value_spec_from_proposal(proposal: dict, measured: dict, entries: dict, currency, horizon_years: int = 1, cost=None) -> dict:
    """The one place a value spec is built. Reads only the text keys of the proposal; every number comes from `measured` (code-run
    SQL), the sourced ranges of `entries` and `cost`."""
    given = measured if isinstance(measured, dict) else {}
    held = {key: given[key] for key in ("exposure", "period_days", "population", "baseline", "unit") if key in given}
    unit = proposal.get("unit")
    if isinstance(unit, str) and unit.strip():
        held["unit"] = unit
    sourced = {}
    for item in proposal.get("assumptions") or []:
        entry = (entries or {}).get(item.get("entry")) if isinstance(item, dict) and isinstance(item.get("entry"), str) else None
        if isinstance(entry, dict) and item.get("slot") in INSPIRE_VALUE_LIB_PARAMETERS:
            sourced[item["slot"]] = {key: held_value for key, held_value in entry.items() if key != "label"}
    spec = {"mechanism": proposal.get("mechanism"), "currency": currency, "flow_or_stock": proposal.get("flow_or_stock"),
            "horizon_years": horizon_years, "measured": held, "sourced": sourced}
    if cost is not None:
        spec["cost"] = cost
    return spec


def inspire_value_formula_spec(value: dict, formula, measured: dict, entries=None, horizon_years: int = 1):
    """The per-record template's spec: built with the same builder from the run's proposal; its constants stay in the per-record expression."""
    proposal = (value or {}).get("proposal")
    if not formula or inspire_value_proposal_problems(proposal):
        return None
    return inspire_value_spec_from_proposal(proposal, measured, entries or {}, value.get("currency"), horizon_years)
